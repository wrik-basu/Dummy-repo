# This reconstructs the THREE final joined datasets exactly as the collation

#   B) Regmap  -> the exact pandas join chain (join4 / join5 / join6b).

# Frame profiling is pandas (frames are small once joined).

# %% [markdown]
# ## The EDA driver — one function, run on every joined frame
# For a given frame it reports: shape/memory, dtype mix, per-column
# null+distinct profile, grain/duplicate check, datetime spans, categorical
# distributions, free-text length stats. Returns the profile table so you
# can keep it; prints the rest for quick reading.
# ==========================================================================

# %%
def explore(df: pd.DataFrame, name: str,
            grain_keys: list[str] | None = None,
            datetime_cols: list[str] | None = None,
            categorical_cols: list[str] | None = None,
            text_cols: list[str] | None = None,
            cat_card_threshold: int = 50,
            top_n: int = 15) -> pd.DataFrame:

    line = "=" * 78
    print(f"\n{line}\n {name}\n{line}")

    # --- shape / memory ----------------------------------------------------
    mem_mb = df.memory_usage(deep=True).sum() / 1024**2
    print(f"shape: {df.shape[0]:,} rows x {df.shape[1]} cols   |   memory: {mem_mb:,.1f} MB")

    # --- dtype mix ---------------------------------------------------------
    print("\ndtype counts:")
    print(df.dtypes.value_counts().to_string())

    # --- per-column null + distinct profile --------------------------------
    n = len(df)
    profile = pd.DataFrame({
        "dtype":    df.dtypes.astype(str),
        "nulls":    df.isna().sum(),
        "null_pct": (100 * df.isna().mean()).round(2),
        "distinct": df.nunique(dropna=True),
    })
    profile["distinct_pct"] = (100 * profile["distinct"] / n).round(2)
    profile.index.name = "column"
    print("\ncolumn profile:")
    print(profile.to_string())

    # --- grain / duplicate check ------------------------------------------
    if grain_keys:
        distinct_keys = df.drop_duplicates(subset=grain_keys).shape[0]
        dup_rows = n - distinct_keys
        print(f"\ngrain check on {grain_keys}:")
        print(f"  rows={n:,}  distinct_key_combos={distinct_keys:,}  "
              f"duplicate_rows={dup_rows:,}  is_unique_grain={n == distinct_keys}")
        if dup_rows:
            worst = (df.groupby(grain_keys).size()
                       .sort_values(ascending=False).head(5))
            print("  worst fan-out (top 5 key combos by row count):")
            print(worst.to_string())

    # --- datetime spans ----------------------------------------------------
    if datetime_cols:
        print("\ndatetime spans:")
        for c in datetime_cols:
            if c in df.columns:
                s = pd.to_datetime(df[c], errors="coerce")
                print(f"  {c}: {s.min()} -> {s.max()}  (nulls={s.isna().sum():,})")

    # --- categorical distributions ----------------------------------------
    if categorical_cols is None:
        categorical_cols = [
            c for c in df.columns
            if df[c].dtype == object and df[c].nunique(dropna=True) <= cat_card_threshold
        ]
    print("\ncategorical distributions:")
    for c in categorical_cols:
        if c in df.columns:
            vc = df[c].value_counts(dropna=False).head(top_n)
            print(f"\n  [{c}]  (distinct={df[c].nunique(dropna=True)})")
            print(vc.to_string())

    # --- free-text length stats -------------------------------------------
    if text_cols:
        print("\nfree-text length (chars):")
        rows = []
        for c in text_cols:
            if c in df.columns:
                ln = df[c].astype(str).str.len().where(df[c].notna())
                rows.append({"column": c,
                             "min": ln.min(), "mean": round(ln.mean(), 1),
                             "median": ln.median(), "max": ln.max()})
        if rows:
            print(pd.DataFrame(rows).to_string(index=False))

    return profile

# %% [markdown]
# ## A. RAPID2 — reconstruct the full joined table
# Exact query from the collation notebook (all joins + both STRING_AGG
# subqueries), parametrised on jurisdiction, then concatenated across the
# three Data-Visa-approved jurisdictions into the full analytical table.
# ==========================================================================

# %%
def load_rapid2(juris: str,
                start: str = "2020-01-01",
                end: str = "2025-12-31",
                category: str = "REGDEV") -> pd.DataFrame:
    """Return the RAPID2 joined dataset for one jurisdiction (24 columns)."""
    sql = f"""
    SELECT
        r.record_id            AS RECORD_ID,
        r.orign_alert_id       AS ALERT_ID,
        H.INGEST_RULE_TYPE     AS INGEST_RULE_TYPE,
        r.jris_cde             AS RECORD_JURISDICTION_CODE,
        c.name                 AS RECORD_JURISDICTION_NAME,
        r.created_on_dtm       AS RECORD_START_DATE,
        r.updated_on_dtm       AS RECORD_END_DATE,
        r.is_actv_ind          AS ACTIVE_INDICATOR,
        r.record_cat_cde       AS RECORD_CATEGORY_CODE,
        r.stat_cde             AS RECORD_STATUS_CODE,
        b.name                 AS RECORD_STATUS,
        r.rglt_bdy_cde         AS REGULATOR_CODE,
        r.risk_stwrd_area_cde  AS RISK_STEWARD_AREA_CODE,
        e.name                 AS RISK_STEWARD_AREA_NAME,
        theme.theme_code       AS RECORD_THEME_CODES,
        theme.theme_desc       AS RECORD_THEMES,
        risk_txnmy.risk_txnmy_lvl_1 AS LEVEL_1_RISK_TAXONOMY,
        risk_txnmy.risk_txnmy_lvl_2 AS LEVEL_2_RISK_TAXONOMY,
        risk_txnmy.risk_txnmy_lvl_3 AS LEVEL_3_RISK_TAXONOMY,
        r.title                AS TITLE,
        r.summ_contnt          AS SUMM_CONTENT,
        r.summ_updt            AS SUMM_UPDATE,
        r.srce_url             AS URL_TEXT,
        r.intro                AS INTRO_TEXT
    FROM `{DATA_PROJECT}.{DATASET}.aa_hsbc_rapid2_hzn_record` r
    LEFT JOIN `{DATA_PROJECT}.{DATASET}.aa_hsbc_rapid2_reg_ref_jris` c
           ON r.jris_cde = c.jris_cde
    LEFT JOIN `{DATA_PROJECT}.{DATASET}.aa_hsbc_rapid2_reg_ref_risk_stwrd_area` e
           ON r.risk_stwrd_area_cde = e.risk_stwrd_area_cde
    LEFT JOIN `{DATA_PROJECT}.{DATASET}.aa_hsbc_rapid2_hzn_ref_stat` b
           ON r.stat_cde = b.stat_cde
    LEFT JOIN `{DATA_PROJECT}.{DATASET}.aa_hsbc_rapid2_hzn_alert` H
           ON R.ORIGN_ALERT_ID = H.ALERT_ID
    LEFT JOIN (
        SELECT t.record_id AS theme_recid,
               STRING_AGG(t.theme_cde, '|' ORDER BY t.record_id) AS theme_code,
               STRING_AGG(rt.name,     '|' ORDER BY t.record_id) AS theme_desc
        FROM `{DATA_PROJECT}.{DATASET}.aa_hsbc_rapid2_hzn_record_theme_map` t
        LEFT JOIN `{DATA_PROJECT}.{DATASET}.aa_hsbc_rapid2_hzn_ref_theme` rt
               ON t.theme_cde = rt.theme_cde
        GROUP BY t.record_id
    ) theme ON r.record_id = theme.theme_recid
    LEFT JOIN (
        SELECT x.record_id AS taxn_recid,
               STRING_AGG(x.risk_txnmy_lvl_1, '|' ORDER BY x.record_id) AS risk_txnmy_lvl_1,
               STRING_AGG(x.risk_txnmy_lvl_2, '|' ORDER BY x.record_id) AS risk_txnmy_lvl_2,
               STRING_AGG(x.risk_txnmy_lvl_3, '|' ORDER BY x.record_id) AS risk_txnmy_lvl_3
        FROM `{DATA_PROJECT}.{DATASET}.aa_hsbc_rapid2_hzn_record_risk_txnmy` x
        GROUP BY x.record_id
    ) risk_txnmy ON r.record_id = risk_txnmy.taxn_recid
    WHERE r.created_on_dtm >= '{start}'
      AND r.created_on_dtm <= '{end}'
      AND c.name = "{juris}"
      AND r.record_cat_cde = '{category}'
    """
    return bq(sql)

# ## C. Helios — reconstruct the joined table
# risk (a) + bridge (b) + L1 control (c) + L2 control (d). The notebook used
# SELECT DISTINCT * ... LIMIT 10; here we drop the LIMIT so we see the true
# joined table, keep the DISTINCT, and keep the taxonomy filter parametrised.
# ==========================================================================

# %%
def load_helios(country: str = "us",
                l1_filter: list[str] | None = ("Financial Crime",)) -> pd.DataFrame:
    l1_ctrl = f"{country}_hsbc_helios_l1_control"
    l2_ctrl = f"{country}_hsbc_helios_l2_control"
    risk    = f"{country}_hsbc_helios_risk"
    mapping = "aa_hsbc_helios_rtcl"

    where_tax = ""
    if l1_filter:
        vals = ", ".join(f"'{v}'" for v in l1_filter)
        where_tax = f"WHERE a.risk_taxonomy_l1 IN UNNEST([{vals}])"

    sql = f"""
    SELECT DISTINCT * FROM (
        SELECT
            a.library_risk_id,
            a.risk_taxonomy_l1_id, a.risk_taxonomy_l1,
            a.risk_taxonomy_l2_id, a.risk_taxonomy_l2,
            a.risk_taxonomy_l3_id, a.risk_taxonomy_l3,
            c.l1_library_control_id, c.l1_library_control_title, c.l1_library_control_description,
            d.l2_library_control_id, d.l2_library_control_title, d.l2_library_control_description,
            c.name AS l1_control_id, c.l1_control_title_alias, c.description AS l1_control_description,
            d.name AS l2_control_id, d.l2_control_title_alias, d.description AS l2_control_description
        FROM `{DATA_PROJECT}.{DATASET}.{risk}` a
        LEFT JOIN `{DATA_PROJECT}.{DATASET}.{mapping}` b
               ON a.library_risk_id = b.risk_library_id
        LEFT JOIN `{DATA_PROJECT}.{DATASET}.{l1_ctrl}` c
               ON b.l1_control_library_id = c.l1_library_control_id
        LEFT JOIN `{DATA_PROJECT}.{DATASET}.{l2_ctrl}` d
               ON b.l2_control_library_id = d.l2_library_control_id
        {where_tax}
    ) e
    WHERE e.l1_library_control_id IS NOT NULL
    """
    return bq(sql)

# %% B.1 — join helpers (reproduced exactly; validation stays commented out
#          so the reconstructed frames match downstream byte-for-byte)
def join_tables(left_df, right_df, join_keys, join_name,
                join_type="left", validation="one_to_many"):
    print(f"\nStarting {join_name}")
    output = pd.merge(left_df, right_df, how=join_type, on=join_keys,
                      suffixes=("", "_drop"))  # validate=validation
    print(f"Completed {join_name}")
    return output

def join_tables_diff_fields(left_df, right_df, join_keys1, join_keys2, join_name,
                            join_type="left", validation="one_to_many"):
    print(f"\nStarting {join_name}")
    output = pd.merge(left_df, right_df, how=join_type,
                      left_on=join_keys1, right_on=join_keys2,
                      suffixes=("", "_drop"))  # validate=validation
    print(f"Completed {join_name}")
    return output
