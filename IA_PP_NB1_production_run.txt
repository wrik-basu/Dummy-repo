# ------------------------------------------------------------------------------
# WHAT THIS NOTEBOOK DOES
#   For every regulatory alert in scope, it finds the GPPS policies and procedures the alert impacts, and the exact
#   SECTION / SUB-SECTION of each, with verbatim evidence from both texts and a one-line "change needed".
#   * POLICIES   : the AI reads EVERY policy for every alert (no search, nothing can be missed by a search step).
#   * PROCEDURES : a search step (method F: alert-text embedding + AI query plan + expanded keywords, fused by rank)
#                  shortlists the TOP 200 procedures per alert; the AI then reads each of them in full.
#   * ONE AI CALL per (alert, document). No second "audit" call in production. The audit (independent AI reviewer,
#     SME pack) and the management storyboard are in NOTEBOOK 2, which reads this notebook's output folder.
#
# DECISION RULES (agreed - see Alert_to_Policy_Procedure_Approach_Decisions)
#   1. LINKED     = the AI says LINKED + link_score >= 5 + BOTH verbatim quotes are found in the texts by code.
#   2. JURISDICTION: every document is labelled GLOBAL / LOCAL (this country) / LOCAL (another country) from GPPS.
#                  A US alert can impact a GLOBAL and a US LOCAL document - never another country's LOCAL one,
#                  unless that country adopts the US requirement -> such links are SCOPE_REVIEW (SME), not impacts.
#   3. FRAMEWORK / PROCESS documents (they govern HOW any regulatory change, read-across or regulatory report is
#      handled - list in SETTINGS D): impacted ONLY if the alert text clearly requires that PROCESS to be updated,
#      changed or rectified. They are read with the same prompt + a FRAMEWORK RULE and must answer
#      framework_change = YES. Every other (BUSINESS) document is read with the validated prompt, unchanged.
#   4. BANDS (one call, a rule on the call's own link_score - SETTINGS D):
#        IMPACTED        link_score >= 8   (reference judge agreed with ~80% on the US development run)
#        POSSIBLE_IMPACT link_score 5-7    (~60% agreement -> SME review queue; nothing is thrown away)
#        SCOPE_REVIEW    linked to another country's LOCAL document
#        NOT_IMPACTED    everything else
#
# HOW TO RUN (any country)
#   1. CELL 0  : your gateway settings (staff id, use case, base url, model) - copy from a gateway cell that works.
#   2. CELL 1  : SETTINGS A (country) and B (5 source file paths). C and D: defaults agreed on the US run.
#   3. Run CELL 0 -> 5 in order. Check: CELL 1 (files found), CELL 2 (document scope examples), CELL 4 (cost gate),
#      CELL 5 (10-pair test gate). Then CELL 6 (policies, the long one), CELL 7 (procedures), CELL 8 (results).
#   4. CELL 9 = status at any time ($0). Then NOTEBOOK 2 (audit + storyboard) with NB1_RUN_DIR = this RUN_DIR.
#   US reference: 161 alerts | 45,402 policy reads + 32,200 procedure reads | ~$150 | ~10-14 h at 24 in parallel.
#
# SAFE TO STOP / RESUME
#   Every paid step is checkpointed (query plans, embeddings, policy reads, procedure reads). Kernel > Interrupt
#   is safe: pending calls are cancelled and progress is saved. After an interrupt / kernel restart / browser
#   disconnect: run CELL 0 -> 5 again (no repeat spend - the test gate is skipped once the run has started), then
#   re-run the cell that was interrupted; finished pairs are never paid for twice.
#   Gateway login expires after ~1 h idle: if calls start failing with 401, the code logs in again by itself; if it
#   keeps failing, re-run CELL 0 and then the interrupted cell.
#
# OUTPUT (RUN_DIR)
#   ia_pp_pairs.parquet            one row per (alert, document) read - every decision, score, quote, reason
#   ia_pp_impacted_sections.parquet one row per impacted SECTION / SUB-SECTION (GPPS numbering + titles)
#   ia_pp_alert_impacts.jsonl      one JSON per alert: impacted / possible / scope-review documents with sections
#   ia_pp_results_summary.xlsx     How to read | summary | per alert | by scope | framework documents | document scope
#   production/                    the exact prompts + production_config.json used in this run

class TrustTokenCache:
    def __init__(self):
        self.token, self.expires_at, self.lock = None, 0.0, asyncio.Lock()
    async def get(self):
        now = time.time()
        if self.token and now < self.expires_at - 5: return self.token
        async with self.lock:
            now = time.time()
            if self.token and now < self.expires_at - 5: return self.token
            self.token = await asyncio.to_thread(getattr(group_ai_client, TOKEN_METHOD))
            self.expires_at = now + 300
            return self.token
    async def invalidate(self):
        async with self.lock:
            self.token, self.expires_at = None, 0.0

trust_token = TrustTokenCache()
USAGE_LOG, ERROR_LOG, CURRENT_STAGE = [], [], "setup"

async def call_llm(system, payload, schema=None, max_tokens=2500, retries=4):
    """One gateway call -> parsed JSON dict, or None if all retries fail (logged in ERROR_LOG)."""
    kw = {"model": CHAT_MODEL, "user": USE_CASE,
          "messages": [{"role": "system", "content": system},
                       {"role": "user", "content": json.dumps(payload, ensure_ascii=False)}]}
    if MODEL_FAMILY == "gemini":
        kw.update(temperature=0.0, seed=42, reasoning_effort="low", max_tokens=max_tokens)
    else:
        kw.update(max_completion_tokens=max_tokens)
    if schema: kw["response_format"] = schema
    last = None
    for attempt in range(retries):
        try:
            token = await trust_token.get()
            r = await openai_client.chat.completions.create(
                extra_headers={"Authorization": f"Bearer {token}"}, **kw)
            u = getattr(r, "usage", None)
            USAGE_LOG.append({"stage": CURRENT_STAGE, "in": getattr(u, "prompt_tokens", 0) or 0,
                              "out": getattr(u, "completion_tokens", 0) or 0, "ts": time.time()})
            txt = (r.choices[0].message.content or "").replace("```json", "").replace("```", "").strip()
            return json.loads(txt[txt.find("{"): txt.rfind("}") + 1])
        except Exception as e:
            last = e
            msg = str(e)
            if "401" in msg: await trust_token.invalidate()
            if ("response_format" in msg or "json_schema" in msg) and "response_format" in kw:
                kw.pop("response_format")
            await asyncio.sleep(min(2 ** attempt, 20))
    ERROR_LOG.append({"stage": CURRENT_STAGE, "error": repr(last)[:300], "ts": time.time()})   # logged after ALL retries
    return None

def cost_report(stage=None):
    u = pd.DataFrame(USAGE_LOG)
    if stage and len(u): u = u[u.stage == stage]
    e = sum(1 for x in ERROR_LOG if not stage or x["stage"] == stage)
    if u.empty:
        print(f"[cost{' ' + stage if stage else ''}] no successful calls | failed calls {e}")
        return {"failed_calls": e}
    c = u["in"].sum() * TOKEN_PRICE["in"] + u["out"].sum() * TOKEN_PRICE["out"]
    print(f"[cost{' ' + stage if stage else ''}] calls {len(u):,} | in {u['in'].sum():,} | out {u['out'].sum():,} | "
          f"${c:.2f} | failed calls {e}")
    return {"calls": len(u), "cost_usd": round(float(c), 2), "failed_calls": e}

res = await call_llm("Reply with JSON only.", {"task": "return {'ok': true}"}, None, 200)
print("gateway ready:", CHAT_MODEL, "| check:", res)
assert res, f"gateway check failed - last error: {ERROR_LOG[-1]['error'] if ERROR_LOG else 'none'}"

# ---- library (same GPPS parsing as the numbers notebook) ----
NA = {"", "n/a", "na", "-", "--", "none", "nan", "null"}

def N(x):
    s = str(x).strip().upper()
    return s[:-2] if re.fullmatch(r"\d+\.0", s) else s

def clean(x):
    return "" if x is None or str(x).strip().lower() in NA else str(x).strip()

def as_str(x):
    if x is None: return ""
    if isinstance(x, list): return " | ".join(str(i).strip() for i in x if i is not None and str(i).strip())
    return str(x).strip()

def as_dict(x):
    if isinstance(x, list): x = next((i for i in x if isinstance(i, dict)), {})
    return x if isinstance(x, dict) else {}

def as_list(x):
    return [i for i in x if isinstance(i, dict)] if isinstance(x, list) else []

def first(d, keys):
    for k in keys:
        v = d.get(k)
        if v not in (None, "", []): return v
    return None

def strip_html(s):
    if not isinstance(s, str) or not s: return ""
    s = re.sub(r"(?is)<!doctype.*?>|<head.*?>.*?</head>", " ", s)
    s = re.sub(r"(?s)<[^>]+>", " ", s)
    return re.sub(r"\s+", " ", html.unescape(s)).strip()

def load_json(raw):
    if isinstance(raw, dict): return raw
    try: return json.loads(raw)
    except Exception: return {}

def read_any(path):
    p = str(path)
    if p.endswith((".xlsx", ".xls")): return pd.read_excel(p)
    if p.endswith(".csv"): return pd.read_csv(p)
    try: return pd.read_parquet(p)
    except Exception:
        import pyarrow.parquet as pq
        return pq.read_table(p).replace_schema_metadata(None).to_pandas(date_as_object=True)

def pick(df, *cands):
    return next((c for c in cands if c in df.columns), None)

# ---- sub-sections: numbered bold / underlined headings inside a segment's HTML (GPPS UI numbering) ----
HEADING = re.compile(r"^\s*([A-Z]?\.?\d+(?:\.\d+){1,3})\.?\s+(.{3,150}?)\s*$")    # "2.2. Model Inputs" / "B.2.7.4 Use of ..."
MARK = "␞SUBHEAD␞"

def split_subsections(raw_html, section_no, strict_prefix):
    if not isinstance(raw_html, str) or not raw_html.strip(): return []
    found = []
    def mark(m):
        inner = strip_html(m.group(0))
        h = HEADING.match(inner)
        top = re.sub(r"^[A-Z]\.?", "", h.group(1)).split(".")[0] if h else ""
        if h and (not strict_prefix or not section_no or top == str(section_no).split(".")[0]):
            found.append((h.group(1), h.group(2).strip(" :")))
            return f" {MARK}{len(found) - 1}{MARK} "
        return m.group(0)
    marked = re.sub(r"(?is)<(strong|b|u)\b[^>]*>.*?</\1>|<span[^>]*underline[^>]*>.*?</span>", mark, raw_html)
    parts = re.split(rf"{MARK}(\d+){MARK}", strip_html(marked))
    subs, seen = [], set()
    if parts[0].strip():
        subs.append({"sub_no": f"{section_no}.0" if section_no else "", "sub_title": "(section introduction)",
                     "text": parts[0].strip()})
    for i in range(1, len(parts), 2):
        num, title = found[int(parts[i])]
        body = parts[i + 1].strip() if i + 1 < len(parts) else ""
        if (num, body[:200]) in seen: continue                     # GPPS sometimes repeats a block
        seen.add((num, body[:200]))
        subs.append({"sub_no": num, "sub_title": title, "text": body})
    return subs

# ---- one GPPS document (policy or procedure) -> sections -> sub-sections ----
KEYS = {
    "policy": dict(app="policyApplication", core=["plcyCoreId"], title=["policyTitle"],
                   status=("versn", ["versStatCde"]), top=[("policyText", "Policy Statement"), ("policyPurpose", "Purpose")],
                   seg_text=["sgmntText"], seg_title=["sgmntTitlText"], seg_disp=["sgmntDispInd"],
                   seg_no=["dispSortOrdrNum", "sortOrdrNum"], seg_reason=["sgmntReasonText"], strict=True),
    "procedure": dict(app="procedureApplication", core=["procCoreId"], title=["procedureTitle"],
                      status=("versn", ["procVersnStatCde", "versStatCde"]), top=[("procedureText", "Procedure Statement")],
                      seg_text=["procSgmntText", "sgmntText"], seg_title=["procSgmntTitlText", "sgmntTitlText"],
                      seg_disp=["procSgmntDispInd", "sgmntDispInd"], seg_no=["procDispSortOrdrNum", "procSortOrdrNum",
                      "dispSortOrdrNum", "sortOrdrNum"], seg_reason=["procSgmntReasonText", "sgmntReasonText"], strict=False),
}

def parse_doc(raw, kind):
    K = KEYS[kind]
    d = load_json(raw)
    if not d: return None
    app = as_dict(d.get(K["app"]))
    secs = []
    for key, label in K["top"]:
        o = as_dict(d.get(key))
        secs.append(dict(section_no="", title=strip_html(as_str(o.get("sgmntTitlText") or label)),
                         html=o.get("sgmntText") or "", displayed="Y", reason="", kind="top"))
    for s in as_list(app.get("segment")):
        secs.append(dict(section_no=clean(as_str(first(s, K["seg_no"]))),
                         title=strip_html(as_str(first(s, K["seg_title"]) or "Section")),
                         html=first(s, K["seg_text"]) or "",
                         displayed=(clean(as_str(first(s, K["seg_disp"]))).upper() or "Y"),
                         reason=strip_html(as_str(first(s, K["seg_reason"]))), kind="segment"))
    out, seen = [], set()
    for s in secs:
        text = strip_html(s["html"])
        if not text: continue
        h = hashlib.md5(text.encode()).hexdigest()
        if h in seen: continue                                          # duplicate segment text
        seen.add(h)
        subs = split_subsections(s["html"], s["section_no"], K["strict"]) if s["kind"] == "segment" else []
        out.append({**{k: v for k, v in s.items() if k != "html"}, "text": text,
                    "subsections": subs or [{"sub_no": "", "sub_title": "", "text": text}]})
    ver = as_dict(d.get(K["status"][0]))
    return dict(core_id=N(first(d, K["core"]) or ""), title=clean(as_str(first(d, K["title"]))),
                status=clean(as_str(first(ver, K["status"][1]))), sections=out, top_keys=sorted(d.keys()))

def load_docs(path, kind, scope):
    df = read_any(path)
    idc = pick(df, "policy_core_id", "procedure_core_id", "policy_id", "procedure_id")
    recs = []
    for _, r in df.iterrows():
        p = parse_doc(r["json_data"], kind)
        if p is None: continue
        p.update(doc_id=N(r[idc]) if idc else p["core_id"], version=r.get("version_num"), scope=scope, kind=kind)
        recs.append(p)
    out = pd.DataFrame(recs)
    n_all = len(out)
    if PUBLISHED_ONLY and len(out):
        out = out[out.status.str.upper().str.startswith("PUBL")]
    out["_v"] = pd.to_numeric(out.version, errors="coerce").fillna(-1)
    out = out.sort_values("_v", ascending=False).drop_duplicates("doc_id").drop(columns="_v").reset_index(drop=True)
    print(f"  {kind:9s} {scope:6s}: {n_all:,} rows -> {len(out):,} documents (published, latest version)")
    return out

def build_units(docs):
    """Retrieval / assessment unit = a visible numbered SUB-SECTION; a visible section without numbered
    sub-sections is one unit. Hidden segments (shown 'N' in the GPPS UI) are excluded."""
    rows = []
    for d in docs.itertuples():
        prefix = "POL" if d.kind == "policy" else "PROC"
        for s in d.sections:
            for sub in s["subsections"]:
                rows.append(dict(
                    unit_id=f"{prefix}-{d.doc_id}|{s['section_no'] or s['title'][:20]}|{sub['sub_no'] or '-'}",
                    doc_type=d.kind, doc_id=f"{prefix}-{d.doc_id}", doc_title=d.title, scope=d.scope,
                    section_no=s["section_no"], section_title=s["title"], subsection_no=sub["sub_no"],
                    subsection_title=sub["sub_title"], hidden=s["displayed"] == "N", hidden_reason=s["reason"],
                    level="sub-section" if sub["sub_no"] else "section", text=sub["text"], chars=len(sub["text"])))
    u = pd.DataFrame(rows)
    return u


def clean_words(s):
    return re.sub(r"[^a-z0-9]+", " ", str(s or "").lower()).strip()
def quote_found(quote, text):
    """Verbatim (case / punctuation-insensitive); long quotes may match on 8-word ends. Same as every run."""
    q = clean_words(quote)
    if len(q) < 12 or q in ("null", "none"): return False
    t = clean_words(text)
    if q in t: return True
    w = q.split()
    return len(w) >= 6 and (" ".join(w[:8]) in t or " ".join(w[-8:]) in t)
def wilson(k, n, z=1.96):
    if n == 0: return (np.nan, np.nan)
    p = k / n; d = 1 + z * z / n; c = (p + z * z / (2 * n)) / d
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return (max(0.0, c - h), min(1.0, c + h))
class Progress:
    """Prints done / total, %, elapsed, time left and cost so far for a stage, every PROGRESS_EVERY_SEC
    seconds and at every checkpoint. 'already' = items finished in earlier sessions (checkpoint)."""
    def __init__(self, stage, total, already=0, what="calls"):
        self.stage, self.total, self.already, self.what = stage, total, already, what
        self.done, self.t0, self.last = 0, time.time(), 0.0
        self.u0 = len(USAGE_LOG) if "USAGE_LOG" in globals() else 0
        todo = total - already
        print(f"[{stage}] {total:,} {what} in total | {already:,} done earlier (checkpoint) | {todo:,} to do now")
    def _cost(self):
        if "USAGE_LOG" not in globals() or len(USAGE_LOG) <= self.u0: return 0.0
        u = pd.DataFrame(USAGE_LOG[self.u0:])
        return float(u["in"].sum() * TOKEN_PRICE["in"] + u["out"].sum() * TOKEN_PRICE["out"])
    def update(self, n=1, force=False, note=""):
        self.done += n
        now = time.time()
        if not force and now - self.last < PROGRESS_EVERY_SEC and self.done < self.total - self.already:
            return
        self.last = now
        el = now - self.t0
        fin = self.already + self.done
        rate = self.done / el if el > 0 else 0
        left = (self.total - fin) / rate if rate > 0 else float("nan")
        cost = self._cost()
        cost_left = cost / self.done * (self.total - fin) if self.done else float("nan")
        errs = sum(1 for x in (ERROR_LOG if "ERROR_LOG" in globals() else []) if x.get("stage") == self.stage)
        fmt = lambda sec: "-" if sec != sec else (f"{int(sec // 3600)}h{int(sec % 3600 // 60):02d}m" if sec >= 3600
                                                  else f"{int(sec // 60)}m{int(sec % 60):02d}s")
        print(f"  [{self.stage}] {fin:,}/{self.total:,} {self.what} ({100 * fin / max(self.total, 1):.1f}%) | "
              f"elapsed {fmt(el)} | time left ~{fmt(left)} | {rate * 60:.0f}/min | cost this session ${cost:.2f} "
              f"(+~${cost_left:.2f} to finish) | failed calls {errs}" + (f" | {note}" if note else ""), flush=True)

def _walk(o, path=""):
    if isinstance(o, dict):
        for k, v in o.items():
            yield from _walk(v, f"{path}.{k}" if path else str(k))
    elif isinstance(o, list):
        for v in o:
            yield from _walk(v, path + "[]")
    else:
        yield path, o

async def embed_missing(texts, cache, path):
    """embeds texts not yet in cache (dict md5 -> vector), 64 per request, 16 in parallel; saved every 2,000"""
    need = list(dict.fromkeys(t for t in texts if _md5(t) not in cache))
    if not need:
        return 0
    sem = asyncio.Semaphore(16)
    async def one(b):
        async with sem:
            for attempt in range(6):
                try:
                    return b, await _emb_call(b)
                except Exception as e:
                    if "401" in str(e): await trust_token.invalidate()
                    await asyncio.sleep(min(2 ** attempt, 20) * (1 + random.random()))
            return b, None
    since = 0
    prog = Progress("embeddings", len(need), 0, "texts embedded")
    _tasks = [asyncio.create_task(one(need[i:i + 64])) for i in range(0, len(need), 64)]
    try:
        for fut in tqdm(asyncio.as_completed(_tasks), total=len(_tasks), desc="embed"):
            b, X = await fut
            if X is not None:
                for t, x in zip(b, X):
                    cache[_md5(t)] = x
                since += len(b)
            prog.update(len(b))
            if since >= 2000:
                _save_npz(path, cache); since = 0
    except BaseException:
        cancel_pending(_tasks, "embedding batches"); _save_npz(path, cache); raise
    _save_npz(path, cache)
    failed = sum(_md5(t) not in cache for t in need)
    assert failed == 0, f"{failed} texts could not be embedded - re-run this cell (finished ones are kept)"
    return len(need)

async def _reauth_if_auth_error():
    """Long runs: the gateway login can expire (~1 h). On an auth-type error, log in again (at most every 5 min)."""
    global group_ai_client

        return
    _LAST_REAUTH[0] = time.time()
    try:
        group_ai_client = await asyncio.to_thread(GroupAIClient.init, staff_id=STAFF_ID)
        await trust_token.invalidate()

def cancel_pending(tasks, what):
    """Interrupt-safe: Jupyter's Interrupt stops the cell but NOT tasks already started -> cancel them."""
    n = sum(1 for t in tasks if not t.done())
    for t in tasks:
        if not t.done():
            t.cancel()
    if n:
        print(f"  interrupted - {n:,} pending {what} cancelled (nothing keeps running in the background)", flush=True)

def section_ids(doc_id, s_ids):
    ids, out = DOC_UNITS[doc_id], []
    raw = " ".join(map(str, s_ids)) if isinstance(s_ids, (list, tuple)) else str(s_ids or "")
    for n in re.findall(r"S\s*(\d+)", raw, flags=re.I) or re.findall(r"\d+", raw):
        if 1 <= int(n) <= len(ids) and ids[int(n) - 1] not in out:
            out.append(ids[int(n) - 1])
    return out

def save_safe(df, path):
    out = df.copy()
    for c in out.columns:
        if out[c].dtype == object:
            out[c] = out[c].map(lambda v: None if v is None else str(v))
    out.to_parquet(path + ".tmp", index=False)
    os.replace(path + ".tmp", path)

        rows = prev[prev["model_decision"] != "ERROR"].to_dict("records")
        print(f"[resume {stage}] {len(rows):,} pairs already done")
    done = {(r["alert"], r["doc"]) for r in rows}
    todo = [r for r in cand.itertuples() if (r.alert, r.doc) not in done]
    prog = Progress(stage, len(cand), len(cand) - len(todo), "documents read")
    sem = asyncio.Semaphore(CONCURRENCY)

    async def one(row):
        role = doc_role(row.doc)
        async with sem:
            a = await ask(PROMPTS_BY_ROLE[role][row.doc_type], build_payload(row.alert, row.doc))
        dec = str(a.get("decision", "ERROR")).upper().strip().replace(" ", "_") if a else "ERROR"
        return {"alert": row.alert, "doc": row.doc, "doc_type": row.doc_type, "doc_scope": row.doc_scope,
                "document_role": role, "search_rank": row.search_rank,
                "model_decision": dec if dec in ("LINKED", "NOT_LINKED", "INSUFFICIENT_EVIDENCE") else "ERROR",
                "link_score": a.get("link_score"), "confidence_level": a.get("confidence_level"),
                "linked_units": "; ".join(section_ids(row.doc, a.get("linked_sections"))),
                "impact_type": a.get("impact_type"), "change_needed": a.get("change_needed"),
                "evidence_from_alert": a.get("evidence_from_alert"),
                "evidence_from_target": a.get(EVIDENCE_FIELD[row.doc_type]), "reasoning": a.get("reasoning"),
                "framework_change": (str(a.get("framework_change") or "NO").upper() if role == "FRAMEWORK"
                                     else "NOT_APPLICABLE")}

    tasks = [asyncio.create_task(one(r)) for r in todo]
    try:
        for i, fut in enumerate(tqdm(asyncio.as_completed(tasks), total=len(tasks), desc=stage), 1):
            rows.append(await fut)
            if ckpt and i % SAVE_EVERY == 0:
                save_safe(pd.DataFrame(rows), ckpt)
                prog.update(1, force=True, note="checkpoint saved")
            else:
                prog.update(1)
    except BaseException:
        cancel_pending(tasks, "document reads")
        if ckpt and rows:
            save_safe(pd.DataFrame(rows), ckpt)
            print(f"  progress saved: {len(rows):,} pairs in {os.path.basename(ckpt)} - re-run this cell to continue")
        raise
    out = pd.DataFrame(rows)
    if ckpt and len(out):
        save_safe(out, ckpt)
    if todo:
        prog.update(0, force=True, note="stage finished")
    for c in out.columns:
        out[c] = out[c].map(lambda v: None if v is None or str(v) in ("None", "nan") else v)
    for c in ("link_score", "confidence_level", "search_rank"):
        out[c] = pd.to_numeric(out[c], errors="coerce")
    return out

def finalise(j):
    o = j.copy()
    a_ok = np.array([quote_found(q, ATEXT[a]) for q, a in zip(o["evidence_from_alert"], o["alert"])])
    t_ok = np.array([quote_found(q, DOC_TEXT[d]) for q, d in zip(o["evidence_from_target"], o["doc"])])
    s_ok = np.array([bool(u) and quote_found(q, " ".join(UNIT_INFO[x]["text"] for x in str(u).split("; ") if x in UNIT_INFO))
                     for q, u in zip(o["evidence_from_target"], o["linked_units"].fillna(""))])
    o["grounding_flag"] = np.select([a_ok & t_ok, a_ok | t_ok], ["GROUNDED", "PARTIAL"], default="UNGROUNDED")
    o["section_grounded"] = s_ok
    o["decision_prob"] = (o["link_score"] / 10).round(2)
    lk = (o["model_decision"] == "LINKED") & (o["link_score"].fillna(0) >= LINK_THRESHOLD)
    o["final_decision"] = np.select(
        [o["model_decision"] == "ERROR", lk & (o["grounding_flag"] == "GROUNDED"), lk,
         o["model_decision"] == "INSUFFICIENT_EVIDENCE"],
        ["ERROR", "LINKED", "INSUFFICIENT_EVIDENCE", "INSUFFICIENT_EVIDENCE"], default="NOT_LINKED")
    return o
