# ==============================================================================
# IMPACT ASSESSMENT - GAP ASSESSMENT  v3  (standalone; NOT part of the ontology work)
# ------------------------------------------------------------------------------
# Question: for every artefact an alert is linked to, does what the bank has today already
#           meet what the alert now requires? If not, what exactly is missing?
#
# DESIGN PRINCIPLE: the LLM never outputs a number. It returns categorical answers, yes/no
# checks and verbatim quotes (evidence BEFORE label in every schema). Code verifies every quote
# and derives every status, tier, confidence and rank from rule tables in CELL 1 (auditable,
# SME-tunable, identical input -> identical result).
#
# INPUTS (only these)
#   linkage outputs : rsm_linkage_pairs + rob_linkage_pairs; Helios Run B links_risk / control / instance
#                     (RGL = our own roll-up from RSM; no RGL file, no linkage audits)
#   raw texts       : Rapid2 (alerts) | RegMap extracts 1, 2, 6 | Helios risk + control files
#   policy / procedure: out of scope (later)
#
# WHAT THE LLM ANSWERS
#   step 1, per alert : requirement points; per point: change_type, mandatory (+ modal quote), the elements it
#                       specifies (who/scope, timing/frequency, threshold, evidence/records); effective date (+ quote)
#   step 2, per link  : per point: remit (IN / OUT / CANNOT_DETERMINE); per element: MATCHES / DIFFERS / ABSENT /
#                       CANNOT_DETERMINE with artefact quote; L1C / 1C: breakage checklist YES / NO / CANNOT_DETERMINE
#                       with both quotes; 1C: entity in scope
#   step 3, judge     : 4 checks + AGREE / DISAGREE / UNCERTAIN on the code-derived result (+ planted probes)
#
# WHAT CODE COMPUTES (rule tables in CELL 1, saved to schemas/rule_tables.json)
#   coverage_status per point : COVERED / PARTIAL / NOT_COVERED / CONTRADICTED / OVER_CONTROLLED / OUT_OF_REMIT /
#                               CANNOT_DETERMINE   (from element results; failed quote -> CANNOT_DETERMINE)
#   gap_tier per point        : T1..T4 / REVIEW / INFO / NONE   (change_type x mandatory(proven) x coverage)
#   artefact                  : status = worst point; tier = most severe point; breakage_types (controls)
#   grounding                 : verified / claimed quotes; HALLUCINATION_SUSPECTED if any claimed quote fails
#   confidence_score 0-100    : 7-component rubric (CONF_RUBRIC) + hard caps; band HIGH >= 80 / MEDIUM >= 50 / LOW
#   blast_radius              : counts from linkage + raw files (no LLM)
#   urgency                   : run date -> verified effective date: OVERDUE / <90D / <365D / LATER / UNKNOWN
#   route                     : gap_tier x confidence_tier
#   alert rank                : sort by worst tier, #T1, #T2, urgency, live instances in gap, #gapped artefacts
#
# OUTPUTS ({GAP_DIR}): schemas/*.schema.json + rule_tables.json | requirements.jsonl | gap_records.jsonl |
#   rgl_rollup.jsonl | alert_summary.jsonl | gap_records.parquet | gap_points.parquet | gap_audit_labelled.parquet |
#   run_manifest.json | GAP_SME_review_pack.xlsx | GAP_management_summary.xlsx | *_ckpt.parquet (re-run to resume)
# GATES: CELL 2 preflight | CELL 6 cost gate | CELL 8 automatic test gate
# ==============================================================================

def resolve(p):
    if any(ch in p for ch in "*?"):
        hits = sorted(glob.glob(p), key=os.path.getmtime)
        return hits[-1] if hits else None
    return p if os.path.exists(p) else None

def read_any(p):
    return (pd.read_excel(p) if p.lower().endswith((".xlsx", ".xls")) else
            pd.read_csv(p) if p.lower().endswith(".csv") else pd.read_parquet(p))

def col(df, key, required=True, exclude=()):
    for c in COLS[key]:
        if c in df.columns and c not in exclude:
            return c
    if required:
        raise KeyError(f"no column for '{key}' (tried {COLS[key]}); columns: {list(df.columns)[:40]}")
    return None

def blank(v):
    s = "" if v is None or (isinstance(v, float) and math.isnan(v)) else str(v).strip()
    return "" if s in ("-", "nan", "None", "NaT") else s

def cid(v):
    s = blank(v)
    return s[:-2] if re.fullmatch(r"\d+\.0", s) else s

def _w(s):
    return re.sub(r"[^a-z0-9]+", " ", str(s or "").lower()).strip()

def _segments(q):
    """Linkage quotes are often several excerpts joined ('...', ';', '|', newlines, bullets) or wrapped in
    quotes / JSON lists. Split into excerpts and keep those long enough to be meaningful."""
    q = str(q or "")
    if q.strip().startswith("["):
        try:
            q = " | ".join(map(str, json.loads(q)))
        except Exception:
            pass
    parts = re.split(r"\.\.\.|\u2026|\||;|\n|\u2022|\" \"", q)
    # drop field labels the linkage model prefixed, e.g. "Obligation Title: ...", "Summary Detail: ..."
    parts = [re.sub(r"^\s*[\"']?(obligation|regulation|summary|control|risk|instance)?\s*"
                    r"(title|detail(ed)?( text)?|text|description|name)\s*:\s*", "", x, flags=re.I) for x in parts]
    return [p for p in (_w(x) for x in parts) if len(p.split()) >= 3]

def _found(q, text):
    """VERIFIED if any excerpt of the quote is in the text: full excerpt, or its first / last 8 words."""
    t = _w(text)
    for p in _segments(q):
        w = p.split()
        if p in t or (len(w) >= 8 and (" ".join(w[:8]) in t or " ".join(w[-8:]) in t)):
            return True
    return False

def _art_full(level, aid):
    """artefact text incl. context + (for 1C) parent control standard - same text the linkage step saw"""
    x = ART[level].get(aid, {})
    s = f"{x.get('title', '')}\n{x.get('text', '')}\n{x.get('ctx', '')}"
    if level == "1C" and x.get("parent"):
        p = ART["L1C"].get(x["parent"], {})
        s += f"\n{p.get('title', '')}\n{p.get('text', '')}"
    if level == "ROB" and ROB_RSM.get(aid):              # ROB linkage quoted 'evidence_from_regulation':
        p = ART["RSM"].get(ROB_RSM[aid], {})            # may come from the obligation or its parent summary
        s += f"\n{p.get('title', '')}\n{p.get('text', '')}"
    if level == "RSM" and RSM_RGL.get(aid):
        s += f"\n{ART['RGL'].get(RSM_RGL[aid], {}).get('title', '')}"
    return s

def link_evidence(r):
    """Impact-link support from the LINKAGE OUTPUT ONLY (no audit files): are the linkage's own evidence
    quotes found in the raw texts? VERIFIED = both | PARTIAL = one | UNVERIFIED = none / no quotes.
    If the linkage file has no artefact-side quote column, the best possible is PARTIAL."""
    a = ALERT.get(r.alert, {})
    n = int(_found(r.link_ev_alert, f"{a.get('title', '')}\n{a.get('text', '')}")) + \
        int(_found(r.link_ev_artefact, _art_full(r.level, r.artefact)))
    return "VERIFIED" if n == 2 else "PARTIAL" if n == 1 else "UNVERIFIED"

def words(s):
    return re.sub(r"[^a-z0-9]+", " ", str(s or "").lower()).strip()

def quote_found(q, text, min_len=12):
    """Verbatim match (case / punctuation-insensitive); 8-word ends for long quotes; rapidfuzz partial >= 92 if installed."""
    q = words(q)
    if len(q) < min_len:
        return False
    t = words(text)
    if q in t:
        return True
    w = q.split()
    if len(w) >= 6 and (" ".join(w[:8]) in t or " ".join(w[-8:]) in t):
        return True
    return bool(fuzz and len(q) >= 25 and fuzz.partial_ratio(q, t) >= 92)

def save_ckpt(rows, path):
    if not path:
        return
    df = pd.DataFrame(rows)
    for c in df.columns:
        if df[c].dtype == object:
            df[c] = df[c].map(lambda v: json.dumps(v, ensure_ascii=False, default=str) if isinstance(v, (dict, list)) else v)
    df.to_parquet(path + ".tmp", index=False)
    os.replace(path + ".tmp", path)

def load_ckpt(path, keys):
    if path and os.path.exists(path):
        df = pd.read_parquet(path)
        df = df[df["call_status"] == "OK"]
        return df.to_dict("records"), set(map(tuple, df[keys].astype(str).values))
    return [], set()

    errs, obj = [], None
    for attempt in range(MAX_RETRIES_SCHEMA + 1):
        p = payload if attempt == 0 else {**payload, "previous_answer_errors":
              f"Your previous JSON broke the schema: {errs}. Return a corrected JSON object only."}
        obj = await call_llm(system, p, schema, MAX_TOKENS)
        errs = schema_errors(obj, schema) if obj is not None else ["no / unparseable response"]
        if not errs:
            return obj, "OK", []
    return obj, "SCHEMA_ERROR", errs

async def run_pool(items, worker, ckpt, keys, stage):
    global CURRENT_STAGE
    CURRENT_STAGE = stage
    rows, done = load_ckpt(ckpt, keys)
    todo = [it for it in items if tuple(str(it[k]) for k in keys) not in done]
    print(f"[{stage}] {len(todo):,} calls to make ({len(rows):,} already done)")
    sem = asyncio.Semaphore(CONCURRENCY)
    async def guarded(it):
        async with sem:
            return await worker(it)
    tasks = [asyncio.create_task(guarded(it)) for it in todo]
    for i, f in enumerate(tqdm(asyncio.as_completed(tasks), total=len(tasks), desc=stage), 1):
        rows.append(await f)
        if i % SAVE_EVERY == 0:
            save_ckpt(rows, ckpt)
    save_ckpt(rows, ckpt)
    return pd.DataFrame(rows)

# ---- step 1 ----------------------------------------------------------------------------
async def req_worker(it):
    a = it["alert"]; txt = alert_text(a)
    obj, st, errs = await call_valid(REQ_SYS, {"alert_text": txt, "alert_metadata": ALERT[a]["meta"]}, LLM_REQ_SCHEMA)
    obj = obj if st == "OK" else {}
    reqs = obj.get("requirements", [])
    for q in reqs:
        q["quote_verified"] = quote_found(q.get("evidence_from_alert"), txt)
        mq = words(q.get("modal_quote"))
        q["mandatory_verified"] = bool(q.get("mandatory")) and bool(mq) and f" {mq} " in f" {words(txt)} "  # proven only
        q["elements_specified"] = ["action"] + [e for e in ELEMENTS[1:] if blank(q.get(e))]
    dq = obj.get("effective_date_quote")
    d_ok = bool(dq) and len(words(dq)) >= 6 and words(dq) in words(txt)
    return {"alert": a, "call_status": st, "schema_errors": errs, "actionable": bool(obj.get("actionable", False)),
            "requirements": reqs, "effective_date": obj.get("effective_date") if d_ok else None,
            "effective_date_quote": dq, "effective_date_verified": d_ok}

def level_points(pts, level):
    """requirement points as sent to the model for this level: only the elements this level is expected to state"""
    keep = LEVEL_ELEMENTS[level]
    return [{"req_id": q["req_id"], "requirement": q.get("requirement"), "change_type": q.get("change_type"),
             **{e: q.get(e) for e in ELEMENTS if e in keep}} for q in pts]

def verified_reqs(a):
    return [q for q in (js(REQ.get(a)) or []) if q.get("quote_verified")]

# ---- step 2 ----------------------------------------------------------------------------
async def gap_worker(it):
    pts = verified_reqs(it["alert"])
    base = {k: it[k] for k in ("alert", "artefact", "level")}

               "requirement_points": level_points(pts, it["level"]),

    return {**base, "call_status": st, "schema_errors": errs, "llm": obj if st == "OK" else None}

# ---- deterministic derivation ------------------------------------------------------------
def coverage_from_elements(remit, elems, required, change_type):
    if remit == "OUT_OF_REMIT":
        return "OUT_OF_REMIT"
    if remit != "IN_REMIT":
        return "CANNOT_DETERMINE"
    res = {e["element"]: e["result_checked"] for e in elems}
    vals = [res.get(e, "CANNOT_DETERMINE") for e in required]      # element skipped by the model -> cannot determine
    # A CONFLICT needs the same subject with an incompatible parameter: DIFFERS on action, timing, threshold or
    # records. A different population (who_scope DIFFERS) means the artefact does not cover who the alert
    # covers -> treated as ABSENT (a coverage gap), never as a contradiction.
    if res.get("who_scope") == "DIFFERS":
        res["who_scope"] = "ABSENT"
        vals = [res.get(e, "CANNOT_DETERMINE") for e in required]
    if "DIFFERS" in vals:
        return "OVER_CONTROLLED" if change_type == "RELAXATION_OR_REMOVAL" else "CONTRADICTED"
    if res.get("action") == "ABSENT":
        return "NOT_COVERED"
    if all(v == "MATCHES" for v in vals):
        return "COVERED"
    if "ABSENT" in vals:
        return "PARTIAL"
    return "CANNOT_DETERMINE"

def tier_for(cov, change_type, mandatory):
    if cov in NO_GAP_COVERAGE:
        return NO_GAP_COVERAGE[cov]
    t = BASE_TIER.get(change_type, 2) + COVERAGE_TIER_SHIFT.get(cov, 0) + (0 if mandatory else NOT_MANDATORY_SHIFT)
    return f"T{min(max(t, 1), 4)}"

def urgency_of(a):
    d = EFF.get(a)
    if not d:
        return "UNKNOWN", None
    try:
        days = (dt.date.fromisoformat(str(d)) - RUN_DATE).days
    except ValueError:
        return "UNKNOWN", None
    return next(lbl for lo, hi, lbl in URGENCY_BANDS if lo <= days < hi), days

def blast_of(level, aid, alert):
    if level == "RSM":
        return {"obligations_under_summary": int(ROBS_PER_RSM.get(aid, 0))}
    if level == "L1C":
        return {"live_instances": int(LIVE_1C.get(aid, 0)), "businesses": int(LIVE_BIZ.get(aid, 0))}
    if level == "1C":
        p = ART["1C"].get(aid, {}).get("parent", "")
        return {"sibling_live_instances": int(LIVE_1C.get(p, 0)), "businesses_under_standard": int(LIVE_BIZ.get(p, 0))}
    if level == "LR":
        return {"controls_linked_to_alert": int(L1C_PER_ALERT.get(alert, 0))}
    return {"obligations": 1}

def derive_record(link, llm, call_status):
    reqs = {q["req_id"]: q for q in verified_reqs(link.alert)}
    atxt, alt = artefact_text(link.level, link.artefact), alert_text(link.alert)
    claimed = verified = 0
    points = []
    for p in (llm or {}).get("points", []):
        q = reqs.get(p["req_id"])
        if not q:
            continue
        elems = []
        for e in p.get("elements", []):
            if e["element"] not in LEVEL_ELEMENTS[link.level]:
                continue                                  # element not expected at this level -> ignored
            r = e["result"]
            ok = quote_found(e.get("artefact_quote"), atxt) if r in ("MATCHES", "DIFFERS") else None
            if ok is not None:
                claimed += 1; verified += int(ok)
            elems.append({**e, "quote_verified": ok, "result_checked": r if ok in (None, True) else "CANNOT_DETERMINE"})
        required = [e for e in q["elements_specified"] if e in LEVEL_ELEMENTS[link.level]]
        cov = coverage_from_elements(p["remit"], elems, required, q["change_type"])
        points.append(dict(req_id=p["req_id"], requirement=q.get("requirement"), change_type=q["change_type"],
                           mandatory=q["mandatory_verified"], remit=p["remit"], remit_reason=p.get("remit_reason"),
                           elements=elems, coverage_status=cov, gap_tier=tier_for(cov, q["change_type"], q["mandatory_verified"]),
                           missing=p.get("missing"), recommended_change=p.get("recommended_change")))
    breakage = []
    for k, c in ((llm or {}).get("breakage_checklist") or {}).items():
        if c["answer"] == "YES":
            claimed += 2
            ok = int(quote_found(c.get("alert_quote"), alt)) + int(quote_found(c.get("artefact_quote"), atxt))
            verified += ok
            if ok == 2:
                breakage.append(k)
    ent = ((llm or {}).get("entity_in_scope") or {}).get("answer") if link.level == "1C" else None
    if call_status != "OK":
        status, tier = "ERROR", "REVIEW"
    elif not reqs:                                        # alert has no verified requirement point
        status, tier = "NO_REQUIREMENT", "NONE"
    elif llm is None or not points:
        status, tier = "NOT_ASSESSABLE", "REVIEW"
    elif ent == "NO":
        status, tier = "ENTITY_OUT_OF_SCOPE", "NONE"
    else:
        status = min((p["coverage_status"] for p in points), key=COVERAGE_ORDER.index)
        tier = min((p["gap_tier"] for p in points), key=TIER_ORDER.get)
    urg, days = urgency_of(link.alert)
    x = ART[link.level].get(link.artefact, {})
    return {
        "schema_version": SCHEMA_VERSION, "run_id": RUN_ID, "record_id": f"{link.alert}|{link.level}|{link.artefact}",
        "alert": {"id": link.alert, "title": ALERT[link.alert]["title"]},
        "artefact": {"level": link.level, "id": link.artefact, "title": x.get("title", ""),
                     "regulation_id": RSM_RGL.get(link.artefact) if link.level == "RSM" else None,
                     "parent_control_id": x.get("parent") if link.level == "1C" else None,
                     "business": x.get("business") if link.level == "1C" else None, "context": x.get("ctx") or None},
        "impact_link": {"link_evidence": link.link_evidence,
                        "link_score": None if pd.isna(link.link_score) else float(link.link_score)},
        "result": {"artefact_status": status, "gap_tier": tier,
                   "status_counts": {k: int(v) for k, v in pd.Series([p["coverage_status"] for p in points],
                                                                     dtype=object).value_counts().items()},
                   "breakage_types": breakage, "entity_in_scope": ent,
                   "gap_summary": (llm or {}).get("gap_summary"), "n_points": len(points)},
        "grounding": {"claimed_quotes": claimed, "verified_quotes": verified,
                      "grounding_ratio": round(verified / claimed, 3) if claimed else 1.0,
                      "hallucination_suspected": verified < claimed},
        "context": {"blast_radius": blast_of(link.level, link.artefact, link.alert), "urgency": urg, "days_to_effective": days},
        "points": points,
        "audit": {"bucket": None, "verdict": None, "label": None, "error_type": None, "checks": None},
        "consistency": {"rerun": False, "element_agreement": None, "status_changed": None},
        "decisiveness": decisiveness(points),
        "requirement_quality": REQ_QUALITY.get(link.alert, 0.0),
        "triage": {"confidence_score": 0.0, "confidence_components": {}, "confidence_caps": [],
                   "confidence_tier": "LOW", "route": "NO_ROUTE"}}

def decisiveness(points):
    """share of decisions (remit per point + each required element) that are not CANNOT_DETERMINE"""
    n = d = 0
    for p in points:
        n += 1; d += p["remit"] != "CANNOT_DETERMINE"
        for e in p["elements"]:
            n += 1; d += e["result_checked"] != "CANNOT_DETERMINE"
    return round(d / n, 3) if n else 0.0

JUDGE_POINTS = {"GAP_CONFIRMED": 1.0, "NO_GAP_CONFIRMED": 1.0, None: 0.5, "NEEDS_SME": 0.27,
                "GAP_DISPUTED": 0.0, "POSSIBLE_MISSED_GAP": 0.0, "ERROR": 0.5}

def confidence_score(rec):
    """Deterministic 0-100 confidence from CONF_RUBRIC + CONF_CAPS. Returns (score, components, caps)."""
    W, g, a, c = CONF_RUBRIC, rec["grounding"], rec["audit"], rec["consistency"]
    comp = {
        "C1_gap_quotes":     W["C1_gap_quotes"] * g["grounding_ratio"],
        "C2_linkage_quotes": W["C2_linkage_quotes"] * {"VERIFIED": 1, "PARTIAL": 0.5}.get(rec["impact_link"]["link_evidence"], 0),
        "C3_requirements":   W["C3_requirements"] * rec["requirement_quality"],
        "C4_decisiveness":   W["C4_decisiveness"] * rec["decisiveness"],
        "C5_consistency":    W["C5_consistency"] * (c["element_agreement"] if c["rerun"] else 0.5),
        "C6_judge_verdict":  W["C6_judge_verdict"] * JUDGE_POINTS.get(a["label"], 0.5),
        "C7_judge_checks":   W["C7_judge_checks"] * (sum(v == "YES" for v in a["checks"].values()) / 4
                                                     if isinstance(a.get("checks"), dict) and a["checks"] else 0.5),
    }
    comp = {k: round(v, 1) for k, v in comp.items()}
    score, caps = sum(comp.values()), []
    if g["hallucination_suspected"]:
        caps.append("hallucination_suspected")
    if a["label"] in ("GAP_DISPUTED", "POSSIBLE_MISSED_GAP"):
        caps.append("judge_disputed")
    if a["label"] is None:
        caps.append("not_judged")
    if c["rerun"] and c["status_changed"]:
        caps.append("inconsistent")
    for k in caps:
        score = min(score, CONF_CAPS[k])
    if rec["result"]["artefact_status"] in ("ERROR", "NOT_ASSESSABLE"):
        score = 0.0
    return round(score, 1), comp, caps

def tier_group(t):
    return "T1-2" if t in ("T1", "T2") else "T3-4" if t in ("T3", "T4") else t

def triage(rec):
    sc, comp, caps = confidence_score(rec)
    band = next(b for th, b in CONF_BANDS if sc >= th)
    rec["triage"] = {"confidence_score": sc, "confidence_components": comp, "confidence_caps": caps,
                     "confidence_tier": band,
                     "route": ("LOG_NO_ACTION" if rec["result"]["artefact_status"] == "NO_REQUIREMENT" else
                               ROUTE.get((tier_group(rec["result"]["gap_tier"]), band), "NO_ROUTE"))}

def build_records(raw):
    lk = L.set_index(["alert", "level", "artefact"])
    out = []
    for r in raw.itertuples():
        link = pd.Series({**lk.loc[(r.alert, r.level, r.artefact)].to_dict(),
                          "alert": r.alert, "level": r.level, "artefact": r.artefact})
        rec = derive_record(link, js(r.llm), r.call_status)
        triage(rec)
        out.append(rec)
    return out

def write_jsonl(recs, path, schema):
    bad = 0
    with open(path + ".tmp", "w", encoding="utf-8") as f:
        for r in recs:
            bad += bool(schema_errors(r, schema))
            f.write(json.dumps(r, ensure_ascii=False, default=str) + "\n")
    os.replace(path + ".tmp", path)
    print(f"  {os.path.basename(path)}: {len(recs):,} records | schema-invalid {bad}")
    return bad

def model_view(rec, probe=False):
    pts = []
    for p in rec["points"]:
        el = [{k: e.get(k) for k in ("element", "artefact_quote", "result_checked")} for e in p["elements"]]
        cov = p["coverage_status"]
        if probe and p["remit"] == "IN_REMIT":            # planted error: every element claimed ABSENT
            el = [{"element": e["element"], "artefact_quote": None, "result_checked": "ABSENT"} for e in el] or \
                 [{"element": "action", "artefact_quote": None, "result_checked": "ABSENT"}]
            cov = "NOT_COVERED"
        pts.append({"req_id": p["req_id"], "remit": p["remit"], "elements": el, "coverage_status": cov})
    return {"artefact_status": "NOT_COVERED" if probe else rec["result"]["artefact_status"], "points": pts}

def wilson(k, n, z=1.96):
    if n == 0:
        return (np.nan, np.nan)
    p = k / n; d = 1 + z * z / n; c = (p + z * z / (2 * n)) / d
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return (max(0, c - h), min(1, c + h))
