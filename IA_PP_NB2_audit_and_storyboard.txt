class TrustTokenCache:
    def __init__(self):
        self.token, self.expires_at, self.lock = None, 0.0, asyncio.Lock()
    async def get(self):
        now = time.time()
        if self.token and now < self.expires_at - 5: return self.token
        async with self.lock:
            now = time.time()
            if self.token and now < self.expires_at - 5: return self.token
            self.token = await asyncio.to_thread(getattr(group_ai_client, TOKEN_METHOD))
            self.expires_at = now + 300
            return self.token
    async def invalidate(self):
        async with self.lock:
            self.token, self.expires_at = None, 0.0

trust_token = TrustTokenCache()
USAGE_LOG, ERROR_LOG, CURRENT_STAGE = [], [], "setup"

async def call_llm(system, payload, schema=None, max_tokens=2500, retries=4):
    """One gateway call -> parsed JSON dict, or None if all retries fail (logged in ERROR_LOG)."""
    kw = {"model": CHAT_MODEL, "user": USE_CASE,
          "messages": [{"role": "system", "content": system},
                       {"role": "user", "content": json.dumps(payload, ensure_ascii=False)}]}
    if MODEL_FAMILY == "gemini":
        kw.update(temperature=0.0, seed=42, reasoning_effort="low", max_tokens=max_tokens)
    else:
        kw.update(max_completion_tokens=max_tokens)
    if schema: kw["response_format"] = schema
    last = None
    for attempt in range(retries):
        try:
            token = await trust_token.get()
            r = await openai_client.chat.completions.create(
                extra_headers={"Authorization": f"Bearer {token}"}, **kw)
            u = getattr(r, "usage", None)
            USAGE_LOG.append({"stage": CURRENT_STAGE, "in": getattr(u, "prompt_tokens", 0) or 0,
                              "out": getattr(u, "completion_tokens", 0) or 0, "ts": time.time()})
            txt = (r.choices[0].message.content or "").replace("```json", "").replace("```", "").strip()
            return json.loads(txt[txt.find("{"): txt.rfind("}") + 1])
        except Exception as e:
            last = e
            msg = str(e)
            if "401" in msg: await trust_token.invalidate()
            if ("response_format" in msg or "json_schema" in msg) and "response_format" in kw:
                kw.pop("response_format")
            await asyncio.sleep(min(2 ** attempt, 20))
    ERROR_LOG.append({"stage": CURRENT_STAGE, "error": repr(last)[:300], "ts": time.time()})   # logged after ALL retries
    return None

def cost_report(stage=None):
    u = pd.DataFrame(USAGE_LOG)
    if stage and len(u): u = u[u.stage == stage]
    e = sum(1 for x in ERROR_LOG if not stage or x["stage"] == stage)
    if u.empty:
        print(f"[cost{' ' + stage if stage else ''}] no successful calls | failed calls {e}")
        return {"failed_calls": e}
    c = u["in"].sum() * TOKEN_PRICE["in"] + u["out"].sum() * TOKEN_PRICE["out"]
    print(f"[cost{' ' + stage if stage else ''}] calls {len(u):,} | in {u['in'].sum():,} | out {u['out'].sum():,} | "
          f"${c:.2f} | failed calls {e}")
    return {"calls": len(u), "cost_usd": round(float(c), 2), "failed_calls": e}

res = await call_llm("Reply with JSON only.", {"task": "return {'ok': true}"}, None, 200)
print("gateway ready:", CHAT_MODEL, "| check:", res)
assert res, f"gateway check failed - last error: {ERROR_LOG[-1]['error'] if ERROR_LOG else 'none'}"

def clean_words(s):
    return re.sub(r"[^a-z0-9]+", " ", str(s or "").lower()).strip()
def quote_found(quote, text):
    """Verbatim (case / punctuation-insensitive); long quotes may match on 8-word ends. Same as every run."""
    q = clean_words(quote)
    if len(q) < 12 or q in ("null", "none"): return False
    t = clean_words(text)
    if q in t: return True
    w = q.split()
    return len(w) >= 6 and (" ".join(w[:8]) in t or " ".join(w[-8:]) in t)
def wilson(k, n, z=1.96):
    if n == 0: return (np.nan, np.nan)
    p = k / n; d = 1 + z * z / n; c = (p + z * z / (2 * n)) / d
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return (max(0.0, c - h), min(1.0, c + h))
class Progress:
    """Prints done / total, %, elapsed, time left and cost so far for a stage, every PROGRESS_EVERY_SEC
    seconds and at every checkpoint. 'already' = items finished in earlier sessions (checkpoint)."""
    def __init__(self, stage, total, already=0, what="calls"):
        self.stage, self.total, self.already, self.what = stage, total, already, what
        self.done, self.t0, self.last = 0, time.time(), 0.0
        self.u0 = len(USAGE_LOG) if "USAGE_LOG" in globals() else 0
        todo = total - already
        print(f"[{stage}] {total:,} {what} in total | {already:,} done earlier (checkpoint) | {todo:,} to do now")
    def _cost(self):
        if "USAGE_LOG" not in globals() or len(USAGE_LOG) <= self.u0: return 0.0
        u = pd.DataFrame(USAGE_LOG[self.u0:])
        return float(u["in"].sum() * TOKEN_PRICE["in"] + u["out"].sum() * TOKEN_PRICE["out"])
    def update(self, n=1, force=False, note=""):
        self.done += n
        now = time.time()
        if not force and now - self.last < PROGRESS_EVERY_SEC and self.done < self.total - self.already:
            return
        self.last = now
        el = now - self.t0
        fin = self.already + self.done
        rate = self.done / el if el > 0 else 0
        left = (self.total - fin) / rate if rate > 0 else float("nan")
        cost = self._cost()
        cost_left = cost / self.done * (self.total - fin) if self.done else float("nan")
        errs = sum(1 for x in (ERROR_LOG if "ERROR_LOG" in globals() else []) if x.get("stage") == self.stage)
        fmt = lambda sec: "-" if sec != sec else (f"{int(sec // 3600)}h{int(sec % 3600 // 60):02d}m" if sec >= 3600
                                                  else f"{int(sec // 60)}m{int(sec % 60):02d}s")
        print(f"  [{self.stage}] {fin:,}/{self.total:,} {self.what} ({100 * fin / max(self.total, 1):.1f}%) | "
              f"elapsed {fmt(el)} | time left ~{fmt(left)} | {rate * 60:.0f}/min | cost this session ${cost:.2f} "
              f"(+~${cost_left:.2f} to finish) | failed calls {errs}" + (f" | {note}" if note else ""), flush=True)

# ====== CELL 2: AUDIT SET + COST GATE ($0) =====================================
# Strata per document type:
#   linked bands  : IMPACTED / POSSIBLE_IMPACT / SCOPE_REVIEW - all (ALL_LINKS) or SAMPLE_RATES (SAMPLE)
#   REJ_TOP_SCORE : the highest-scored rejections (near misses)
#   REJ_TOP_RANK  : procedures ranked in the search top 20 but rejected
#   REJ_RANDOM    : a random sample of all rejections -> estimate of missed impacts across the whole population
#   PROBE         : planted wrong links - the reviewer must DISAGREE
P = PAIRS[PAIRS.final_decision != "ERROR"].rename(columns={"RECORD_ID": "alert", "DOC_ID": "doc", "DOC_TYPE": "doc_type"})
parts, POPULATION = [], {}
for kind, t in P.groupby("doc_type"):
    for band in ("IMPACTED", "POSSIBLE_IMPACT", "SCOPE_REVIEW"):
        g = t[t.impact_status == band]
        n = len(g) if AUDIT_MODE == "ALL_LINKS" else min(len(g), max(SAMPLE_MIN, int(round(SAMPLE_RATES[band] * len(g)))))
        parts.append(g.sample(n, random_state=SEED).assign(stratum=band))
        POPULATION[(kind, band)] = len(g)
    rj = t[t.impact_status == "NOT_IMPACTED"]
    POPULATION[(kind, "REJ_RANDOM")] = len(rj)
    s1 = rj.sort_values("link_score", ascending=False).head(REJECTED["top_score"])
    s2 = rj[(rj.search_rank <= 20)].drop(s1.index, errors="ignore") if kind == "procedure" else rj.iloc[0:0]
    s2 = s2.sample(min(REJECTED["top_rank"], len(s2)), random_state=SEED)
    rest = rj.drop(s1.index.union(s2.index))
    s3 = rest.sample(min(REJECTED["random"], len(rest)), random_state=SEED)
    clear = rest.drop(s3.index)
    clear = clear[clear.link_score.fillna(0) <= 2]
    pr = clear.sample(min(PROBES, len(clear)), random_state=SEED + 1).copy()
    pr["model_decision"] = "LINKED"; pr["link_score"] = 7; pr["impact_type"] = "REVIEW_REQUIRED"; pr["linked_units"] = ""
    pr["evidence_from_alert"] = None; pr["evidence_from_target"] = None
    pr["reasoning"] = (f"The {kind}'s scope covers regulatory compliance for the bank's products and customers; "
                       f"the alert introduces related regulatory requirements, so the {kind} is impacted.")
    parts += [s1.assign(stratum="REJ_TOP_SCORE"), s2.assign(stratum="REJ_TOP_RANK"), s3.assign(stratum="REJ_RANDOM"),
              pr.assign(stratum="PROBE")]
AUDIT_SET = pd.concat(parts, ignore_index=True)

_LAST_REAUTH = [0.0]
async def _reauth_if_auth_error():
    """Long runs: the gateway login can expire (~1 h). On an auth-type error, log in again (at most every 5 min)."""
    global group_ai_client
    last = (ERROR_LOG[-1]["error"] if ERROR_LOG else "").lower()
    if not any(k in last for k in ("401", "403", "unauthor", "expired", "token", "auth")):
        return
    if time.time() - _LAST_REAUTH[0] < 300 or "GroupAIClient" not in globals():
        return
    _LAST_REAUTH[0] = time.time()
    try:
        group_ai_client = await asyncio.to_thread(GroupAIClient.init, staff_id=STAFF_ID)
        await trust_token.invalidate()
        print("  gateway: logged in again after an authentication error", flush=True)
    except Exception as e:
        print("  gateway: re-login failed -", repr(e)[:200], flush=True)

async def ask(system, payload):
    """call_llm with real retries; re-logs in to the gateway if the failures are authentication errors."""
    for attempt in range(4):
        a = await call_llm(system, payload, None, MAX_TOKENS)
        if isinstance(a, dict) and isinstance(a.get("items"), list) and a["items"]:
            a = a["items"][0]
        if isinstance(a, dict) and a:
            return a
        await _reauth_if_auth_error()
        await asyncio.sleep(min(2 ** attempt, 20) * (1 + random.random()))
    return {}

def cancel_pending(tasks, what):
    """Interrupt-safe: Jupyter's Interrupt stops the cell but NOT tasks already started -> cancel them."""
    n = sum(1 for t in tasks if not t.done())
    for t in tasks:
        if not t.done():
            t.cancel()
    if n:
        print(f"  interrupted - {n:,} pending {what} cancelled (nothing keeps running in the background)", flush=True)

def build_payload(alert, doc_id):
    return {"alert_text": ATEXT[alert], "alert_metadata": META.get(alert, {}),
            "document_scope": SCOPE_LABEL.get(DOC_SCOPE.get(doc_id, ""), "unknown")
                              + (f" | {APPLIC_NOTE[DOC_APPLIC.get(doc_id, '')]}" if DOC_APPLIC.get(doc_id) else ""),
            "document_text": DOC_TEXT[doc_id]}

async def run_judge(audit, ckpt):
    global CURRENT_STAGE
    CURRENT_STAGE = "audit"
    key = lambda df: list(zip(df["alert"], df["doc"], df["stratum"]))
    rows, done = [], set()
    if os.path.exists(ckpt):
        prev = pd.read_parquet(ckpt)
        prev = prev[prev["verdict"] != "ERROR"]
        rows, done = prev.to_dict("records"), set(key(prev))
    todo = audit[[k not in done for k in key(audit)]]
    prog = Progress("audit", len(audit), len(audit) - len(todo), "reviews")
    sem = asyncio.Semaphore(CONCURRENCY)
    async def one(row):
        p = build_payload(row.alert, row.doc)
        labels = [UNIT_INFO[u]["gpps_label"] for u in str(row.linked_units or "").split("; ") if u in UNIT_INFO]
        payload = {"alert_text": p["alert_text"], "alert_metadata": p["alert_metadata"], "target_id": row.doc,
                   "document_scope": p["document_scope"], "target_text": p["document_text"],
                   "model_output": {"decision": "LINKED" if row.stratum in ("IMPACTED", "POSSIBLE_IMPACT", "SCOPE_REVIEW", "PROBE")
                                    else "NOT_LINKED",
                                    "link_score": None if pd.isna(row.link_score) else float(row.link_score),
                                    "linked_sections": labels, "impact_type": row.impact_type, "reasoning": row.reasoning,
                                    "evidence_from_alert": row.evidence_from_alert,
                                    "evidence_from_target": row.evidence_from_target}}
        async with sem:
            a = await ask(JUDGE_PROMPTS[row.doc_type], payload)
        v = str(a.get("verdict", "ERROR")).upper().strip() if a else "ERROR"
        return {"alert": row.alert, "doc": row.doc, "stratum": row.stratum,
                "verdict": v if v in ("AGREE", "DISAGREE", "UNCERTAIN") else "ERROR",
                "error_type": a.get("error_type"), "correct_decision": a.get("correct_decision"),
                "impact_type_ok": a.get("impact_type_correct"), "reviewer_alert_quote": a.get("alert_quote"),
                "reviewer_target_quote": a.get("target_quote"), "reviewer_confidence": a.get("confidence_level"),
                "reviewer_rationale": a.get("rationale")}
    tasks = [asyncio.create_task(one(r)) for r in todo.itertuples()]
    try:
        for i, fut in enumerate(tqdm(asyncio.as_completed(tasks), total=len(tasks), desc="audit"), 1):
            rows.append(await fut)
            if i % SAVE_EVERY == 0:
                save_safe(pd.DataFrame(rows), ckpt); prog.update(1, force=True, note="checkpoint saved")
            else:
                prog.update(1)
    except BaseException:
        cancel_pending(tasks, "reviews")
        if rows: save_safe(pd.DataFrame(rows), ckpt)
        raise
    out = pd.DataFrame(rows)
    save_safe(out, ckpt)
    if len(todo): prog.update(0, force=True, note="audit finished")
    for c in out.columns:
        out[c] = out[c].map(lambda v: None if v is None or str(v) in ("None", "nan") else v)
    print("[audit] verdicts:", out["verdict"].value_counts().to_dict())
    return out
