"""Re-create the project folder from the text bundle part(s) (the bank git / file transfer accepts text only).

    python unpack_bundle.py IA_PP_BUNDLE_part1.txt IA_PP_BUNDLE_part2.txt [...] --dest .

Each file in a bundle sits between '#### FILE: <path> | sha256=<hash> ####' and '#### END FILE ####'.
The hash of every unpacked file is checked; a mismatch (damaged copy) is reported and the run stops.
"""
import os
import re
import sys
import hashlib
import argparse

HEAD = re.compile(r"^#### FILE: (.+?) \| sha256=([0-9a-f]{64}) ####$")


def unpack(parts, dest):
    files, cur, buf, digest = {}, None, [], None
    for part in parts:
        with open(part, encoding="utf-8", newline="") as f:
            for line in f.read().split("\n"):
                m = HEAD.match(line)
                if m:
                    cur, digest, buf = m.group(1), m.group(2), []
                elif line == "#### END FILE ####" and cur:
                    files[cur] = ("\n".join(buf), digest)
                    cur = None
                elif cur is not None:
                    buf.append(line)
    bad = []
    for path, (text, digest) in sorted(files.items()):
        if ".." in path.split("/") or path.startswith("/"):
            bad.append(path)
            continue
        if hashlib.sha256(text.encode("utf-8")).hexdigest() != digest:
            bad.append(path)
            continue
        full = os.path.join(dest, path)
        os.makedirs(os.path.dirname(full), exist_ok=True)
        with open(full, "w", encoding="utf-8", newline="") as f:
            f.write(text)
    print(f"unpacked {len(files) - len(bad)} file(s) into {os.path.abspath(dest)}")
    if bad:
        print(f"DAMAGED or unsafe, not written: {bad}")
        sys.exit(1)


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("parts", nargs="+")
    ap.add_argument("--dest", default=".")
    a = ap.parse_args()
    unpack(a.parts, a.dest)

def check(name, cond):
    print(("PASS " if cond else "FAIL ") + name)
    if not cond:
        FAILS.append(name)


class APIErr(Exception):
    def __init__(self, status, msg="err"):
        super().__init__(msg)
        self.status_code = status


class StubCompletions:
    def __init__(self, script):
        self.script, self.calls = list(script), []

    async def create(self, extra_headers=None, **kw):
        self.calls.append((extra_headers, kw))
        step = self.script.pop(0)
        if isinstance(step, Exception):
            raise step
        msg = types.SimpleNamespace(content=step)
        return types.SimpleNamespace(choices=[types.SimpleNamespace(message=msg, finish_reason="stop")],
                                     usage=types.SimpleNamespace(prompt_tokens=1000, completion_tokens=200))


def settings():
    cfg = os.path.join(PROJECT, "config", "config.yaml")
    os.environ.setdefault("IA_GATEWAY_BASE_URL", "https://example.invalid")
    os.environ.setdefault("IA_STAFF_ID", "test")
    os.environ.setdefault("IA_USE_CASE", "UC-test")
    return load_settings(cfg, {"sources.alert_scope": "/tmp/x.parquet", "execution.backoff_max_sec": 1})


def test_live_stub():
    s = settings()
    log = RunLog(tempfile.mkdtemp(), "test")
    good = json.dumps({"policy_id": "POL-1", "decision": "LINKED", "link_score": 8})
    comp = StubCompletions([APIErr(429), APIErr(503), "```json\n" + good + "\n```"])
    client = types.SimpleNamespace(chat=types.SimpleNamespace(completions=comp))
    gw = LiveGateway(s, log, Usage(0.2, 1.2), client=client, token_fn=lambda: "TOKEN")
    r = asyncio.run(gw.chat("system", {"a": 1}, ref={"alert": "1", "doc": "POL-1"}))
    check("live: retries 429 + 503 then succeeds", r.ok and r.attempts == 3 and r.data["decision"] == "LINKED")
    hdr, kw = comp.calls[-1]
    check("live: bearer token sent", hdr == {"Authorization": "Bearer TOKEN"})
    check("live: gpt family sends max_completion_tokens from config",
          kw.get("max_completion_tokens") == s.model["max_tokens"] and "temperature" not in kw)
    check("live: model and use case from config", kw["model"] == s.model["chat_model"] and kw["user"] == "UC-test")

    comp2 = StubCompletions([APIErr(404, "no Route matched")])
    gw2 = LiveGateway(s, log, Usage(0.2, 1.2), client=types.SimpleNamespace(chat=types.SimpleNamespace(completions=comp2)),
                      token_fn=lambda: "T")
    r2 = asyncio.run(gw2.chat("system", {}, ref={}))
    check("live: 404 is not retried and coded E-GW-008", (not r2.ok) and r2.code == "E-GW-008" and len(comp2.calls) == 1)

    s.model["family"] = "gemini"
    gw3 = LiveGateway(s, log, Usage(0.2, 1.2), client=client, token_fn=lambda: "T")
    kw3 = gw3._kwargs("sys", {})
    check("live: gemini family sends temperature, seed, reasoning_effort, max_tokens",
          kw3["temperature"] == s.model["temperature"] and kw3["seed"] == s.model["seed"]
          and kw3["reasoning_effort"] == s.model["reasoning_effort"] and kw3["max_tokens"] == s.model["max_tokens"])
    comp4 = StubCompletions(["not json at all", "still not", "no", "nope"])
    gw4 = LiveGateway(settings(), log, Usage(0.2, 1.2),
                      client=types.SimpleNamespace(chat=types.SimpleNamespace(completions=comp4)), token_fn=lambda: "T")
    r4 = asyncio.run(gw4.chat("system", {}, ref={}))

def test_validation():
    s = settings()
    p = PromptStore(s)
    a, err, w = p.validate("policy:BUSINESS", {"decision": "linked", "link_score": "8", "impact_type": "maybe"})
    check("validate: coerces decision / score, nulls bad optional field",
          err is None and a["decision"] == "LINKED" and a["link_score"] == 8 and a["impact_type"] is None and w)
    a, err, w = p.validate("policy:BUSINESS", {"link_score": 9})
    check("validate: missing decision is an error", err is not None)
    a, err, w = p.validate("policy:FRAMEWORK", {"decision": "NOT_LINKED", "framework_change": "yes - process"})
    check("validate: framework_change coerced", err is None and a["framework_change"] == "YES")

| E-GW-009 | Bad request (400/413/422) | PAIR | Input too long for the model, or a parameter the model does not accept. | See error_detail. Not retried. If 'context length': the document is too long for the model. |
| E-GW-010 | Circuit breaker: too many consecutive failures | STAGE | The gateway is down or rejecting every call. | The stage stops and saves progress. Fix the cause (see the codes before it in errors.jsonl), then re-run. |
| E-GW-011 | Gateway check failed at start | RUN | The one-call connectivity check before the run failed. | See the detail; usually E-GW-003 / 008 / 007. |
| E-GW-012 | Unknown gateway error | PAIR | An error type the code does not classify. | See error_detail and run.log; retried automatically. |
| E-LLM-001 | Empty answer | PAIR | The model returned no content (often the answer budget was spent on thinking). | Retried. If frequent, raise model.max_tokens. |
| E-LLM-002 | Answer is not valid JSON | PAIR | The model wrote text around the JSON or broke the format. | Retried automatically. |
| E-LLM-003 | Answer misses a required field | PAIR | e.g. 'decision' missing or not LINKED / NOT_LINKED / INSUFFICIENT_EVIDENCE. | Retried automatically. |
| E-LLM-004 | Answer cut off (max tokens reached) | PAIR | model.max_tokens too low for this answer. | Raise model.max_tokens (8000 covered 3x the longest US answer). |
| W-LLM-001 | Optional answer field invalid | NONE | e.g. link_score 'eight', impact_type misspelt. | Value set to null (or coerced); the answer is kept. |
| W-LLM-002 | Cited section ids not in the document | NONE | The model cited S-ids that do not exist. | Those ids are ignored; the link is still checked by its quotes. |
| E-EMB-001 | Embedding batch failed after retries | STAGE | Gateway problem on the embedding model. | Re-run: finished embeddings are cached and not paid again. |
| E-EMB-002 | Embedding size mismatch | STAGE | The cache was built with another embedding model. | Delete the emb_*.npz files in the output folder, or restore the model. |
| W-EMB-001 | Embedding cache unreadable - rebuilt | NONE | Corrupt .npz file (e.g. disk full during a save). | Rebuilt automatically (costs under $1). |
| E-SHL-001 | Query plans missing after retries | STAGE | The model failed for some alerts. | Re-run: finished plans are kept. |
| W-SHL-001 | Saved shortlist does not match the current alerts | NONE | New alerts, new query plans or a different top-K. | Alerts already in the saved shortlist keep it; missing alerts get a fresh shortlist. |
| E-RUN-001 | Cost cap reached | RUN | Estimated or measured spend reached run.max_cost_usd. | Progress is saved. Raise the cap if the spend is expected, then re-run the same command. |
| E-RUN-002 | Interrupted | RUN | Ctrl+C, Kernel > Interrupt or SIGTERM from the scheduler. | Progress is saved; re-run the same command to continue. |
| E-RUN-003 | Test gate failed | RUN | The 10-pair test before the full run had errors, unverifiable quotes or a cost above the cap. | Read the test table in run.log; fix the cause before the full run. |
| E-RUN-004 | Unexpected error on one pair | PAIR | A bug or an unusual document. The traceback is in run.log. | The pair is marked ERROR and retried on the next run. Report the traceback. |
| E-RUN-005 | Unexpected error in a stage | RUN | A bug or an unusual input. The traceback is in run.log. | Report run.log and errors.jsonl to RCAI&A. |
| E-RUN-006 | Document text shorter than its source | RUN | The no-truncation check failed: a document would not be sent in full. | Report to RCAI&A; nothing was spent. |
| W-RUN-001 | Pairs without an answer at the end of a stage | NONE | Their calls failed every retry (see their error_code). | Re-run the same command: only those pairs are retried. |
| W-CKP-001 | Damaged checkpoint lines skipped | NONE | The process was killed while writing the last line. | Those pairs are simply read again. |
| E-CKP-001 | Checkpoint unreadable | STAGE | The checkpoint file is damaged beyond line level. | Rename it (to keep a copy) and re-run; the stage restarts. |
| E-OUT-001 | Cannot write output | RUN | Folder not writable, disk full, or a file open in another program. | Free space / fix permissions; re-run. |
| W-OUT-001 | Excel summary not written | NONE | openpyxl missing or the file is open elsewhere. | Parquet, CSV and JSONL outputs are still written. |
| E-OUT-002 | No pair results to report | RUN | The results stage ran before the policy / procedure stages. | Run --stage policies and procedures first (or --stage all). |

## Exit codes

| Exit code | Meaning |
|---|---|
| 0 | Completed, every pair answered |
| 1 | Completed, some pairs without answer: re-run the same command to retry only those |
| 2 | Config / environment / prompt / schema / source problem (nothing spent) |
| 3 | Gateway unusable (authentication, route, circuit breaker) |
| 4 | Cost cap reached, progress saved |
| 5 | Pre-run test gate failed |
| 6 | Unexpected internal error (see run.log) |
| 130 | Interrupted, progress saved |
