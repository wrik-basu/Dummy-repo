# ---- 4. source files: exist, readable, required columns -------------------------
def header(path):
    """Column names (+ row count when cheap) without loading the file."""
    if path.endswith(".parquet"):
        import pyarrow.parquet as pq
        f = pq.ParquetFile(path)
        return list(f.schema_arrow.names), f.metadata.num_rows
    if path.endswith((".xlsx", ".xls")):
        return list(pd.read_excel(path, nrows=3).columns), None
    return list(pd.read_csv(path, nrows=3).columns), None


def one_of(cols, *alts):
    return next((a for a in alts if a in cols), None)

COLS_SEEN = {}

        try:

            add("source", f"{key} readable", "PASS", f"{len(cols)} cols" + (f", {n:,} rows" if n is not None else ""))
        except Exception as e:

            hit = one_of(cols, *alts)
            add("columns", f"{key}: {' | '.join(alts)}", "PASS" if hit else "FAIL", hit or "missing",
                f"{key} must have column {' or '.join(alts)} (renamed in this extract?)")
        if key in NEEDS_DECISION:
            hit = one_of(cols, *C.COLS["decision"])
            add("columns", f"{key}: decision column", "PASS" if hit else "FAIL", hit or "missing",
                f"add the column name to COLS['decision'] in pipeline_config.py")

            hit = one_of(cols, *C.COLS["verdict"])
            add("columns", f"{key}: judge verdict column", "PASS" if hit else "FAIL", hit or "missing",

            hit = one_of(cols, *C.COLS[field])
            add("columns", f"{key}: {field}", "PASS" if hit else "FAIL", hit or f"none of {C.COLS[field]}",
                f"add this extract's column name to COLS['{field}'] in pipeline_config.py")

            share = s.str.match(r"^1C-\d+").mean()

        s = (pd.read_excel(p, usecols=["Obligation ID"], nrows=200) if p.endswith(("xlsx", "xls"))
             else pd.read_parquet(p, columns=["Obligation ID"]).head(200))["Obligation ID"].astype(str)
        share = s.str.match(r"^ROB-\d+").mean()
