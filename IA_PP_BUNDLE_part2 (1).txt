def _ids(arg, path):
    ids = []
    if arg:
        ids += [x.strip() for x in str(arg).split(",") if x.strip()]
    if path:
        with open(path, encoding="utf-8") as f:
            ids += [ln.strip() for ln in f if ln.strip() and not ln.startswith("#")]
    return ids


def _run_coro(coro):
    """asyncio.run, also from inside Jupyter (whose event loop is already running): then a worker thread is used."""
    try:
        asyncio.get_running_loop()
    except RuntimeError:
        return asyncio.run(coro)
    box = {}

    def target():
        box["v"] = asyncio.run(coro)
    t = threading.Thread(target=target, daemon=False)
    t.start()
    t.join()
    return box.get("v", EXIT_INTERNAL)


def _overrides(a):
    o = {}
    ids = _ids(getattr(a, "alert_ids", None), getattr(a, "alert_ids_file", None))
    if ids:
        o["run.alert_ids"] = ids
    if getattr(a, "alerts_limit", None):
        o["run.alerts_limit"] = a.alerts_limit
    if getattr(a, "output_dir", None):
        o["run.output_dir"] = a.output_dir
    if getattr(a, "mock", False):
        o["gateway.mode"] = "mock"
    if getattr(a, "max_cost", None) is not None:
        o["run.max_cost_usd"] = a.max_cost
    if getattr(a, "no_test_gate", False):
        o["run.test_gate"] = False
    return o


def cmd_run(a, stages):
    from .config import load_settings
    from .logs import RunLog
    from .pipeline import run_pipeline
    try:
        s = load_settings(a.config, _overrides(a))
    except IAError as e:                                   # config problems happen before logging exists
        print(f"{e}\n{explain(e.code)}", file=sys.stderr)
        return exit_code_for(e.code)
    log = RunLog(s.log_dir, s.run_id, s.logging["level"], s.logging["log_llm_calls"], s.logging["log_payloads"])

    async def main_async():
        stop = asyncio.Event()
        if threading.current_thread() is threading.main_thread():
            loop = asyncio.get_running_loop()
            for sig in (signal.SIGINT, signal.SIGTERM):
                try:
                    loop.add_signal_handler(sig, lambda: (log.info("stop signal received - finishing calls in hand, "
                                                                   "saving progress"), stop.set()))
                except (NotImplementedError, RuntimeError, ValueError):
                    pass
        return await run_pipeline(s, log, stages, stop)

    try:
        return _run_coro(main_async())
    finally:
        log.close()

"""Compare two runs (e.g. the validated notebook run and this package on the same 10 alerts).

python -m ia_pp compare --run-a <notebook output folder> --run-b <package output folder> [--alert-ids 1,2,3]

Writes compare_report.xlsx (or CSVs if openpyxl is missing) in run B's folder and prints the agreement.
Expect small differences: the model does not answer identically twice (about 10% of links flip between two runs
of the same prompt), and procedures differ if the shortlist was rebuilt (use run.reuse_search_from to share it).
"""
import os

import pandas as pd

KEYS = ["RECORD_ID", "DOC_ID"]
LINKISH = ["IMPACTED", "POSSIBLE_IMPACT", "SCOPE_REVIEW"]


def _load(folder):
    p = os.path.join(folder, "ia_pp_pairs.parquet")
    if not os.path.exists(p):
        raise FileNotFoundError(f"{p} not found - run the results stage there first")
    df = pd.read_parquet(p)
    df["RECORD_ID"] = df["RECORD_ID"].astype(str)
    return df


def compare(run_a, run_b, alert_ids=None, out_dir=None):
    A, B = _load(run_a), _load(run_b)
    common = sorted(set(A.RECORD_ID) & set(B.RECORD_ID))
    if alert_ids:
        common = [a for a in common if a in {str(x) for x in alert_ids}]
    if not common:
        raise ValueError("no alert in common between the two runs")
    A, B = A[A.RECORD_ID.isin(common)], B[B.RECORD_ID.isin(common)]
    cols = ["DOC_TYPE", "impact_status", "link_score", "final_decision", "linked_sections"]
    m = A[KEYS + cols].merge(B[KEYS + cols], on=KEYS, how="outer", suffixes=("_a", "_b"), indicator=True)
    m["DOC_TYPE"] = m.DOC_TYPE_a.fillna(m.DOC_TYPE_b)
    both = m[m._merge == "both"].copy()
    both["same_status"] = both.impact_status_a == both.impact_status_b
    both["linked_a"] = both.impact_status_a.isin(LINKISH)
    both["linked_b"] = both.impact_status_b.isin(LINKISH)
    rows = []
    for t, g in both.groupby("DOC_TYPE"):
        la, lb = int(g.linked_a.sum()), int(g.linked_b.sum())
        inter = int((g.linked_a & g.linked_b).sum())
        rows.append(dict(doc_type=t, pairs_in_both=len(g), same_status_pct=round(100 * g.same_status.mean(), 1),
                         linked_a=la, linked_b=lb, linked_in_both=inter,
                         link_overlap_pct=round(100 * inter / max(la + lb - inter, 1), 1),
                         impacted_a=int((g.impact_status_a == "IMPACTED").sum()),
                         impacted_b=int((g.impact_status_b == "IMPACTED").sum()),
                         only_in_a=int(((m._merge == "left_only") & (m.DOC_TYPE == t)).sum()),
                         only_in_b=int(((m._merge == "right_only") & (m.DOC_TYPE == t)).sum())))
    summary = pd.DataFrame(rows)
    per_alert = both.groupby(["RECORD_ID", "DOC_TYPE"]).agg(linked_a=("linked_a", "sum"), linked_b=("linked_b", "sum"),
                                                           same_status_pct=("same_status", "mean")).reset_index()
    per_alert["same_status_pct"] = (100 * per_alert.same_status_pct).round(1)
    diffs = both[~both.same_status].sort_values(["RECORD_ID", "DOC_TYPE", "DOC_ID"])
    out_dir = out_dir or run_b
    try:
        with pd.ExcelWriter(os.path.join(out_dir, "compare_report.xlsx")) as xl:
            pd.DataFrame({"How to read": [
                f"Run A: {run_a}", f"Run B: {run_b}", f"Alerts compared: {len(common)}",
                "same_status_pct: share of pairs with the same impact_status in both runs.",
                "link_overlap_pct: links (impacted / possible / scope review) found by both, over links found by either.",
                "only_in_a / only_in_b: pairs read in one run only (procedure shortlist differences)."]}
            ).to_excel(xl, sheet_name="How to read", index=False)
            summary.to_excel(xl, sheet_name="summary", index=False)
            per_alert.to_excel(xl, sheet_name="per alert", index=False)
            diffs.drop(columns=["_merge"]).to_excel(xl, sheet_name="different status", index=False)
        report = os.path.join(out_dir, "compare_report.xlsx")
    except Exception:
        summary.to_csv(os.path.join(out_dir, "compare_summary.csv"), index=False)
        diffs.to_csv(os.path.join(out_dir, "compare_differences.csv"), index=False)
        report = os.path.join(out_dir, "compare_summary.csv")
    return summary, per_alert, diffs, report

"""Load, check and resolve config/config.yaml into a Settings object. No execution parameter is hard-coded elsewhere."""
import os
import re
import copy
from pathlib import Path

from .errors import IAError

REQUIRED = {
    "run": ["run_id", "output_dir", "log_dir", "alert_ids", "alerts_limit", "max_cost_usd", "test_gate", "reuse_search_from"],
    "country": ["code", "name", "terms", "codes", "entity_codes"],
    "sources": ["rapid2", "alert_scope", "policy_local", "policy_global", "procedure"],
    "gateway": ["mode", "base_url", "base_url_env", "staff_id", "staff_id_env", "use_case", "use_case_env",
                "quota_project", "quota_project_env", "token_lifetime_sec", "relogin_min_interval_sec",
                "http_timeout_sec", "max_connections", "mock_failure_rate"],
    "model": ["chat_model", "family", "max_tokens", "temperature", "seed", "reasoning_effort",
              "price_per_million_tokens", "embedding_model", "embedding_batch_size", "embedding_max_chars"],
    "execution": ["concurrency", "query_plan_concurrency", "embedding_concurrency", "transport_retries",
                  "answer_retries", "backoff_max_sec", "circuit_breaker_failures", "save_every", "progress_every_sec"],
    "search": ["proc_top_k", "rrf_k", "tfidf_ngram_max", "tfidf_min_df", "tfidf_max_df"],
    "rules": ["published_only", "link_threshold", "headline_min_score"],
    "logging": ["level", "log_llm_calls", "log_payloads"],
    "paths": ["prompts_dir", "schemas_dir", "exceptions_dir"],
}
PLACEHOLDER = re.compile(r"^<.*>$|<your |<EXECUTION_OUTPUT_DIR>")


class Settings:
    """Attribute access to the config sections: s.run["output_dir"], s.model["max_tokens"], ... plus resolved paths."""

    def __init__(self, raw, config_path):
        self.raw = raw
        self.config_path = Path(config_path).resolve()
        self.project_dir = self.config_path.parent.parent      # config/ sits in the project folder
        for section in REQUIRED:
            setattr(self, section, raw[section])
        self.country_code = str(raw["country"]["code"])

    # -- paths -------------------------------------------------------------------------------------------
    def path(self, value):
        """Resolve a path from the config: {country} filled, relative paths taken from the project folder."""
        if value is None:
            return None
        p = str(value).replace("{country}", self.country_code)
        p = os.path.expandvars(os.path.expanduser(p))
        return str(p if os.path.isabs(p) else (self.project_dir / p))

    @property
    def run_id(self):
        return str(self.run["run_id"]).replace("{country}", self.country_code)

    @property
    def output_dir(self):
        return self.path(self.run["output_dir"])

    @property
    def log_dir(self):
        return self.path(self.run["log_dir"])

    def secret(self, name):
        """Gateway setting: environment variable first, then the config value."""
        env_name = self.gateway.get(f"{name}_env")
        value = os.environ.get(env_name) if env_name else None
        return value if value else self.gateway.get(name)

    def masked(self):
        """Copy of the config safe to write to the logs (gateway secrets masked)."""
        r = copy.deepcopy(self.raw)
        for k in ("base_url", "staff_id", "use_case", "quota_project"):
            v = self.secret(k)
            r["gateway"][k] = (str(v)[:4] + "***") if v and not PLACEHOLDER.search(str(v)) else v
        return r


def _num(cfg, sec, key, kind, lo=None, hi=None):
    v = cfg[sec][key]
    try:
        v = kind(v)
    except (TypeError, ValueError):
        raise IAError("E-CFG-004", f"{sec}.{key} = {cfg[sec][key]!r} is not a {kind.__name__}")
    if (lo is not None and v < lo) or (hi is not None and v > hi):
        raise IAError("E-CFG-004", f"{sec}.{key} = {v} must be between {lo} and {hi}")
    cfg[sec][key] = v


def load_settings(config_path, overrides=None):
    """Read and validate the YAML config. overrides: dict of 'section.key' -> value (from the command line)."""
    try:
        import yaml
    except ImportError:
        raise IAError("E-ENV-001", "PyYAML (package 'pyyaml') is needed to read config.yaml")
    if not config_path or not os.path.isfile(config_path):
        raise IAError("E-CFG-001", f"config file not found: {config_path}")
    try:
        with open(config_path, encoding="utf-8") as f:
            raw = yaml.safe_load(f)
    except yaml.YAMLError as e:
        raise IAError("E-CFG-002", f"{config_path}: {e}")
    if not isinstance(raw, dict):
        raise IAError("E-CFG-002", f"{config_path}: the file is empty or not a mapping")

    for sec, keys in REQUIRED.items():
        if not isinstance(raw.get(sec), dict):
            raise IAError("E-CFG-003", f"section '{sec}' missing")
        for k in keys:
            if k not in raw[sec]:
                raise IAError("E-CFG-003", f"'{sec}.{k}' missing")

    for dotted, value in (overrides or {}).items():
        sec, key = dotted.split(".", 1)
        raw[sec][key] = value

    # types and ranges
    _num(raw, "run", "max_cost_usd", float, 0)
    _num(raw, "model", "max_tokens", int, 256, 128000)
    _num(raw, "model", "temperature", float, 0, 2)
    _num(raw, "model", "seed", int, 0)
    _num(raw, "model", "embedding_batch_size", int, 1, 250)
    _num(raw, "model", "embedding_max_chars", int, 100)
    for k, lo, hi in (("concurrency", 1, 500), ("query_plan_concurrency", 1, 200), ("embedding_concurrency", 1, 100),
                      ("transport_retries", 1, 10), ("answer_retries", 1, 10), ("backoff_max_sec", 1, 300),
                      ("circuit_breaker_failures", 10, 100000), ("save_every", 1, 100000), ("progress_every_sec", 1, 3600)):
        _num(raw, "execution", k, int, lo, hi)
    _num(raw, "search", "proc_top_k", int, 1, 100000)
    _num(raw, "search", "rrf_k", int, 1, 10000)
    _num(raw, "search", "tfidf_ngram_max", int, 1, 3)
    _num(raw, "rules", "link_threshold", int, 0, 10)
    _num(raw, "rules", "headline_min_score", int, 0, 10)
    _num(raw, "gateway", "token_lifetime_sec", int, 30, 3600)
    _num(raw, "gateway", "relogin_min_interval_sec", int, 0, 86400)
    _num(raw, "gateway", "http_timeout_sec", float, 5, 3600)
    _num(raw, "gateway", "max_connections", int, 1, 2000)
    _num(raw, "gateway", "mock_failure_rate", float, 0, 1)
    if raw["rules"]["headline_min_score"] < raw["rules"]["link_threshold"]:
        raise IAError("E-CFG-004", "rules.headline_min_score must be >= rules.link_threshold")
    if raw["gateway"]["mode"] not in ("live", "mock"):
        raise IAError("E-CFG-004", "gateway.mode must be 'live' or 'mock'")
    if str(raw["model"]["family"]).lower() not in ("auto", "gpt", "gemini"):
        raise IAError("E-CFG-004", "model.family must be auto, gpt or gemini")
    price = raw["model"]["price_per_million_tokens"]
    if not isinstance(price, dict) or "input" not in price or "output" not in price:
        raise IAError("E-CFG-004", "model.price_per_million_tokens needs 'input' and 'output'")
    for k in ("terms", "codes", "entity_codes"):
        if not isinstance(raw["country"][k], list):
            raise IAError("E-CFG-004", f"country.{k} must be a list")
    if not isinstance(raw["run"]["alert_ids"], list):
        raise IAError("E-CFG-004", "run.alert_ids must be a list (use [] for all alerts)")
    if raw["run"]["alerts_limit"] is not None:
        _num(raw, "run", "alerts_limit", int, 1)
    if raw["logging"]["level"] not in ("DEBUG", "INFO", "WARNING", "ERROR"):
        raise IAError("E-CFG-004", "logging.level must be DEBUG, INFO, WARNING or ERROR")

    s = Settings(raw, config_path)
    for k, v in raw["sources"].items():
        if PLACEHOLDER.search(str(v)):

# ------------------------------------------------------------------------------------------- config text editing
def _set_yaml_value(text, section, key, value):
    """Replace 'key: value' inside a top-level section, keeping every comment and the rest of the file unchanged."""
    lines = text.split("\n")
    start = next((i for i, ln in enumerate(lines) if re.match(rf"^{re.escape(section)}:\s*(#.*)?$", ln)), None)
    if start is None:
        raise IAError("E-CFG-003", f"section '{section}' not found in the config text")
    for i in range(start + 1, len(lines)):
        ln = lines[i]
        if ln and not ln.startswith((" ", "#")):
            break
        m = re.match(rf"^(\s+{re.escape(key)}:\s*)(\[[^\]]*\]|\"[^\"]*\"|'[^']*'|[^#]*?)(\s+#.*)?$", ln)
        if m:
            lines[i] = m.group(1) + value + (m.group(3) or "")
            return "\n".join(lines)
    raise IAError("E-CFG-003", f"'{section}.{key}' not found in the config text")


def _flow_list(values):

def init_country(project_dir, code, name, template="config/config.yaml", force=False):
    code = re.sub(r"[^A-Za-z0-9_]", "", code).upper()
    if not code:
        raise IAError("E-CFG-004", "country code must be letters / digits, e.g. UK")
    src = os.path.join(project_dir, template)
    dst = os.path.join(project_dir, "config", f"config_{code}.yaml")
    if os.path.exists(dst) and not force:
        raise IAError("E-CFG-004", f"{dst} exists (use --force to overwrite)")
    with open(src, encoding="utf-8") as f:
        t = f.read()
    t = _set_yaml_value(t, "country", "code", json.dumps(code))
    t = _set_yaml_value(t, "country", "name", json.dumps(name))
    for k in ("terms", "codes", "entity_codes"):

# ------------------------------------------------------------------------------------------- discover
def _scan_file(path, kind, prefix):
    df = read_table(path)
    idc = pick(df, "policy_core_id", "procedure_core_id", "policy_id", "procedure_id")
    docs = {}
    if idc is None or "json_data" not in df.columns:
        return docs
    for _, r in df.iterrows():
        d = load_json(r.get("json_data"))
        if not d:
            continue
        doc = prefix + N(r[idc])
        title = clean(as_str(first(d, KEYS[kind]["title"])))
        ent_codes, ent_names, countries, level = set(), set(), set(), set()
        for p, v in _walk(d):
            if v is None or isinstance(v, (dict, list)):
                continue
            pl, sv = p.lower(), str(v).strip()
            leaf = re.sub(r"\[\]", "", pl).split(".")[-1]
            if not sv or len(sv) > 200 or "sgmnt" in pl or "appendix" in pl:
                continue
            if leaf in ("plcylglentycde", "proclglentycde", "region"):
                level.add(sv.upper())
            elif leaf == "legalentycde":
                ent_codes.add(sv.upper())
            elif leaf == "legalentyname":
                ent_names.add(sv)
            elif re.search(r"cntry|country", leaf):
                countries.add(sv)
        docs[doc] = dict(doc_id=doc, title=title, level=", ".join(sorted(level)), entity_codes=sorted(ent_codes),
                         entity_names=sorted(ent_names), countries=sorted(countries))
    return docs


def hub_candidates(pairs, framework_docs, share=0.4, min_alerts=20):
    """Documents the model linked to >= share of the alerts (excluding known framework documents)."""
    if pairs is None or not len(pairs):
        return pd.DataFrame(columns=["DOC_ID", "doc_title", "DOC_TYPE", "alerts_linked", "share_of_alerts"])
    n_al = pairs.RECORD_ID.nunique()
    if n_al < min_alerts:
        return pd.DataFrame(columns=["DOC_ID", "doc_title", "DOC_TYPE", "alerts_linked", "share_of_alerts"])
    lk = pairs[(pairs.model_decision == "LINKED") & ~pairs.DOC_ID.isin(framework_docs)]
    g = lk.groupby(["DOC_ID", "DOC_TYPE"]).RECORD_ID.nunique().rename("alerts_linked").reset_index()
    g["share_of_alerts"] = (g.alerts_linked / n_al).round(2)
    g["doc_title"] = g.DOC_ID.map(dict(zip(pairs.DOC_ID, pairs.get("doc_title", pairs.DOC_ID))))
    return g[g.share_of_alerts >= share].sort_values("share_of_alerts", ascending=False)[
        ["DOC_ID", "doc_title", "DOC_TYPE", "alerts_linked", "share_of_alerts"]].reset_index(drop=True)

# ------------------------------------------------------------------------------------------- forget
def forget(output_dir, docs, echo=print):
    docs = {d.strip().upper() for d in docs if d.strip()}
    total = 0
    for name in ("policy_ckpt.jsonl", "procedure_ckpt.jsonl"):
        path = os.path.join(output_dir, name)
        if not os.path.exists(path):
            continue
        keep, drop = [], 0
        with open(path, encoding="utf-8") as f:
            for line in f:
                try:
                    if json.loads(line).get("doc") in docs:
                        drop += 1
                        continue
                except Exception:
                    pass
                keep.append(line)
        if drop:
            shutil.copy2(path, f"{path}.bak_{time.strftime('%Y%m%d_%H%M%S')}")
            with open(path + ".tmp", "w", encoding="utf-8") as f:
                f.writelines(keep)
            os.replace(path + ".tmp", path)
        total += drop
        echo(f"{name}: {drop} saved answer(s) removed for {sorted(docs)}")
    echo("next: run the same 'python -m ia_pp run ...' command - only these pairs are read again")
    return total

class IAError(Exception):
    """An expected, classified failure. `code` is a key of ERRORS; `detail` says what exactly happened."""

    def __init__(self, code, detail="", **context):
        self.code = code if code in ERRORS else "E-RUN-005"
        self.detail = str(detail)
        self.context = context
        info = ERRORS[self.code]
        super().__init__(f"[{self.code}] {info.title}: {self.detail}")

def exit_code_for(code):
    if code == "E-RUN-001":
        return EXIT_COST_CAP
    if code == "E-RUN-002":

    if code == "E-RUN-003":

    if not e:

    return (f"{e.code}  {e.title}\n  stops      : {e.scope}\n  likely cause: {e.likely_cause}\n"
            f"  what to do : {e.action}")

@dataclass
class ChatResult:
    ok: bool

class Usage:
    """Token and cost counters, per stage and total."""

    def __init__(self, price_in_per_m, price_out_per_m):
        self.p_in, self.p_out = float(price_in_per_m) / 1e6, float(price_out_per_m) / 1e6
        self.stages = {}

    def add(self, stage, tin, tout, ok=True):
        st = self.stages.setdefault(stage, {"calls": 0, "failed_calls": 0, "in": 0, "out": 0})
        st["calls" if ok else "failed_calls"] += 1
        st["in"] += int(tin or 0)
        st["out"] += int(tout or 0)

    def cost(self, stage=None):
        sts = [self.stages.get(stage, {})] if stage else list(self.stages.values())
        return float(sum(x.get("in", 0) * self.p_in + x.get("out", 0) * self.p_out for x in sts))

    def report(self):
        return {k: {**v, "cost_usd": round(self.cost(k), 4)} for k, v in self.stages.items()} | \
            {"_total_cost_usd": round(self.cost(), 4)}


def classify(exc):
    """Exception from the openai / httpx client -> error code."""
    status = getattr(exc, "status_code", None) or getattr(getattr(exc, "response", None), "status_code", None)
    name, msg = type(exc).__name__.lower(), str(exc).lower()
    if status in (401, 403) or "unauthor" in msg or "forbidden" in msg:
        return "E-GW-003"
    if status == 404 or "no route" in msg:
        return "E-GW-008"
    if status == 429 or "rate limit" in msg:
        return "E-GW-004"
    if status in (400, 413, 422):
        return "E-GW-009"
    if status is not None and 500 <= int(status) < 600:
        return "E-GW-005"
    if "timeout" in name or "timed out" in msg:
        return "E-GW-006"
    if "connect" in name or "ssl" in msg or "certificate" in msg or "connection" in msg:
        return "E-GW-007"
    return "E-GW-012"


def parse_json_answer(text):
    """Model text -> dict. Tolerates ```json fences and text around the JSON object."""
    t = (text or "").replace("```json", "").replace("```", "").strip()
    if not t:
        raise ValueError("empty")
    a, b = t.find("{"), t.rfind("}")
    if a < 0 or b <= a:
        raise ValueError("no JSON object in the answer")
    d = json.loads(t[a:b + 1])
    if isinstance(d, dict) and isinstance(d.get("items"), list) and d["items"] and isinstance(d["items"][0], dict):
        d = d["items"][0]                                    # some models wrap the object in {"items": [...]}
    if not isinstance(d, dict):
        raise ValueError("JSON is not an object")
    return d


class _Base:
    def __init__(self, s, log, usage):
        self.s, self.log, self.usage = s, log, usage
        self.stage = "setup"
        ex = s.execution
        self.transport_retries, self.backoff_max = ex["transport_retries"], ex["backoff_max_sec"]

    async def _sleep(self, attempt, code):
        base = min(2 ** attempt, self.backoff_max)
        if code == "E-GW-004":                               # rate limited: back off harder
            base = min(base * 3, self.backoff_max * 3)
        await asyncio.sleep(base * (0.5 + random.random()))

    def _log_call(self, ref, attempt, status, code, t0, tin=0, tout=0, finish=None):
        self.log.llm_call(alert=ref.get("alert"), doc=ref.get("doc"), attempt=attempt, status=status, code=code,
                          ms=int((time.time() - t0) * 1000), tokens_in=tin, tokens_out=tout, finish=finish)


class LiveGateway(_Base):
    """OpenAI-compatible bank gateway with a Bearer token from py_group_ai_tools."""

    def __init__(self, s, log, usage, client=None, token_fn=None):
        super().__init__(s, log, usage)
        g, m = s.gateway, s.model
        self.model = m["chat_model"]
        fam = str(m["family"]).lower()
        self.family = ("gemini" if "gemini" in self.model.lower() else "gpt") if fam == "auto" else fam
        self.use_case = s.secret("use_case")
        self.token_lifetime, self.relogin_gap = g["token_lifetime_sec"], g["relogin_min_interval_sec"]
        self._token, self._token_exp, self._lock = None, 0.0, None
        self._last_relogin = 0.0
        self._group_client = None
        if client is not None:                               # injected (tests)
            self.client, self._token_fn = client, token_fn
            return
        try:
            import httpx
            import openai
        except ImportError as e:
            raise IAError("E-ENV-001", f"{e.name} (needed for the live gateway)")
        qp = s.secret("quota_project")
        if qp and not str(qp).startswith("<"):
            os.environ.setdefault("GOOGLE_CLOUD_QUOTA_PROJECT", str(qp))
        try:
            try:
                import truststore
                verify = truststore.SSLContext(ssl.PROTOCOL_TLS_CLIENT)
            except ImportError:
                verify = True
            http = httpx.AsyncClient(http2=self._has_h2(), verify=verify, timeout=httpx.Timeout(g["http_timeout_sec"]),
                                     limits=httpx.Limits(max_connections=g["max_connections"],
                                                         max_keepalive_connections=min(200, g["max_connections"])))
            self.client = openai.AsyncOpenAI(
                api_key="N/A", http_client=http, base_url=s.secret("base_url"),
                default_headers={"Content-Type": "application/json", "x-correlation-id": str(uuid.uuid4()),
                                 "x-usersession-id": str(uuid.uuid4())})
        except Exception as e:
            raise IAError("E-GW-001", f"{type(e).__name__}: {e}")
        self._token_fn = self._login()

    @staticmethod
    def _has_h2():
        try:
            import h2  # noqa: F401
            return True
        except ImportError:
            return False

    def _login(self):
        try:
            from py_group_ai_tools import GroupAIClient
        except ImportError:
            raise IAError("E-ENV-001", "py_group_ai_tools (bank gateway client)")
        try:
            gc = GroupAIClient.init(staff_id=self.s.secret("staff_id"))
        except Exception as e:
            raise IAError("E-GW-001", f"GroupAIClient.init failed: {type(e).__name__}: {e}")
        names = ["get_dsp_token", "get_dap_token", "get_token", "get_access_token", "get_bearer_token"]
        names += sorted(n for n in dir(gc) if "token" in n.lower() and not n.startswith("_"))
        method = next((n for n in names if callable(getattr(gc, n, None))), None)
        if not method:
            raise IAError("E-GW-002", "no token method found on GroupAIClient")
        self._group_client = gc
        self.log.debug(f"gateway token method: {method}")
        return getattr(gc, method)

    async def _get_token(self):
        if self._lock is None:
            self._lock = asyncio.Lock()
        now = time.time()
        if self._token and now < self._token_exp - 5:
            return self._token
        async with self._lock:
            now = time.time()
            if self._token and now < self._token_exp - 5:
                return self._token
            try:
                self._token = await asyncio.to_thread(self._token_fn)
            except Exception as e:
                raise IAError("E-GW-002", f"{type(e).__name__}: {e}")
            self._token_exp = now + self.token_lifetime
            return self._token

    async def _on_auth_error(self):
        self._token, self._token_exp = None, 0.0
        if self._group_client is None or time.time() - self._last_relogin < self.relogin_gap:
            return
        self._last_relogin = time.time()
        try:
            self._token_fn = await asyncio.to_thread(self._login)
            self.log.info("  gateway: logged in again after an authentication error")
        except Exception as e:
            self.log.event("E-GW-002", f"re-login failed: {e}")

    def _kwargs(self, system, payload):
        m = self.s.model
        kw = {"model": self.model, "user": self.use_case,
              "messages": [{"role": "system", "content": system},
                           {"role": "user", "content": json.dumps(payload, ensure_ascii=False)}]}
        if self.family == "gemini":
            kw.update(temperature=m["temperature"], seed=m["seed"], reasoning_effort=m["reasoning_effort"],
                      max_tokens=m["max_tokens"])
        else:
            kw.update(max_completion_tokens=m["max_tokens"])
        return kw

    async def chat(self, system, payload, ref=None):
        ref = ref or {}
        kw = self._kwargs(system, payload)
        code, detail, tin, tout, finish = None, "", 0, 0, None
        for attempt in range(1, self.transport_retries + 1):
            t0 = time.time()
            try:
                token = await self._get_token()
                r = await self.client.chat.completions.create(extra_headers={"Authorization": f"Bearer {token}"}, **kw)
            except IAError as e:
                code, detail = e.code, e.detail
                self._log_call(ref, attempt, "error", code, t0)
                await self._sleep(attempt, code)
                continue
            except Exception as e:
                code, detail = classify(e), f"{type(e).__name__}: {str(e)[:300]}"
                self._log_call(ref, attempt, "error", code, t0)
                self.usage.add(self.stage, 0, 0, ok=False)
                if code == "E-GW-003":
                    await self._on_auth_error()
                if code not in RETRYABLE:
                    break
                await self._sleep(attempt, code)
                continue
            u = getattr(r, "usage", None)
            tin, tout = getattr(u, "prompt_tokens", 0) or 0, getattr(u, "completion_tokens", 0) or 0
            self.usage.add(self.stage, tin, tout, ok=True)
            try:
                choice = r.choices[0]
                finish = getattr(choice, "finish_reason", None)
                text = choice.message.content or ""
            except Exception:
                text = ""
            self.log.payload(alert=ref.get("alert"), doc=ref.get("doc"), attempt=attempt, payload=payload, answer=text)
            if not text.strip():
                code, detail = ("E-LLM-004" if finish == "length" else "E-LLM-001"), f"finish_reason={finish}"
                self._log_call(ref, attempt, "error", code, t0, tin, tout, finish)
                await self._sleep(attempt, code)
                continue
            try:
                data = parse_json_answer(text)
            except Exception as e:
                code = "E-LLM-004" if finish == "length" else "E-LLM-002"
                detail = f"{e}; finish_reason={finish}; starts: {text[:120]!r}"
                self._log_call(ref, attempt, "error", code, t0, tin, tout, finish)
                await self._sleep(attempt, code)
                continue
            self._log_call(ref, attempt, "ok", None, t0, tin, tout, finish)
            return ChatResult(True, data, None, "", tin, tout, attempt, finish)
        return ChatResult(False, None, code or "E-GW-012", detail, tin, tout, self.transport_retries, finish)

    async def embed(self, texts):
        last = None
        for attempt in range(1, 7):
            t0 = time.time()
            try:
                token = await self._get_token()
                r = await self.client.embeddings.create(model=self.s.model["embedding_model"], input=texts,
                                                        user=self.use_case,
                                                        extra_headers={"Authorization": f"Bearer {token}"})
                self.log.llm_call(alert=None, doc=f"embed x{len(texts)}", attempt=attempt, status="ok", code=None,
                                  ms=int((time.time() - t0) * 1000))
                return np.asarray([d.embedding for d in r.data], dtype=np.float32)
            except Exception as e:
                code = e.code if isinstance(e, IAError) else classify(e)
                last = f"{code} {type(e).__name__}: {str(e)[:200]}"
                self.log.llm_call(alert=None, doc=f"embed x{len(texts)}", attempt=attempt, status="error", code=code,
                                  ms=int((time.time() - t0) * 1000))
                if code == "E-GW-003":
                    await self._on_auth_error()
                if code in GATEWAY_FATAL:
                    break
                await self._sleep(attempt, code)
        raise IAError("E-EMB-001", last or "unknown")


class MockGateway(_Base):
    """Offline simulator: no network, no cost. Deterministic answers built from the real texts, so every downstream
    rule (quote check, sections, bands, framework rule, outputs) runs exactly as in production.
    gateway.mock_failure_rate injects failures to exercise retries, ERROR rows and the re-run path."""

    def __init__(self, s, log, usage):
        super().__init__(s, log, usage)
        self.fail_rate = float(s.gateway["mock_failure_rate"])
        self.latency = float(s.gateway.get("mock_latency_sec", 0) or 0)
        self.rng = random.Random(int(s.model["seed"]))

    @staticmethod
    def _h(*parts):
        return int(hashlib.md5("|".join(map(str, parts)).encode()).hexdigest(), 16)

    async def chat(self, system, payload, ref=None):
        ref = ref or {}
        await asyncio.sleep(self.latency * (0.5 + self.rng.random()) if self.latency else 0)
        for attempt in range(1, self.transport_retries + 1):
            t0 = time.time()
            if self.rng.random() < self.fail_rate:
                code = self.rng.choice(["E-GW-004", "E-GW-005", "E-GW-006", "E-LLM-002"])
                self._log_call(ref, attempt, "error", code, t0)
                self.usage.add(self.stage, 0, 0, ok=False)
                continue
            data = self._answer(system, payload)
            tin = (len(system) + len(json.dumps(payload))) // 4
            self.usage.add(self.stage, tin, 400, ok=True)
            self._log_call(ref, attempt, "ok", None, t0, tin, 400, "stop")
            return ChatResult(True, data, None, "", tin, 400, attempt, "stop")
        return ChatResult(False, None, "E-GW-005", "mock: injected failures on every attempt", 0, 0,
                          self.transport_retries)

    def _answer(self, system, p):
        if "SEARCH QUERIES" in system:
            words = p.get("alert_text", "").split()
            return {"alert_id": p.get("alert_id"), "core_mandate": " ".join(words[:12]),
                    "requirements": [" ".join(words[i:i + 8]) for i in range(0, min(len(words), 24), 8)] or ["n/a"],
                    "affected_activities": words[3:6], "policy_language": [" ".join(words[:10])],
                    "procedure_language": [" ".join(words[2:12])], "keywords": words[:8]}
        kind = "policy" if "policy_id" in p else "procedure"
        doc = p.get(f"{kind}_id")
        h = self._h(p.get("alert_id"), doc)
        linked = h % 4 == 0
        score = 5 + h % 6 if linked else h % 5

               f"evidence_from_{kind}": first_line[:90] if linked else None,
               "linked_sections": sections[:1] if linked else [],
               "impact_type": ("CHANGE_REQUIRED" if h % 3 == 0 else "REVIEW_REQUIRED") if linked else "NONE",
               "change_needed": "Mock: review the cited section." if linked else None}
        if p.get("document_role") == "FRAMEWORK":
            ans["framework_change"] = "YES" if linked and h % 2 == 0 else "NO"
        return ans

    async def embed(self, texts):

            out.append(v)
        self.log.llm_call(alert=None, doc=f"embed x{len(texts)}", attempt=1, status="ok", code=None, ms=0)
        return np.vstack(out)

"""GPPS JSON -> documents -> sections -> sub-sections (GPPS UI numbering). Same parsing as the validated notebook."""
import re
import json

import pandas as pd

NA = {"", "n/a", "na", "-", "--", "none", "nan", "null"}


def N(x):
    """Normalised id: upper case, '123.0' -> '123'."""
    s = str(x).strip().upper()
    return s[:-2] if re.fullmatch(r"\d+\.0", s) else s


def clean(x):
    return "" if x is None or str(x).strip().lower() in NA else str(x).strip()


def as_str(x):
    if x is None:
        return ""
    if isinstance(x, list):
        return " | ".join(str(i).strip() for i in x if i is not None and str(i).strip())
    return str(x).strip()


def as_dict(x):
    if isinstance(x, list):
        x = next((i for i in x if isinstance(i, dict)), {})
    return x if isinstance(x, dict) else {}


def as_list(x):
    return [i for i in x if isinstance(i, dict)] if isinstance(x, list) else []


def first(d, keys):
    for k in keys:
        v = d.get(k)
        if v not in (None, "", []):
            return v
    return None


def strip_html(s):
    if not isinstance(s, str) or not s:
        return ""
    s = re.sub(r"(?is)<!doctype.*?>|<head.*?>.*?</head>", " ", s)
    s = re.sub(r"(?s)<[^>]+>", " ", s)
    return re.sub(r"\s+", " ", html.unescape(s)).strip()


def load_json(raw):
    if isinstance(raw, dict):
        return raw
    try:
        return json.loads(raw)
    except Exception:
        return {}


def pick(df, *cands):
    return next((c for c in cands if c in df.columns), None)


HEADING = re.compile(r"^\s*([A-Z]?\.?\d+(?:\.\d+){1,3})\.?\s+(.{3,150}?)\s*$")   # "2.2. Model Inputs" / "B.2.7.4 Use of"
MARK = "␞SUBHEAD␞"


def split_subsections(raw_html, section_no, strict_prefix):
    """Numbered bold / underlined headings inside a segment's HTML become sub-sections."""
    if not isinstance(raw_html, str) or not raw_html.strip():
        return []
    found = []

    def mark(m):
        inner = strip_html(m.group(0))
        h = HEADING.match(inner)
        top = re.sub(r"^[A-Z]\.?", "", h.group(1)).split(".")[0] if h else ""
        if h and (not strict_prefix or not section_no or top == str(section_no).split(".")[0]):
            found.append((h.group(1), h.group(2).strip(" :")))
            return f" {MARK}{len(found) - 1}{MARK} "
        return m.group(0)

    marked = re.sub(r"(?is)<(strong|b|u)\b[^>]*>.*?</\1>|<span[^>]*underline[^>]*>.*?</span>", mark, raw_html)
    parts = re.split(rf"{MARK}(\d+){MARK}", strip_html(marked))
    subs, seen = [], set()
    if parts[0].strip():
        subs.append({"sub_no": f"{section_no}.0" if section_no else "", "sub_title": "(section introduction)",
                     "text": parts[0].strip()})
    for i in range(1, len(parts), 2):
        num, title = found[int(parts[i])]
        body = parts[i + 1].strip() if i + 1 < len(parts) else ""
        if (num, body[:200]) in seen:
            continue                                                  # GPPS sometimes repeats a block
        seen.add((num, body[:200]))
        subs.append({"sub_no": num, "sub_title": title, "text": body})
    return subs


KEYS = {
    "policy": dict(app="policyApplication", core=["plcyCoreId"], title=["policyTitle"],
                   status=("versn", ["versStatCde"]), top=[("policyText", "Policy Statement"), ("policyPurpose", "Purpose")],
                   seg_text=["sgmntText"], seg_title=["sgmntTitlText"], seg_disp=["sgmntDispInd"],
                   seg_no=["dispSortOrdrNum", "sortOrdrNum"], seg_reason=["sgmntReasonText"], strict=True),
    "procedure": dict(app="procedureApplication", core=["procCoreId"], title=["procedureTitle"],
                      status=("versn", ["procVersnStatCde", "versStatCde"]), top=[("procedureText", "Procedure Statement")],
                      seg_text=["procSgmntText", "sgmntText"], seg_title=["procSgmntTitlText", "sgmntTitlText"],
                      seg_disp=["procSgmntDispInd", "sgmntDispInd"],
                      seg_no=["procDispSortOrdrNum", "procSortOrdrNum", "dispSortOrdrNum", "sortOrdrNum"],
                      seg_reason=["procSgmntReasonText", "sgmntReasonText"], strict=False),
}
ID_COLUMNS = ("policy_core_id", "procedure_core_id", "policy_id", "procedure_id")


def parse_doc(raw, kind):
    K = KEYS[kind]
    d = load_json(raw)
    if not d:
        return None
    app = as_dict(d.get(K["app"]))
    secs = []
    for key, label in K["top"]:
        o = as_dict(d.get(key))
        secs.append(dict(section_no="", title=strip_html(as_str(o.get("sgmntTitlText") or label)),
                         html=o.get("sgmntText") or "", displayed="Y", reason="", kind="top"))
    for s in as_list(app.get("segment")):
        secs.append(dict(section_no=clean(as_str(first(s, K["seg_no"]))),
                         title=strip_html(as_str(first(s, K["seg_title"]) or "Section")),
                         html=first(s, K["seg_text"]) or "",
                         displayed=(clean(as_str(first(s, K["seg_disp"]))).upper() or "Y"),
                         reason=strip_html(as_str(first(s, K["seg_reason"]))), kind="segment"))
    out, seen = [], set()
    for s in secs:
        text = strip_html(s["html"])
        if not text:
            continue
        h = hashlib.md5(text.encode()).hexdigest()
        if h in seen:
            continue                                                  # duplicate segment text
        seen.add(h)
        subs = split_subsections(s["html"], s["section_no"], K["strict"]) if s["kind"] == "segment" else []
        out.append({**{k: v for k, v in s.items() if k != "html"}, "text": text,
                    "subsections": subs or [{"sub_no": "", "sub_title": "", "text": text}]})
    ver = as_dict(d.get(K["status"][0]))
    return dict(core_id=N(first(d, K["core"]) or ""), title=clean(as_str(first(d, K["title"]))),
                status=clean(as_str(first(ver, K["status"][1]))), sections=out)


def load_docs(path, kind, scope, published_only, log):
    df = read_table(path)
    if "json_data" not in df.columns:
        raise IAError("E-SRC-003", f"{path}: column 'json_data' missing (columns: {list(df.columns)[:15]})")
    idc = pick(df, *ID_COLUMNS)
    recs, bad = [], 0
    for _, r in df.iterrows():
        try:
            p = parse_doc(r["json_data"], kind)
        except Exception as e:                                        # one odd document never stops the run
            log.debug(f"parse error in {path}: {type(e).__name__}: {e}")
            p = None
        if p is None:
            bad += 1
            continue
        p.update(doc_id=N(r[idc]) if idc else p["core_id"], version=r.get("version_num"), scope=scope, kind=kind)
        recs.append(p)
    if bad:
        log.event("W-SRC-001", f"{bad:,} of {len(df):,} rows in {path} not parseable - skipped")
    out = pd.DataFrame(recs)
    n_all = len(out)
    if not len(out):
        return out
    if published_only:
        out = out[out.status.astype(str).str.upper().str.startswith("PUBL")]
    out = out.assign(_v=pd.to_numeric(out.version, errors="coerce").fillna(-1))
    out = out.sort_values("_v", ascending=False).drop_duplicates("doc_id").drop(columns="_v").reset_index(drop=True)
    log.info(f"  {kind:9s} {scope:6s}: {n_all:,} rows -> {len(out):,} documents (published, latest version)")
    return out


def build_units(docs):
    """Assessment unit = a visible numbered SUB-SECTION; a visible section without numbered sub-sections is one
    unit. Hidden segments (shown 'N' in the GPPS UI) are flagged hidden=True and excluded later."""
    rows = []
    for d in docs.itertuples():
        prefix = "POL" if d.kind == "policy" else "PROC"
        for s in d.sections:
            for sub in s["subsections"]:
                rows.append(dict(
                    unit_id=f"{prefix}-{d.doc_id}|{s['section_no'] or s['title'][:20]}|{sub['sub_no'] or '-'}",
                    doc_type=d.kind, doc_id=f"{prefix}-{d.doc_id}", doc_title=d.title, scope=d.scope,
                    section_no=s["section_no"], section_title=s["title"], subsection_no=sub["sub_no"],
                    subsection_title=sub["sub_title"], hidden=s["displayed"] == "N", hidden_reason=s["reason"],
                    level="sub-section" if sub["sub_no"] else "section", text=sub["text"], chars=len(sub["text"])))
    return pd.DataFrame(rows)

"""File helpers: tolerant readers, atomic writers, append-only checkpoints."""
import os
import glob

import pandas as pd

def resolve_source(pattern):
    """A path, or a glob (newest match). Returns None if nothing exists."""
    p = str(pattern)
    if any(ch in p for ch in "*?["):
        hits = sorted(glob.glob(p, recursive=True), key=os.path.getmtime)
        return hits[-1] if hits else None
    return p if os.path.exists(p) else None


def read_table(path):
    """parquet / csv / xlsx -> DataFrame, with a fallback for parquet files that carry odd metadata."""
    p = str(path)
    try:
        if p.lower().endswith((".xlsx", ".xls")):
            return pd.read_excel(p)
        if p.lower().endswith(".csv"):
            return pd.read_csv(p)
        try:
            return pd.read_parquet(p)
        except Exception:
            import pyarrow.parquet as pq
            return pq.read_table(p).replace_schema_metadata(None).to_pandas(date_as_object=True)
    except FileNotFoundError:
        raise IAError("E-SRC-001", p)
    except Exception as e:
        raise IAError("E-SRC-002", f"{p}: {type(e).__name__}: {e}")


def ensure_dir(path):
    try:
        os.makedirs(path, exist_ok=True)
        probe = os.path.join(path, ".write_test")
        with open(probe, "w") as f:
            f.write("ok")
        os.remove(probe)
    except OSError as e:
        raise IAError("E-OUT-001", f"folder not writable: {path} ({e})")


def _stringify_objects(df):
    out = df.copy()
    for c in out.columns:
        if out[c].dtype == object:
            out[c] = out[c].map(lambda v: None if v is None or (isinstance(v, float) and v != v) else str(v))
    return out


def write_parquet(df, path):
    """Atomic: written to a temporary file, then renamed (a crash never leaves half a file)."""
    tmp = path + ".tmp"
    try:
        _stringify_objects(df).to_parquet(tmp, index=False)
        os.replace(tmp, path)
    except Exception as e:
        try:
            if os.path.exists(tmp):
                os.remove(tmp)
        except OSError:
            pass
        raise IAError("E-OUT-001", f"{path}: {type(e).__name__}: {e}")


def write_text(text, path):
    tmp = path + ".tmp"
    try:
        with open(tmp, "w", encoding="utf-8") as f:
            f.write(text)
        os.replace(tmp, path)
    except Exception as e:
        raise IAError("E-OUT-001", f"{path}: {type(e).__name__}: {e}")


def write_json(obj, path):
    write_text(json.dumps(obj, indent=2, ensure_ascii=False, default=str), path)


class JsonlCheckpoint:
    """Append-only checkpoint, one JSON line per finished (alert, document) pair.

    Appending never rewrites earlier work, so a crash loses at most the lines not yet flushed. On load the last
    line per key wins, ERROR rows are dropped (so they are retried) and damaged lines are skipped (W-CKP-001)."""

    def __init__(self, path, key=("alert", "doc"), log=None):
        self.path, self.key, self.log = path, key, log
        self._buf = []

    def load(self):
        if not os.path.exists(self.path):
            return []
        rows, bad = {}, 0
        try:
            with open(self.path, encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        r = json.loads(line)
                        rows[tuple(r[k] for k in self.key)] = r
                    except Exception:
                        bad += 1
        except OSError as e:
            raise IAError("E-CKP-001", f"{self.path}: {e}")
        if bad and self.log:
            self.log.event("W-CKP-001", f"{bad} damaged line(s) in {os.path.basename(self.path)} skipped")
        return [r for r in rows.values() if r.get("model_decision") != "ERROR"]

    def add(self, row):
        self._buf.append(row)

    def flush(self):
        if not self._buf:
            return
        try:
            with open(self.path, "a", encoding="utf-8") as f:
                for r in self._buf:
                    f.write(json.dumps(r, ensure_ascii=False, default=str) + "\n")
                f.flush()
                os.fsync(f.fileno())
            self._buf = []
        except OSError as e:
            raise IAError("E-OUT-001", f"checkpoint {self.path}: {e}")
