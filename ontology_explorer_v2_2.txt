function esc(v) {
  if (v === null || v === undefined) return "";
  if (Array.isArray(v)) v = v.join(" | ");
  return String(v).replace(/[&<>"']/g, (c) =>
    ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
}
function short(s, n) { s = String(s || ""); return s.length > n ? s.slice(0, n - 1) + "…" : s; }
function blank(v) { return v === null || v === undefined || v === "" || String(v).toLowerCase() === "nan"; }

function verdictClass(v) {
  const s = String(v || "").toUpperCase();
  if (["AGREE", "CONFIRMED"].includes(s)) return "agree";
  if (["DISAGREE", "DISPUTED"].includes(s)) return "disagree";
  if (["UNCERTAIN", "NEEDS_SME"].includes(s)) return "uncertain";
  return "";
}
function verdictText(v) {
  return { agree: "Judge confirms", disagree: "Judge disputes", uncertain: "Judge unsure → SME" }[verdictClass(v)]
         || "Not judged";
}
function typeColor(t) { return (TYPES[t] || {}).color || "#777"; }
function typeLabel(t) { return (TYPES[t] || {}).label || t; }

// Accepts ONE whole JSON file, or ALL split parts selected together
// (e.g. ontology_demo_data.json.part01 ... part03). Parts are joined in
// file-name order, so the part numbers must sort correctly (01, 02, ...).
function readFiles(fileList) {
  const files = [...fileList].sort((a, b) => a.name.localeCompare(b.name));
  if (!files.length) return;
  Promise.all(files.map((f) => f.text()))
    .then((texts) => {
      // Python writes missing numbers as bare NaN / Infinity, which is not
      // valid JSON. Turn those values (never text inside quotes) into null.
      const clean = texts.join("").replace(/([:\[,]\s*)(NaN|-?Infinity)(?=\s*[,\]}])/g, "$1null");
      try { loadData(JSON.parse(clean)); }
      catch (err) {
        alert("Could not read the data.\n\nIf you loaded split parts, select ALL parts together.\n\n" + err.message);
      }
    });
}

  .catch(() => {});

function loadData(d) {
  if (!d || !Array.isArray(d.nodes) || !Array.isArray(d.edges)) {
    alert("This file does not look like ontology_demo_data.json (needs 'nodes' and 'edges').");
    return;
  }

  // --- Normalise types and index the graph ---------------------------
  d.nodes.forEach((n) => { n.type = TYPE_ALIAS[n.type] || n.type; });
  d.edges.forEach((e, i) => { e._i = i; });
  DATA = d;
  nodeById = {}; inEdges = {}; outEdges = {};
  d.nodes.forEach((n) => { nodeById[n.id] = n; inEdges[n.id] = []; outEdges[n.id] = []; });
  d.edges = d.edges.filter((e) => nodeById[e.source] && nodeById[e.target]);
  d.edges.forEach((e) => { outEdges[e.source].push(e); inEdges[e.target].push(e); });

  // --- Alert filter ---------------------------------------------------
  const alerts = d.nodes.filter((n) => n.type === "Alert");
  const sel = $("alertSelect");
  sel.innerHTML = `<option value="ALL">All showcase alerts (${alerts.length})</option>` +
    alerts.map((a) => `<option value="${esc(a.id)}">${esc(a.title)} – ${esc(short(a.text, 50))}</option>`).join("");
  sel.disabled = false;
  $("search").disabled = false;
  $("viewGraph").disabled = false;
  $("loadBtn").classList.remove("pulse");
  $("empty").hidden = true;

  const ai = d.edges.filter((e) => e.kind === "AI_LINK").length;
  $("status").textContent =
    `Loaded ${d.nodes.length} nodes · ${d.edges.length} links (${ai} AI links with evidence)` +
    (d.meta && d.meta.generated ? ` · generated ${d.meta.generated}` : "") +
    " · sample of real data";

  selected = null;
  closePanel();
  setView("graph");
}

function classInfo(id) {
  const base = SCHEMA_NODES.find((s) => s.id === id) || {};
  const t = TYPES[id];
  let count = 0;
  if (DATA) {
    if (id === "LinkAssertion") count = DATA.edges.filter((e) => e.kind === "AI_LINK").length;
    else count = DATA.nodes.filter((n) => n.type === id).length;
  }
  return {
    id,
    label: base.label || (t ? t.label : id),
    desc: base.desc || (t ? t.desc : ""),
    color: t ? t.color : (id === "LinkAssertion" ? "#555" : "#999"),
    count,
    active: count > 0,
  };
}

function renderSchema() {
  $("stats").hidden = true;
  const info = Object.fromEntries(SCHEMA_NODES.map((s) => [s.id, classInfo(s.id)]));
  const pos = Object.fromEntries(SCHEMA_NODES.map((s) => [s.id, s]));
  let html = "";

  // --- Relations (drawn first, under the boxes) ----------------------
  SCHEMA_EDGES.forEach(([a, b, label]) => {
    const A = pos[a], B = pos[b];
    const x1 = A.x + SCHEMA_W / 2, y1 = A.y + SCHEMA_H / 2;
    const x2 = B.x + SCHEMA_W / 2, y2 = B.y + SCHEMA_H / 2;
    const on = info[a].active && info[b].active;
    html += `<line class="sedge ${on ? "" : "off"}" x1="${x1}" y1="${y1}" x2="${x2}" y2="${y2}" marker-end="url(#arrow)"/>`;
    html += `<text class="slabel ${on ? "" : "off"}" x="${(x1 + x2) / 2}" y="${(y1 + y2) / 2 - 4}" text-anchor="middle">${esc(label)}</text>`;
  });

  // --- Classes: active ones are clickable, inactive ones are not -----
  SCHEMA_NODES.forEach((s) => {
    const c = info[s.id];
    const isSel = selected && selected.kind === "class" && selected.id === s.id;
    html += `<g class="cls ${c.active ? "" : "off"} ${isSel ? "sel" : ""}" ${c.active ? `data-class="${s.id}"` : ""}
                transform="translate(${s.x},${s.y})">
      <title>${esc(c.desc)}</title>
      <rect width="${SCHEMA_W}" height="${SCHEMA_H}" rx="6"/>
      ${c.active ? `<rect width="6" height="${SCHEMA_H}" rx="2" fill="${c.color}" stroke="none"/>` : ""}
      <text x="16" y="23" font-weight="600">${esc(c.label)}</text>
      <text class="cnt" x="16" y="41">${c.active ? `${c.count} in sample` : "no data linked yet"}</text>
    </g>`;
  });

  vp.innerHTML = arrowDefs() + html;
}

function arrowDefs() {
  return `<defs><marker id="arrow" viewBox="0 0 10 10" refX="10" refY="5" markerWidth="7" markerHeight="7"
           orient="auto-start-reverse"><path d="M0,0 L10,5 L0,10 z" fill="#8A8A8A"/></marker></defs>`;
}


/* =====================================================================
   7. DATA GRAPH VIEW
   ---------------------------------------------------------------------
   Layout: two lanes.
     Regulatory chain (top) : Alert | RGL | RSM | ROB | Policy+Procedure | Gaps
     Risk & control (bottom): Alert | LR  | L1C | 1C  |                  | Gaps
   Within a column, nodes are ordered next to their parents to keep
   lines short. The alert sits between the two lanes.
   ===================================================================== */

const COL_W = 270, ROW_H = 36, NODE_W = 220, NODE_H = 28, LANE_GAP = 90, TOP = 50;

function visibleSet() {
  if (!DATA) return new Set();
  if (currentAlert === "ALL") return new Set(DATA.nodes.map((n) => n.id));
  // Everything reachable from the alert along outgoing links.
  const seen = new Set([currentAlert]);
  const queue = [currentAlert];
  while (queue.length) {
    const id = queue.shift();
    (outEdges[id] || []).forEach((e) => {
      if (!seen.has(e.target)) { seen.add(e.target); queue.push(e.target); }
    });
  }
  return seen;
}

function laneOf(id) {
  const n = nodeById[id];
  if (n.type === "Gap") {
    const parent = (inEdges[id] || [])[0];
    return parent ? laneOf(parent.source) : "reg";
  }
  return (TYPES[n.type] || { lane: "reg" }).lane;
}

// Edges actually drawn: hide Alert -> X when X already hangs off another
// parent in view (the proof stays available in the side panel).
function drawnEdges(vis) {
  return DATA.edges.filter((e) => {
    if (!vis.has(e.source) || !vis.has(e.target)) return false;
    if (nodeById[e.source].type !== "Alert") return true;
    return !(inEdges[e.target] || []).some(
      (p) => vis.has(p.source) && nodeById[p.source].type !== "Alert");
  });
}

function layoutGraph(vis, edges) {
  positions = {};
  const cols = {};                    // "lane|col" -> [ids]
  const alerts = [];
  vis.forEach((id) => {
    const n = nodeById[id];
    if (n.type === "Alert") { alerts.push(id); return; }
    const key = laneOf(id) + "|" + ((TYPES[n.type] || {}).col ?? 5);
    (cols[key] = cols[key] || []).push(id);
  });

  // Parent order index, used to sort children near their parents.
  const order = {};
  alerts.sort().forEach((id, i) => { order[id] = i / Math.max(1, alerts.length); });
  const parentsOf = (id) => edges.filter((e) => e.target === id).map((e) => e.source);
  const typeRank = { Policy: 0, Procedure: 1 };

  ["reg", "ctl"].forEach((lane) => {
    for (let c = 1; c <= 5; c++) {
      const list = cols[lane + "|" + c];
      if (!list) continue;
      const bary = (id) => {
        const ps = parentsOf(id).filter((p) => p in order);
        return ps.length ? ps.reduce((s, p) => s + order[p], 0) / ps.length : 1;
      };
      list.sort((a, b) =>
        (typeRank[nodeById[a].type] ?? 0) - (typeRank[nodeById[b].type] ?? 0) ||
        bary(a) - bary(b) ||
        String(nodeById[a].title).localeCompare(String(nodeById[b].title)));
      list.forEach((id, i) => { order[id] = i / list.length; });
    }
  });

  // Lane heights and y positions (each column centred in its lane).
  const laneHeight = (lane) => Math.max(1, ...[1, 2, 3, 4, 5].map((c) => (cols[lane + "|" + c] || []).length)) * ROW_H;
  const regH = laneHeight("reg"), ctlH = laneHeight("ctl");
  const regTop = TOP, ctlTop = TOP + regH + LANE_GAP;
  const lanes = { reg: { top: regTop, h: regH }, ctl: { top: ctlTop, h: ctlH } };

  Object.entries(cols).forEach(([key, list]) => {
    const [lane, c] = key.split("|");
    const L = lanes[lane];
    const offset = (L.h - list.length * ROW_H) / 2;
    list.forEach((id, i) => { positions[id] = { x: 40 + Number(c) * COL_W, y: L.top + offset + i * ROW_H }; });
  });

  const midY = regTop + regH + LANE_GAP / 2 - NODE_H / 2;
  alerts.forEach((id, i) => {
    positions[id] = { x: 40, y: midY - ((alerts.length - 1) / 2) * ROW_H * 1.3 + i * ROW_H * 1.3 };
  });

  return { cols, lanes };
}

function renderGraph() {
  if (!DATA) { renderSchema(); return; }
  const vis = visibleSet();
  const edges = drawnEdges(vis);
  const { cols, lanes } = layoutGraph(vis, edges);

  // --- Highlight set for the current selection ------------------------
  let focus = null;
  if (selected && selected.kind === "node" && vis.has(selected.id)) {
    focus = new Set([selected.id]);
    [...(inEdges[selected.id] || []), ...(outEdges[selected.id] || [])]
      .forEach((e) => { focus.add(e.source); focus.add(e.target); });
  } else if (selected && selected.kind === "edge") {
    const e = DATA.edges[selected.id];
    focus = new Set([e.source, e.target]);
  }

  let html = arrowDefs();

  // --- Lane backgrounds and titles -------------------------------------
  const width = 40 + 6 * COL_W;
  [["reg", "Regulatory chain · RegMap + GPPS"], ["ctl", "Risk & control chain · Helios"]].forEach(([lane, title]) => {
    const L = lanes[lane];
    html += `<rect class="lane-bg" x="${COL_W - 30}" y="${L.top - 34}" width="${width - COL_W + 10}" height="${L.h + 50}" rx="8"/>`;
    html += `<text class="lane-title" x="${COL_W - 14}" y="${L.top - 14}">${title}</text>`;
    for (let c = 1; c <= 5; c++) {
      const list = cols[lane + "|" + c];
      if (!list) continue;
      const types = [...new Set(list.map((id) => typeLabel(nodeById[id].type)))].join(" / ");
      html += `<text class="col-title" x="${40 + c * COL_W}" y="${L.top - 14}" dx="${c === 1 ? 260 - COL_W + 160 : 0}">${esc(c === 1 ? "" : types)}</text>`;
    }
  });

  // --- Edges (with a wide invisible hit area for clicking) -------------
  edges.forEach((e) => {
    const a = positions[e.source], b = positions[e.target];
    if (!a || !b) return;
    const x1 = a.x + NODE_W, y1 = a.y + NODE_H / 2, x2 = b.x, y2 = b.y + NODE_H / 2;
    const dx = Math.max(40, (x2 - x1) / 2);
    const d = `M${x1},${y1} C${x1 + dx},${y1} ${x2 - dx},${y2} ${x2},${y2}`;
    const v = e.kind === "AI_LINK" ? verdictClass(e.judge_verdict) : "";
    const dim = focus && !(focus.has(e.source) && focus.has(e.target)) ? "dim" : "";
    const hl = selected && selected.kind === "edge" && selected.id === e._i ? "hl" : "";
    html += `<path class="edge ${e.kind} ${v ? "v-" + v : ""} ${dim} ${hl}" d="${d}"/>`;
    html += `<path class="edge-hit" data-edge="${e._i}" d="${d}"><title>${esc(REL_LABEL[e.relation] || e.relation)}</title></path>`;
  });

  // --- Nodes ------------------------------------------------------------
  vis.forEach((id) => {
    const n = nodeById[id], p = positions[id];
    if (!p) return;
    const dim = focus && !focus.has(id) ? "dim" : "";
    const sel = selected && selected.kind === "node" && selected.id === id ? "sel" : "";
    const gap = n.type === "Gap" ? "gap" : "";
    // Verdict dot: judge result on the alert's direct link to this node.
    const ai = (inEdges[id] || []).find((e) => e.kind === "AI_LINK" && vis.has(e.source) && !blank(e.judge_verdict));
    const vc = ai ? verdictClass(ai.judge_verdict) : "";
    const label = n.title && n.title !== id ? n.title : id;
    const w = n.type === "Alert" ? NODE_W : NODE_W;
    html += `<g class="node ${dim} ${sel} ${gap}" data-node="${esc(id)}" transform="translate(${p.x},${p.y})">
      <title>${esc(typeLabel(n.type))}: ${esc(id)}${n.title && n.title !== id ? "\n" + esc(n.title) : ""}</title>
      <rect class="box" width="${w}" height="${NODE_H}" rx="5"/>
      <rect width="5" height="${NODE_H}" rx="2" fill="${typeColor(n.type)}"/>
      <text x="12" y="${NODE_H / 2 + 4}" ${n.type === "Alert" ? 'font-weight="700"' : ""}>${esc(short(label, 30))}</text>
      ${vc ? `<circle cx="${w - 10}" cy="${NODE_H / 2}" r="5" fill="var(--${vc})"/>` : ""}
    </g>`;
  });

  vp.innerHTML = html;
  renderStats(vis);
}

function renderStats(vis) {
  const counts = {};
  vis.forEach((id) => { const t = nodeById[id].type; counts[t] = (counts[t] || 0) + 1; });
  const ai = DATA.edges.filter((e) => e.kind === "AI_LINK" && vis.has(e.source) && vis.has(e.target));
  const v = { agree: 0, disagree: 0, uncertain: 0, "": 0 };
  ai.forEach((e) => { v[verdictClass(e.judge_verdict)]++; });
  const order = ["Alert", "RGL", "RSM", "ROB", "Policy", "Procedure", "LR", "L1C", "1C", "Gap"];
  $("stats").innerHTML =
    order.filter((t) => counts[t]).map((t) => `${typeLabel(t)} <b>${counts[t]}</b>`).join(" · ") +
    `<br>AI links <b>${ai.length}</b>: ` +
    `<span class="verdict agree">${v.agree} confirmed</span> · ` +
    `<span class="verdict disagree">${v.disagree} disputed</span> · ` +
    `<span class="verdict uncertain">${v.uncertain} to SME</span> · ${v[""]} not judged`;
  $("stats").hidden = false;
}

function renderLegend() {
  if (view === "schema") {
    $("legend").innerHTML =
      `<b>Ontology view</b><br>Coloured class = data in the sample (click it)<br>` +
      `Grey dashed class = defined, no data linked yet`;
    return;
  }
  $("legend").innerHTML =
    `<span class="sw" style="border-color:var(--agree)"></span>AI link · judge confirms<br>` +
    `<span class="sw" style="border-color:var(--disagree)"></span>AI link · judge disputes<br>` +
    `<span class="sw" style="border-color:#555"></span>AI link · not judged<br>` +
    `<span class="sw" style="border-color:var(--struct);border-top-style:dotted"></span>Structural (RegMap / Helios)<br>` +
    `<span class="sw" style="border-color:var(--disagree);border-top-style:dashed"></span>Gap finding (rule)<br>` +
    `<span class="muted">Click any box or line to see the evidence</span>`;
}

function openPanel(html) { $("panelBody").innerHTML = html; $("panel").classList.add("open"); }
function closePanel() { $("panel").classList.remove("open"); }
$("closePanel").onclick = () => { selected = null; closePanel(); render(false); };

function proofCard(e) {
  const src = nodeById[e.source], tgt = nodeById[e.target];
  const vc = verdictClass(e.judge_verdict);
  let h = `<div class="proof">
    <div><b>${esc(src.title || src.id)}</b> → <i>${esc(REL_LABEL[e.relation] || e.relation)}</i> → <b>${esc(tgt.title || tgt.id)}</b></div>
    <div class="muted">${esc(KIND_LABEL[e.kind] || e.kind)}</div>`;
  if (e.kind === "AI_LINK") {
    h += `<div class="row" style="margin-top:6px">
      ${!blank(e.score) ? `<span>Score <b>${esc(e.score)}</b>/10</span>` : ""}
      ${!blank(e.confidence) ? `<span>Confidence <b>${esc(e.confidence)}</b></span>` : ""}
      <span class="verdict ${vc}">${verdictText(e.judge_verdict)}</span>
      ${!blank(e.grounding) ? `<span>Grounding <b>${esc(e.grounding)}</b></span>` : ""}
      ${!blank(e.route) ? `<span>Route <b>${esc(e.route)}</b></span>` : ""}
      ${!blank(e.tier) ? `<span>Tier <b>${esc(e.tier)}</b></span>` : ""}
    </div>`;
    if (!blank(e.sections)) h += `<div class="row"><span>Linked sections <b>${esc(e.sections)}</b></span></div>`;
    if (!blank(e.evidence_from)) h += `<div class="quote">“${esc(e.evidence_from)}”<div class="muted">from ${esc(typeLabel(src.type).toLowerCase())}</div></div>`;
    if (!blank(e.evidence_to)) h += `<div class="quote to">“${esc(e.evidence_to)}”<div class="muted">from ${esc(typeLabel(tgt.type).toLowerCase())}</div></div>`;
    if (!blank(e.reasoning)) h += `<div style="margin-top:6px"><b>Reasoning.</b> ${esc(e.reasoning)}</div>`;
  }
  return h + `</div>`;
}

function showNode(id) {
  const n = nodeById[id];
  const ins = inEdges[id] || [], outs = outEdges[id] || [];
  const why = ins.filter((e) => e.kind === "AI_LINK");
  const conn = (e, dir) => {
    const other = nodeById[dir === "in" ? e.source : e.target];
    return `<div class="conn">${dir === "in" ? "←" : "→"} ${esc(REL_LABEL[e.relation] || e.relation)}
      <span class="link-chip" data-goto="${esc(other.id)}">${esc(short(other.title || other.id, 40))}</span>
      ${e.kind === "AI_LINK" ? `<span class="link-chip" data-proof="${e._i}">evidence</span>` : ""}</div>`;
  };
  let h = `<span class="chip" style="background:${typeColor(n.type)}">${esc(typeLabel(n.type))}</span>
    <span class="muted">${esc(n.id)}</span>
    <h2>${esc(n.title || n.id)}</h2>`;
  if (!blank(n.text)) h += `<div class="text">${esc(n.text)}</div>`;
  else h += `<div class="muted">${esc((TYPES[n.type] || {}).desc || "")}</div>`;
  if (why.length) h += `<h3>Why this is linked</h3>` + why.map(proofCard).join("");
  if (outs.length) h += `<h3>Links onward (${outs.length})</h3>` + outs.map((e) => conn(e, "out")).join("");
  const otherIns = ins.filter((e) => e.kind !== "AI_LINK");
  if (otherIns.length) h += `<h3>Sits under</h3>` + otherIns.map((e) => conn(e, "in")).join("");
  openPanel(h);
}

function showEdge(i) {
  const e = DATA.edges[i];
  openPanel(`<h2>Link evidence</h2>` + proofCard(e) +
    `<h3>Ends</h3><span class="link-chip" data-goto="${esc(e.source)}">${esc(nodeById[e.source].title || e.source)}</span>
     <span class="link-chip" data-goto="${esc(e.target)}">${esc(nodeById[e.target].title || e.target)}</span>`);
}

function showClass(id) {
  const c = classInfo(id);
  const rel = SCHEMA_EDGES.filter(([a, b]) => a === id || b === id)
    .map(([a, b, l]) => `<div class="conn">${esc(classInfo(a).label)} → <i>${esc(l)}</i> → ${esc(classInfo(b).label)}</div>`).join("");
  let h = `<span class="chip" style="background:${c.color}">Class</span><h2>${esc(c.label)}</h2>
    <div>${esc(c.desc)}</div><h3>Relations</h3>${rel}`;
  if (DATA && id !== "LinkAssertion") {
    const inst = DATA.nodes.filter((n) => n.type === id);
    h += `<h3>In the sample (${inst.length})</h3>` +
      inst.slice(0, 80).map((n) => `<span class="link-chip" data-goto="${esc(n.id)}">${esc(short(n.title || n.id, 40))}</span>`).join("");
  }
  if (id === "LinkAssertion" && DATA) {
    h += `<h3>In the sample</h3><div>${c.count} AI links, each with score, evidence quotes, reasoning and judge verdict. Open the data graph and click any line.</div>`;
  }
  openPanel(h);
}

// Clicks inside the panel: jump to a node, or open a link's evidence.
$("panelBody").addEventListener("click", (ev) => {
  const g = ev.target.closest("[data-goto]");
  const p = ev.target.closest("[data-proof]");
  if (g) gotoNode(g.dataset.goto);
  else if (p) { selected = { kind: "edge", id: Number(p.dataset.proof) }; showEdge(Number(p.dataset.proof)); render(false); }
});

function gotoNode(id) {
  if (view !== "graph") setView("graph");
  if (!visibleSet().has(id)) { currentAlert = "ALL"; $("alertSelect").value = "ALL"; }
  selected = { kind: "node", id };
  render(false);
  showNode(id);
  centerOn(id);
}


/* =====================================================================
   9. CANVAS INTERACTION: click, pan, zoom, fit, search
   ===================================================================== */

let drag = null, moved = false;

svg.addEventListener("pointerdown", (e) => {
  drag = { x: e.clientX, y: e.clientY, tx: T.x, ty: T.y };
  moved = false;
});
window.addEventListener("pointermove", (e) => {
  if (!drag) return;
  const dx = e.clientX - drag.x, dy = e.clientY - drag.y;
  if (Math.abs(dx) + Math.abs(dy) > 4) { moved = true; svg.classList.add("dragging"); }
  if (moved) { T.x = drag.tx + dx; T.y = drag.ty + dy; applyT(); }
});
window.addEventListener("pointerup", () => { drag = null; svg.classList.remove("dragging"); });

svg.addEventListener("click", (e) => {
  if (moved) return;                       // it was a pan, not a click
  const n = e.target.closest("[data-node]");
  const ed = e.target.closest("[data-edge]");
  const c = e.target.closest("[data-class]");
  if (n) { selected = { kind: "node", id: n.dataset.node }; showNode(n.dataset.node); }
  else if (ed) { selected = { kind: "edge", id: Number(ed.dataset.edge) }; showEdge(Number(ed.dataset.edge)); }
  else if (c) { selected = { kind: "class", id: c.dataset.class }; showClass(c.dataset.class); }
  else { selected = null; closePanel(); }
  render(false);
});

svg.addEventListener("wheel", (e) => {
  e.preventDefault();
  const r = svg.getBoundingClientRect();
  zoomAt(e.clientX - r.left, e.clientY - r.top, Math.exp(-e.deltaY * 0.0015));
}, { passive: false });

function zoomAt(mx, my, f) {
  const k = Math.min(3, Math.max(0.15, T.k * f));
  T.x = mx - (mx - T.x) * (k / T.k);
  T.y = my - (my - T.y) * (k / T.k);
  T.k = k;
  applyT();
}
function applyT() { vp.setAttribute("transform", `translate(${T.x},${T.y}) scale(${T.k})`); }

function fit() {
  const bb = vp.getBBox(), r = svg.getBoundingClientRect();
  if (!bb.width || !bb.height) return;
  const k = Math.min((r.width - 60) / bb.width, (r.height - 60) / bb.height, 1.3);
  T = { k, x: (r.width - bb.width * k) / 2 - bb.x * k, y: (r.height - bb.height * k) / 2 - bb.y * k };
  applyT();
}
function centerOn(id) {
  const p = positions[id];
  if (!p) return;
  const r = svg.getBoundingClientRect();
  const k = Math.max(T.k, 0.9);
  T = { k, x: r.width * 0.4 - (p.x + NODE_W / 2) * k, y: r.height / 2 - (p.y + NODE_H / 2) * k };
  applyT();
}

$("zoomIn").onclick = () => { const r = svg.getBoundingClientRect(); zoomAt(r.width / 2, r.height / 2, 1.25); };
$("zoomOut").onclick = () => { const r = svg.getBoundingClientRect(); zoomAt(r.width / 2, r.height / 2, 0.8); };
$("fitBtn").onclick = fit;
window.addEventListener("resize", () => fit());
document.addEventListener("keydown", (e) => { if (e.key === "Escape") { selected = null; closePanel(); render(false); } });

$("search").addEventListener("keydown", (e) => {
  if (e.key !== "Enter" || !DATA) return;
  const q = e.target.value.trim().toLowerCase();
  if (!q) return;
  const hit = DATA.nodes.find((n) => n.id.toLowerCase().includes(q)) ||
              DATA.nodes.find((n) => String(n.title || "").toLowerCase().includes(q));
  if (hit) gotoNode(hit.id); else alert("No node matches: " + q);
});
