"""Run logging. One folder per run: <log_dir>/<run_id>/<YYYYmmdd_HHMMSS>/

  run.log            everything (console lines + DEBUG detail + tracebacks)
  errors.jsonl       one JSON line per warning / error: time, code, stage, alert, doc, attempt, detail
  llm_calls.jsonl    one JSON line per gateway call: time, stage, alert, doc, attempt, status, code, ms, tokens
  payloads.jsonl     (only if logging.log_payloads) full model input and answer text - sensitive, debug only
  config_used.yaml   the effective config, gateway secrets masked
  run_summary.json   written at the end: stages, counts, cost, error-code counts, exit code
"""
import os
import sys
import json
import time
import logging
import datetime
from collections import Counter

from .errors import ERRORS


class RunLog:
    def __init__(self, log_dir, run_id, level="INFO", log_llm_calls=True, log_payloads=False):
        stamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        self.dir = os.path.join(log_dir, run_id, stamp)
        try:
            os.makedirs(self.dir, exist_ok=True)
        except OSError:
            # last resort: a writable folder, so that a bad log path never stops the diagnosis
            self.dir = os.path.join(os.getcwd(), "ia_pp_logs", run_id, stamp)
            os.makedirs(self.dir, exist_ok=True)
        self.logger = logging.getLogger(f"ia_pp.{run_id}.{stamp}")
        self.logger.setLevel(logging.DEBUG)
        self.logger.propagate = False
        fh = logging.FileHandler(os.path.join(self.dir, "run.log"), encoding="utf-8")
        fh.setLevel(logging.DEBUG)
        fh.setFormatter(logging.Formatter("%(asctime)s %(levelname)-7s %(message)s"))
        ch = logging.StreamHandler(sys.stdout)
        ch.setLevel(getattr(logging, level, logging.INFO))
        ch.setFormatter(logging.Formatter("%(asctime)s %(message)s", "%H:%M:%S"))
        self.logger.addHandler(fh)
        self.logger.addHandler(ch)
        self._err = open(os.path.join(self.dir, "errors.jsonl"), "a", encoding="utf-8")
        self._calls = open(os.path.join(self.dir, "llm_calls.jsonl"), "a", encoding="utf-8") if log_llm_calls else None
        self._payloads = open(os.path.join(self.dir, "payloads.jsonl"), "a", encoding="utf-8") if log_payloads else None
        self.codes = Counter()
        self.stage = "setup"
        self.t0 = time.time()

    # ---- plain lines --------------------------------------------------------------------------------------
    def info(self, msg):
        self.logger.info(msg)

    def debug(self, msg):
        self.logger.debug(msg)

    # ---- classified events --------------------------------------------------------------------------------
    def event(self, code, detail="", echo=True, **ctx):
        """Record a warning / error with its code. echo=False keeps per-pair noise out of the console."""
        self.codes[code] += 1
        rec = {"ts": datetime.datetime.now().isoformat(timespec="seconds"), "code": code,
               "title": ERRORS[code].title if code in ERRORS else "", "stage": self.stage,
               "detail": str(detail)[:2000], **{k: _safe(v) for k, v in ctx.items()}}
        try:
            self._err.write(json.dumps(rec, ensure_ascii=False) + "\n")
            self._err.flush()
        except Exception:
            pass
        line = f"[{code}] {rec['title']}: {str(detail)[:500]}" + (f" | {ctx}" if ctx and echo else "")
        if echo:
            (self.logger.warning if code.startswith("W") else self.logger.error)(line)
        else:
            self.logger.debug(line)

    def exception(self, code, exc, **ctx):
        """Unexpected exception: classified event + full traceback in run.log."""
        self.event(code, f"{type(exc).__name__}: {exc}", **ctx)
        self.logger.debug("traceback", exc_info=exc)

    def llm_call(self, **rec):
        if self._calls is None:
            return
        try:
            rec = {"ts": round(time.time(), 3), "stage": self.stage, **{k: _safe(v) for k, v in rec.items()}}
            self._calls.write(json.dumps(rec, ensure_ascii=False) + "\n")
        except Exception:
            pass

    def payload(self, **rec):
        if self._payloads is None:
            return
        try:
            self._payloads.write(json.dumps({k: _safe(v, 10 ** 7) for k, v in rec.items()}, ensure_ascii=False) + "\n")
        except Exception:
            pass

    def write_json(self, name, obj):
        try:
            with open(os.path.join(self.dir, name), "w", encoding="utf-8") as f:
                json.dump(obj, f, indent=2, ensure_ascii=False, default=str)
        except Exception as e:
            self.logger.error(f"could not write {name}: {e}")

    def write_yaml(self, name, obj):
        try:
            import yaml
            with open(os.path.join(self.dir, name), "w", encoding="utf-8") as f:
                yaml.safe_dump(obj, f, sort_keys=False, allow_unicode=True)
        except Exception as e:
            self.logger.error(f"could not write {name}: {e}")

    def close(self):
        for f in (self._err, self._calls, self._payloads):
            try:
                if f:
                    f.flush()
                    f.close()
            except Exception:
                pass
        for h in list(self.logger.handlers):
            try:
                h.flush()
                h.close()
            except Exception:
                pass
            self.logger.removeHandler(h)


def _safe(v, limit=2000):
    if v is None or isinstance(v, (bool, int, float)):
        return v
    s = str(v)
    return s if len(s) <= limit else s[:limit] + "..."

def write_results(s, ctx, res, prompts, log, usage_report):
    out = s.output_dir
    titles = dict(zip(ctx.alerts.alert, ctx.alerts.title))
    r = res.copy()
    r["doc_applicability"] = r.doc.map(ctx.doc_applic).fillna("")
    r["doc_title"] = r.doc.map(ctx.doc_title)
    r["alert_title"] = r.alert.map(titles)
    r["linked_sections"] = r.linked_units.fillna("").map(
        lambda u: " | ".join(ctx.unit_info[x]["gpps_label"] for x in str(u).split("; ") if x in ctx.unit_info))
    r["run_id"] = s.run_id
    pairs = r.rename(columns={"alert": "RECORD_ID", "doc": "DOC_ID", "doc_type": "DOC_TYPE"})
    for c in PAIR_COLUMNS:
        if c not in pairs.columns:
            pairs[c] = None
    pairs = pairs[PAIR_COLUMNS]
    write_parquet(pairs, os.path.join(out, "ia_pp_pairs.parquet"))

    srows = []
    for p in pairs[pairs.impact_status.isin(SHOWN)].itertuples():
        for u in str(p.linked_units or "").split("; "):
            if u in ctx.unit_info:
                I = ctx.unit_info[u]
                srows.append(dict(RECORD_ID=p.RECORD_ID, DOC_TYPE=p.DOC_TYPE, DOC_ID=p.DOC_ID, doc_title=p.doc_title,
                                  doc_scope=p.doc_scope, impact_status=p.impact_status, unit_id=u,
                                  section_no=I["section_no"], section_title=I["section_title"],
                                  subsection_no=I["subsection_no"], subsection_title=I["subsection_title"],
                                  gpps_label=I["gpps_label"], quote_in_this_section=quote_found(p.evidence_from_target, I["text"]),
                                  impact_type=p.impact_type, link_score=p.link_score, change_needed=p.change_needed))
    write_parquet(pd.DataFrame(srows, columns=["RECORD_ID", "DOC_TYPE", "DOC_ID", "doc_title", "doc_scope", "impact_status",
                                               "unit_id", "section_no", "section_title", "subsection_no", "subsection_title",
                                               "gpps_label", "quote_in_this_section", "impact_type", "link_score",
                                               "change_needed"]),
                  os.path.join(out, "ia_pp_impacted_sections.parquet"))

    def items(g):
        return [{"doc_id": x.DOC_ID, "title": x.doc_title, "scope": x.doc_scope, "role": x.document_role,
                 "impact_status": x.impact_status, "confidence_tier": x.confidence_tier,
                 "link_score": None if pd.isna(x.link_score) else int(x.link_score),
                 "impact_type": _j(x.impact_type), "change_needed": _j(x.change_needed),
                 "sections": [{k: ctx.unit_info[u][k] for k in ("section_no", "section_title", "subsection_no",
                                                                "subsection_title", "gpps_label")}
                              for u in str(x.linked_units or "").split("; ") if u in ctx.unit_info],
                 "section_grounded": bool(x.section_grounded) if x.section_grounded is not None else False,
                 "evidence_from_alert": _j(x.evidence_from_alert), "evidence_from_target": _j(x.evidence_from_target),
                 "reasoning": _j(x.reasoning)}
                for x in g.sort_values("link_score", ascending=False).itertuples()]

    shown = pairs[pairs.impact_status.isin(SHOWN)]
    lines = []
    for a in ctx.alerts.alert:
        g = shown[shown.RECORD_ID == a]
        sel = lambda st, t: g[(g.impact_status == st) & (g.DOC_TYPE == t)]
        lines.append(json.dumps({
            "record_id": a, "alert_title": titles.get(a), "run_id": s.run_id,
            "counts": {st.lower(): {t: int(((g.impact_status == st) & (g.DOC_TYPE == t)).sum())
                                    for t in ("policy", "procedure")} for st in SHOWN},
            "impacted": {"policies": items(sel("IMPACTED", "policy")), "procedures": items(sel("IMPACTED", "procedure"))},
            "possible_impact": {"policies": items(sel("POSSIBLE_IMPACT", "policy")),
                                "procedures": items(sel("POSSIBLE_IMPACT", "procedure"))},
            "scope_review": items(g[g.impact_status == "SCOPE_REVIEW"])}, ensure_ascii=False, default=str))
    write_text("\n".join(lines) + "\n", os.path.join(out, "ia_pp_alert_impacts.jsonl"))

    n_alerts = len(ctx.alerts)
    summary = []
    for kind in ("policy", "procedure"):
        pk = pairs[pairs.DOC_TYPE == kind]
        imp, pos = pk[pk.impact_status == "IMPACTED"], pk[pk.impact_status == "POSSIBLE_IMPACT"]
        both = pk[pk.impact_status.isin(["IMPACTED", "POSSIBLE_IMPACT"])]
        summary.append(dict(doc_type=kind, pairs_read=len(pk), no_answer=int((pk.final_decision == "ERROR").sum()),
                            impacted=len(imp), impacted_per_alert=round(len(imp) / max(n_alerts, 1), 1),
                            alerts_with_impacted=imp.RECORD_ID.nunique(), possible_impact=len(pos),
                            possible_per_alert=round(len(pos) / max(n_alerts, 1), 1),
                            alerts_with_any=both.RECORD_ID.nunique(),
                            scope_review=int((pk.impact_status == "SCOPE_REVIEW").sum()),
                            framework_links=int((both.document_role == "FRAMEWORK").sum()),
                            change_required_pct=round(100 * both.impact_type.astype(str).eq("CHANGE_REQUIRED").mean(), 1)
                            if len(both) else np.nan,
                            section_grounded_pct=round(100 * both.section_grounded.astype(str).eq("True").mean(), 1)
                            if len(both) else np.nan))
    summary = pd.DataFrame(summary)
    log.info("RESULTS " + s.run_id + f" | {n_alerts} alerts | IMPACTED = link_score >= {s.rules['headline_min_score']} | "
             f"POSSIBLE_IMPACT = {s.rules['link_threshold']}-{s.rules['headline_min_score'] - 1}\n"

    try:
        by_scope = shown.groupby(["DOC_TYPE", "doc_scope", "impact_status"]).size().unstack(fill_value=0).reset_index()
        per_alert = shown.pivot_table(index="RECORD_ID", columns=["DOC_TYPE", "impact_status"], values="DOC_ID",

        per_alert.insert(1, "alert_title", per_alert.RECORD_ID.map(titles))
        fw = pairs[pairs.document_role == "FRAMEWORK"].groupby(["DOC_ID", "doc_title"]).agg(
            pairs_read=("RECORD_ID", "size"), model_linked=("model_decision", lambda x: int((x == "LINKED").sum())),
            process_change_required=("impact_status", lambda x: int(x.isin(["IMPACTED", "POSSIBLE_IMPACT"]).sum()))
        ).reset_index()
        errs = pairs[pairs.final_decision == "ERROR"].groupby(["DOC_TYPE", "error_code"]).size().rename("pairs").reset_index()
        with pd.ExcelWriter(os.path.join(out, "ia_pp_results_summary.xlsx")) as xl:
            pd.DataFrame({"How to read": [
                f"Run {s.run_id}: every policy read for every alert; procedures = search top {s.search['proc_top_k']} per "
                "alert, each read in full. ONE model call per (alert, document).",
                f"IMPACTED = model LINKED with link_score >= {s.rules['headline_min_score']} and both verbatim quotes "
                "verified by code.",
                f"POSSIBLE_IMPACT = model LINKED with link_score {s.rules['link_threshold']}-"
                f"{s.rules['headline_min_score'] - 1}, quotes verified - SME review queue.",
                f"SCOPE_REVIEW = linked to another country's LOCAL document: a {s.country['name']} alert should not "
                "impact it unless that country adopts the requirement.",
                "Framework / process documents count only when the alert requires the process itself to change.",
                "no_answer / sheet 'pairs without answer': re-run the same command to retry them (see error codes)."]}
            ).to_excel(xl, sheet_name="How to read", index=False)
            summary.to_excel(xl, sheet_name="summary", index=False)
            per_alert.to_excel(xl, sheet_name="per alert", index=False)
            by_scope.to_excel(xl, sheet_name="by scope", index=False)
            fw.to_excel(xl, sheet_name="framework documents", index=False)
            errs.to_excel(xl, sheet_name="pairs without answer", index=False)
            ctx.scope.to_excel(xl, sheet_name="document scope", index=False)
    except Exception as e:
        log.event("W-OUT-001", f"{type(e).__name__}: {e}")

    # run record: exact prompts, schemas, exceptions and (masked) settings
    prod = os.path.join(out, "production")
    os.makedirs(prod, exist_ok=True)
    for key, (ppath, spath) in prompts.files.items():
        for src in (ppath, spath):
            try:
                shutil.copy2(src, os.path.join(prod, os.path.basename(src)))
            except OSError as e:
                log.event("E-OUT-001", f"copy {src}: {e}")
    write_json({"run_id": s.run_id, "country": s.country_code, "country_name": s.country["name"],
                "n_alerts": n_alerts, "rules": s.rules, "search": s.search, "model": {k: v for k, v in s.model.items()},
                "framework_documents": ctx.framework_docs, "skipped_documents": ctx.skip_docs,
                "sources": ctx.sources_used, "usage": usage_report,
                "calls": "ONE per (alert, document) - no audit call in production"},
               os.path.join(prod, "production_config.json"))
    return summary

def check_environment(s):
    need = ["pandas", "numpy", "pyarrow", "sklearn", "yaml"]
    if s.gateway["mode"] == "live":
        need += ["openai", "httpx", "py_group_ai_tools"]
    missing = []
    for mod in need:
        try:
            __import__(mod)
        except ImportError:
            missing.append(mod)
    if missing:
        raise IAError("E-ENV-001", ", ".join(missing))


def _ledger(s):
    path = os.path.join(s.output_dir, "spend_ledger.json")
    try:
        with open(path, encoding="utf-8") as f:
            return path, json.load(f)
    except Exception:
        return path, {"total_usd": 0.0, "sessions": []}


def _pairs_frame(ctx, alerts, docs, kind, rank=None):
    rows = [dict(alert=a, doc=d, doc_type=kind, search_rank=float("nan")) for a in alerts for d in docs]
    df = pd.DataFrame(rows, columns=["alert", "doc", "doc_type", "search_rank"])
    df["doc_scope"] = df.doc.map(ctx.doc_scope).fillna("GLOBAL")
    return df


async def run_pipeline(s, log, stages, stop):
    """Returns the process exit code. Never raises IAError (they are logged and mapped to an exit code)."""
    t0 = time.time()
    usage = Usage(s.model["price_per_million_tokens"]["input"], s.model["price_per_million_tokens"]["output"])
    summary = {"run_id": s.run_id, "version": __version__, "started": datetime.datetime.now().isoformat(timespec="seconds"),
               "stages_requested": stages, "stages_done": [], "gateway_mode": s.gateway["mode"],
               "output_dir": s.output_dir, "log_dir": log.dir}
    ledger_path, ledger = None, None
    exit_code = EXIT_OK
    try:
        # ---------------------------------------------------------------- validate
        log.stage = "validate"
        log.info(f"IA policy / procedure pipeline v{__version__} | run {s.run_id} | gateway {s.gateway['mode']} | "
                 f"model {s.model['chat_model']} | stages {stages}")
        log.write_yaml("config_used.yaml", s.masked())
        check_environment(s)
        ensure_dir(s.output_dir)
        ledger_path, ledger = _ledger(s)
        if not (s.country["terms"] or s.country["codes"] or s.country["entity_codes"]):
            log.event("W-CFG-001", f"country {s.country_code}: terms, codes and entity_codes are empty")
        prompts = PromptStore(s)
        log.info(f"  prompts and schemas OK: {len(prompts.text)} prompts")
        gw = make_gateway(s, log, usage)
        if s.gateway["mode"] == "live" and any(x in stages for x in ("gateway_check", "shortlist", "test_gate", "policies", "procedures")):
            gw.stage = log.stage = "gateway_check"
            r = await gw.chat("Reply with JSON only.", {"task": "return {\"ok\": true}"}, ref={"doc": "gateway_check"})
            if not r.ok:
                raise IAError(r.code if r.code == "E-GW-008" else "E-GW-011", f"{r.code}: {r.detail}")
            log.info("  gateway check OK")
        summary["stages_done"].append("validate")

        # ---------------------------------------------------------------- prepare
        log.stage = "prepare"
        ctx = build_context(s, log)
        ctx.atext_cache = ctx.atext
        compute_scope(s, ctx, log)
        alerts = list(ctx.alerts.alert)
        summary["stages_done"].append("prepare")
        summary["n_alerts"] = len(alerts)
        budget = Budget(s.run["max_cost_usd"], usage, ledger.get("total_usd", 0.0))

        pol_ckpt = os.path.join(s.output_dir, "policy_ckpt.jsonl")
        proc_ckpt = os.path.join(s.output_dir, "procedure_ckpt.jsonl")
        pol_cand = _pairs_frame(ctx, alerts, ctx.docs_of("policy"), "policy")

        # ---------------------------------------------------------------- shortlist
        proc_cand, deferred = None, None                    # deferred: a stage error that must not stop policies
        if any(x in stages for x in ("shortlist", "test_gate", "procedures")):
            log.stage = "shortlist"
            try:
                proc_cand = await build_procedure_shortlist(s, ctx, gw, prompts, log, stop)
                proc_cand["doc_scope"] = proc_cand.doc.map(ctx.doc_scope).fillna("GLOBAL")
                summary["stages_done"].append("shortlist")
            except IAError as e:
                if e.code in ("E-RUN-002", "E-RUN-001", "E-GW-008"):
                    raise
                log.event(e.code, e.detail + " | procedures are skipped in this session; policies continue")
                deferred = e
                stages = [x for x in stages if x != "procedures"]
        summary["pairs"] = {"policy": len(pol_cand), "procedure": None if proc_cand is None else len(proc_cand)}
        log.info(f"  pairs: policies {len(pol_cand):,} (all) | procedures "
                 f"{'-' if proc_cand is None else f'{len(proc_cand):,}'} (top-{s.search['proc_top_k']})")

        # ---------------------------------------------------------------- no-truncation proof + cost gate
        reading = [c for c, st in ((pol_cand, "policies"), (proc_cand, "procedures")) if st in stages and c is not None]
        if reading:
            cand = pd.concat(reading, ignore_index=True)
            short = [d for d in cand.doc.unique() if len(ctx.doc_text[d]) < ctx.doc_src_chars[d]]
            if short:
                raise IAError("E-RUN-006", f"{len(short)} document(s), e.g. {short[:5]}")
            done = set()
            for p in (pol_ckpt, proc_ckpt):
                if os.path.exists(p):
                    from .io_utils import JsonlCheckpoint
                    done |= {(r["alert"], r["doc"]) for r in JsonlCheckpoint(p).load()}
            left = cand[[(a, d) not in done for a, d in zip(cand.alert, cand.doc)]]
            chars = sum(len(ctx.doc_text[d]) + len(ctx.atext_cache[a]) for a, d in zip(left.alert, left.doc)) \
                + len(left) * (len(prompts.get("policy:BUSINESS")) + 600)

A prompt file is sent to the model VERBATIM as the system message, after one substitution: {{COUNTRY_NAME}}.
A schema file is the JSON Schema of that prompt's answer. Fields in 'required' must be valid (else the answer is

PLACEHOLDERS = {"COUNTRY_NAME"}
_LEFTOVER = re.compile(r"\{\{\s*([A-Za-z0-9_]+)\s*\}\}")


class PromptStore:
    def __init__(self, s):
        pdir, sdir = s.path(s.paths["prompts_dir"]), s.path(s.paths["schemas_dir"])
        values = {"COUNTRY_NAME": str(s.country["name"])}
        self.text, self.schema, self.files = {}, {}, {}
        for key, (pf, sf) in PROMPTS.items():
            ppath, spath = os.path.join(pdir, pf), os.path.join(sdir, sf)
            try:
                with open(ppath, encoding="utf-8") as f:

                raise IAError("E-PRM-001", f"{ppath} is empty")

            for name, v in values.items():
                raw = re.sub(r"\{\{\s*" + name + r"\s*\}\}", v, raw)
            if _LEFTOVER.search(raw):
                raise IAError("E-PRM-002", f"{ppath}: unresolved placeholder")
            try:
                with open(spath, encoding="utf-8") as f:
                    sch = json.load(f)
                assert isinstance(sch.get("properties"), dict)
            except Exception as e:
                raise IAError("E-SCH-001", f"{spath}: {type(e).__name__}: {e}")
            self.text[key], self.schema[key], self.files[key] = raw, sch, (ppath, spath)

        """-> (clean answer, error or None, [warnings]). Lenient coercion: '8' -> 8, 'linked' -> 'LINKED'."""
        sch = self.schema[key]
        if not isinstance(answer, dict) or not answer:
            return None, "answer is empty or not a JSON object", []
        out, warns = dict(answer), []
        required = set(sch.get("required", []))
        for name, spec in sch["properties"].items():
            if name not in out or out[name] is None:
                if name in required:
                    return None, f"required field '{name}' missing", warns
                continue
            ok, value = _coerce(out[name], spec)
            if ok:
                out[name] = value
            elif name in required:
                return None, f"required field '{name}' invalid: {str(out[name])[:80]!r}", warns
            else:
                warns.append(f"{name}={str(out[name])[:60]!r}")
                out[name] = None
        return out, None, warns


def _coerce(v, spec):
    types = spec.get("type")
    types = types if isinstance(types, list) else [types]
    if v is None:
        return ("null" in types), None
    if "enum" in spec:
        t = str(v).upper().strip().replace(" ", "_")
        t = next((e for e in spec["enum"] if t == e or t.startswith(e)), None)
        return (t is not None), t
    if "integer" in types:
        try:
            n = int(round(float(str(v).strip())))
        except (TypeError, ValueError):
            return False, None
        lo, hi = spec.get("minimum"), spec.get("maximum")
        if (lo is not None and n < lo) or (hi is not None and n > hi):
            return False, None
        return True, n
    if "array" in types:
        if isinstance(v, list):
            return True, [str(i).strip() for i in v if i is not None and str(i).strip()]
        if isinstance(v, str):
            return True, [x.strip() for x in re.split(r"[;\n]", v) if x.strip()]
        return False, None
    if "string" in types:
        return True, str(v)
    return True, v

"""Decision rules applied by code after the model call (identical to the validated notebook).

1. LINKED        = model LINKED + link_score >= rules.link_threshold + BOTH verbatim quotes found in the texts.
2. JURISDICTION  = a link to another country's LOCAL document -> SCOPE_REVIEW (never IMPACTED).
3. FRAMEWORK     = a framework / process document (exceptions/framework_documents.csv) also needs framework_change=YES.
4. BANDS         = IMPACTED (link_score >= rules.headline_min_score) / POSSIBLE_IMPACT (below) / NOT_IMPACTED.
"""
import re

import numpy as np
import pandas as pd


def clean_words(s):
    return re.sub(r"[^a-z0-9]+", " ", str(s or "").lower()).strip()


def quote_found(quote, text):
    """Verbatim (case / punctuation-insensitive); long quotes may match on their first or last 8 words."""
    q = clean_words(quote)
    if len(q) < 12 or q in ("null", "none"):
        return False
    t = clean_words(text)
    if q in t:
        return True
    w = q.split()
    return len(w) >= 6 and (" ".join(w[:8]) in t or " ".join(w[-8:]) in t)


def finalise(df, s, ctx):
    """Adds grounding_flag, section_grounded, decision_prob, final_decision (rule 1)."""
    o = df.copy()
    for c in ("link_score", "confidence_level", "search_rank"):
        if c in o.columns:
            o[c] = pd.to_numeric(o[c], errors="coerce")
    for c in ("evidence_from_alert", "evidence_from_target", "linked_units", "framework_change", "error_code",
              "error_detail", "impact_type", "change_needed", "reasoning"):
        if c not in o.columns:
            o[c] = None
    atext = ctx.atext_cache
    a_ok = np.array([quote_found(q, atext.get(a, "")) for q, a in zip(o["evidence_from_alert"], o["alert"])], dtype=bool)
    t_ok = np.array([quote_found(q, ctx.doc_text.get(d, "")) for q, d in zip(o["evidence_from_target"], o["doc"])], dtype=bool)
    s_ok = np.array([bool(u) and quote_found(q, " ".join(ctx.unit_info[x]["text"] for x in str(u).split("; ")
                                                         if x in ctx.unit_info))
                     for q, u in zip(o["evidence_from_target"], o["linked_units"].fillna(""))], dtype=bool)
    o["grounding_flag"] = np.select([a_ok & t_ok, a_ok | t_ok], ["GROUNDED", "PARTIAL"], default="UNGROUNDED")
    o["section_grounded"] = s_ok
    o["decision_prob"] = (o["link_score"] / 10).round(2)
    lk = (o["model_decision"] == "LINKED") & (o["link_score"].fillna(0) >= s.rules["link_threshold"])
    o["final_decision"] = np.select(
        [o["model_decision"] == "ERROR", lk & (o["grounding_flag"] == "GROUNDED"), lk,
         o["model_decision"] == "INSUFFICIENT_EVIDENCE"],
        ["ERROR", "LINKED", "INSUFFICIENT_EVIDENCE", "INSUFFICIENT_EVIDENCE"], default="NOT_LINKED")
    return o


def apply_impact_rules(res, s):
    """Adds impact_status and confidence_tier (rules 2-4)."""
    r = res.copy()
    linked = (r.final_decision == "LINKED") & ((r.document_role != "FRAMEWORK") |
                                                r.framework_change.astype(str).str.upper().str.startswith("YES"))
    other_country = r.doc_scope.eq("LOCAL_OTHER")
    r["impact_status"] = np.select([linked & other_country, linked & (r.link_score >= s.rules["headline_min_score"]), linked],
                                   ["SCOPE_REVIEW", "IMPACTED", "POSSIBLE_IMPACT"], default="NOT_IMPACTED")
    r["confidence_tier"] = np.select([linked & ((r.link_score >= 9) | (r.confidence_level >= 5)),
                                      linked & (r.link_score >= 8), linked], ["HIGH", "MEDIUM", "LOW"], default="")
    return r

"""Runs the (alert, document) reads: ONE model call per pair, a fixed pool of workers, checkpoint, progress, guard rails.

Guard rails
  * a failed pair never stops the run: it is saved as ERROR with its error_code and retried on the next run
  * every answer is checked against its response schema (retried if a required field is missing)
  * cost cap: the stage stops cleanly when the measured spend reaches run.max_cost_usd (E-RUN-001)
  * circuit breaker: N consecutive failed pairs stop the stage (E-GW-010); a wrong model / url stops it at once
  * stop signal (Ctrl+C / SIGTERM): workers finish the call in hand, the checkpoint is flushed, the run exits
"""
import re

EVIDENCE_FIELD = {"policy": "evidence_from_policy", "procedure": "evidence_from_procedure"}
SCOPE_LABEL = {"GLOBAL": "GLOBAL (applies to all countries, including {c})", "LOCAL_COUNTRY": "LOCAL - {c}",

APPLIC_NOTE = {"APPLIES": "GPPS lists {c} legal entities among the applicable entities",

class Progress:
    """Done / total, %, elapsed, time left, calls per minute, cost - every progress_every_sec and at checkpoints."""

    def __init__(self, log, stage, total, already, what, every_sec, usage):
        self.log, self.stage, self.total, self.already, self.what = log, stage, total, already, what
        self.every, self.usage = every_sec, usage
        self.done, self.failed, self.t0, self.last = 0, 0, time.time(), 0.0
        self.c0 = usage.cost(stage) if usage else 0.0
        log.info(f"[{stage}] {total:,} {what} in total | {already:,} done earlier (checkpoint) | {total - already:,} to do now")

    def update(self, n=1, failed=0, force=False, note=""):
        self.done += n
        self.failed += failed
        now = time.time()
        if not force and now - self.last < self.every and self.already + self.done < self.total:
            return
        self.last = now
        el = now - self.t0
        fin = self.already + self.done
        rate = self.done / el if el > 0 else 0
        left = (self.total - fin) / rate if rate > 0 else float("nan")
        cost = (self.usage.cost(self.stage) - self.c0) if self.usage else 0.0
        fmt = lambda sec: "-" if sec != sec else (f"{int(sec // 3600)}h{int(sec % 3600 // 60):02d}m" if sec >= 3600
                                                  else f"{int(sec // 60)}m{int(sec % 60):02d}s")
        self.log.info(f"  [{self.stage}] {fin:,}/{self.total:,} {self.what} ({100 * fin / max(self.total, 1):.1f}%) | "
                      f"elapsed {fmt(el)} | left ~{fmt(left)} | {rate * 60:.0f}/min | cost ${cost:.2f} | "
                      f"no answer {self.failed:,}" + (f" | {note}" if note else ""))

def build_payload(s, ctx, alert, doc):

def section_ids(ctx, doc, s_ids, log=None, ref=None):
    """['S4', 'S7'] -> unit ids of that document; unknown ids are ignored (W-LLM-002)."""
    ids, out, bad = ctx.doc_units[doc], [], []
    raw = " ".join(map(str, s_ids)) if isinstance(s_ids, (list, tuple)) else str(s_ids or "")
    for n in re.findall(r"S\s*(\d+)", raw, flags=re.I) or re.findall(r"\d+", raw):
        if 1 <= int(n) <= len(ids):
            if ids[int(n) - 1] not in out:
                out.append(ids[int(n) - 1])
        else:
            bad.append(n)
    if bad and log:
        log.event("W-LLM-002", f"S-ids {bad} not in {doc} ({len(ids)} sections)", echo=False, **(ref or {}))
    return out


def error_row(row, role, code, detail, attempts=0):

async def read_one(s, ctx, gw, prompts, log, row):
    """One pair -> result row. Never raises (except cancellation): any problem becomes an ERROR row with a code."""
    role = doc_role(ctx, row.doc)
    key = f"{row.doc_type}:{role}"
    ref = {"alert": row.alert, "doc": row.doc}
    try:
        payload = build_payload(s, ctx, row.alert, row.doc)
        system = prompts.get(key)
        code, detail, attempts = None, "", 0
        for _ in range(s.execution["answer_retries"]):
            r = await gw.chat(system, payload, ref=ref)
            attempts += r.attempts
            if not r.ok:
                code, detail = r.code, r.detail
                if code in ("E-GW-008", "E-GW-009"):            # retrying cannot help
                    break
                continue
            a, err, warns = prompts.validate(key, r.data)
            if err:
                code, detail = "E-LLM-003", err
                continue
            if warns:
                log.event("W-LLM-001", "; ".join(warns), echo=False, **ref)
            return {"alert": row.alert, "doc": row.doc, "doc_type": row.doc_type, "doc_scope": row.doc_scope,
                    "document_role": role, "search_rank": row.search_rank, "model_decision": a["decision"],
                    "link_score": a.get("link_score"), "confidence_level": a.get("confidence_level"),
                    "linked_units": "; ".join(section_ids(ctx, row.doc, a.get("linked_sections"), log, ref)),
                    "impact_type": a.get("impact_type"), "change_needed": a.get("change_needed"),
                    "evidence_from_alert": a.get("evidence_from_alert"),
                    "evidence_from_target": a.get(EVIDENCE_FIELD[row.doc_type]), "reasoning": a.get("reasoning"),
                    "framework_change": ((a.get("framework_change") or "NO") if role == "FRAMEWORK" else "NOT_APPLICABLE"),
                    "error_code": None, "error_detail": None, "attempts": attempts}
        log.event(code or "E-GW-012", detail, echo=False, **ref)
        return error_row(row, role, code or "E-GW-012", detail, attempts)
    except asyncio.CancelledError:
        raise
    except Exception as e:                                       # a bug or an odd document: isolate it to this pair
        log.event("E-RUN-004", f"{type(e).__name__}: {e}", echo=False, **ref)
        log.debug(traceback.format_exc())
        return error_row(row, role, "E-RUN-004", f"{type(e).__name__}: {e}")


async def run_pairs(s, ctx, gw, prompts, log, cand, stage, ckpt_path, stop, budget):
    """Read every pair in cand (columns alert, doc, doc_type, doc_scope, search_rank). Returns all rows of the stage
    (earlier checkpointed rows + this session). ckpt_path=None -> no checkpoint (test gate)."""
    gw.stage = log.stage = stage
    ckpt = JsonlCheckpoint(ckpt_path, log=log) if ckpt_path else None
    rows = ckpt.load() if ckpt else []
    wanted = set(zip(cand.alert, cand.doc))
    rows = [r for r in rows if (r["alert"], r["doc"]) in wanted]
    done = {(r["alert"], r["doc"]) for r in rows}
    todo = [r for r in cand.itertuples(index=False) if (r.alert, r.doc) not in done]
    ex = s.execution
    prog = Progress(log, stage, len(cand), len(cand) - len(todo), "documents read", ex["progress_every_sec"], gw.usage)
    if not todo:
        return pd.DataFrame(rows)

    queue = asyncio.Queue()
    for r in todo:
        queue.put_nowait(r)
    state = {"consecutive_fail": 0, "route_fail": 0, "ok": 0, "since_flush": 0, "last_flush": time.time(),
             "abort": None}

    def flush(note=None):
        if ckpt:
            ckpt.flush()
            state["since_flush"], state["last_flush"] = 0, time.time()
            if note:
                prog.update(0, force=True, note=note)

    async def worker():
        while not stop.is_set() and state["abort"] is None:
            try:
                row = queue.get_nowait()
            except asyncio.QueueEmpty:
                return
            res = await read_one(s, ctx, gw, prompts, log, row)
            rows.append(res)
            if ckpt:
                ckpt.add(res)
            failed = res["model_decision"] == "ERROR"
            if failed:
                state["consecutive_fail"] += 1
                if res.get("error_code") == "E-GW-008":
                    state["route_fail"] += 1
            else:
                state["consecutive_fail"] = 0
                state["ok"] += 1
            state["since_flush"] += 1
            prog.update(1, failed=int(failed))
            if state["since_flush"] >= ex["save_every"] or time.time() - state["last_flush"] > 60:
                flush("checkpoint saved")
            if state["route_fail"] >= 5 and state["ok"] == 0:
                state["abort"] = IAError("E-GW-008", f"model '{s.model['chat_model']}' not routed at this base url")
            elif state["consecutive_fail"] >= ex["circuit_breaker_failures"]:
                state["abort"] = IAError("E-GW-010", f"{state['consecutive_fail']} consecutive pairs failed; "
                                                     f"last: {res.get('error_code')} {res.get('error_detail')}")
            elif budget is not None and budget.exceeded():
                state["abort"] = IAError("E-RUN-001", budget.describe())

    workers = [asyncio.create_task(worker()) for _ in range(min(ex["concurrency"], len(todo)))]
    try:
        await asyncio.gather(*workers)
    except BaseException:
        for w in workers:
            if not w.done():
                w.cancel()
        flush()
        raise
    flush()
    prog.update(0, force=True, note="stage finished" if not (stop.is_set() or state["abort"]) else "stopped")
    if state["abort"] is not None:
        raise state["abort"]
    if stop.is_set():
        raise IAError("E-RUN-002", f"stage {stage}: progress saved")
    return pd.DataFrame(rows)


class Budget:
    """Measured spend of this process + the spend recorded by earlier sessions of the same run, against the cap."""

    def __init__(self, cap_usd, usage, spent_before=0.0):
        self.cap, self.usage, self.before = float(cap_usd), usage, float(spent_before)

    def spent(self):
        return self.before + self.usage.cost()

    def exceeded(self):
        return self.spent() >= self.cap

    def describe(self):
        return f"spent ${self.spent():.2f} (this session ${self.usage.cost():.2f}) >= cap ${self.cap:.2f}"

def _walk(o, path=""):
    if isinstance(o, dict):
        for k, v in o.items():
            yield from _walk(v, f"{path}.{k}" if path else str(k))
    elif isinstance(o, list):
        for v in o:
            yield from _walk(v, path + "[]")
    else:
        yield path, o


def _has(term, text):
    return re.search(rf"(?<![a-z]){re.escape(term)}(?![a-z])", text) is not None


def compute_scope(s, ctx, log):
    home_terms = [t.lower() for t in s.country["terms"]]
    home_codes = {str(c).upper() for c in s.country["codes"]}
    entity_codes = {str(c).upper() for c in s.country["entity_codes"]}
    others = [t for t in OTHER_COUNTRIES if t not in home_terms]
    other_codes = OTHER_CODES - home_codes
    rows = []
    for kind, keys, idcands, prefix in (("policy", ("policy_local", "policy_global"), ("policy_core_id", "policy_id"), "POL-"),
                                        ("procedure", ("procedure",), ("procedure_core_id", "procedure_id"), "PROC-")):
        known = {d for d in ctx.doc_text if ctx.doc_type[d] == kind}
        for key in keys:
            df = read_table(ctx.sources_used[key])
            idc = next((c for c in idcands if c in df.columns), None)
            if idc is None or "json_data" not in df.columns:
                log.info(f"  scope: {key} has no id / json_data column - source file decides")
                continue
            if "version_num" in df.columns:
                df = df.assign(_v=pd.to_numeric(df["version_num"], errors="coerce")).sort_values("_v", ascending=False)
            df = df.drop_duplicates(idc)
            for _, r in df.iterrows():
                doc = prefix + N(r[idc])
                if doc not in known:
                    continue
                try:
                    vals = []
                    for p, v in _walk(load_json(r["json_data"])):
                        leaf = re.sub(r"\[\]", "", p).split(".")[-1]
                        if v is None or isinstance(v, (dict, list)) or not SCOPE_KEY.search(leaf) or "sgmnt" in p.lower():
                            continue
                        sv = str(v).strip()
                        if not sv or len(sv) > 200 or sv.lower() in NA:
                            continue
                        vals.append((leaf.lower(), sv, p.lower()))
                    lvl = [sv.upper() for k, sv, _ in vals if k in ("plcylglentycde", "proclglentycde", "region")
                           or "lvl" in k or "level" in k]
                    appl = [(k, sv) for k, sv, pth in vals if ("apblctry" in pth or "apblenty" in pth or "legalenty" in k
                            or "cntry" in k or "country" in k) and "appendix" not in pth
                            and k not in ("plcylglentycde", "proclglentycde")]
                    low = [sv.lower() for _, sv in appl]
                    home = any(_has(t, x) for t in home_terms for x in low) or \
                        any(sv.upper() in home_codes | entity_codes for _, sv in appl)
                    found = sorted({t for t in others for x in low if _has(t, x)} |
                                   {sv.upper() for k, sv in appl if re.search("cntry|country", k) and sv.upper() in other_codes})
                except Exception as e:
                    log.debug(f"scope parse error {doc}: {e}")
                    vals, lvl, home, found = [], [], False, []
                if "GLOBAL" in lvl:
                    scope, how = "GLOBAL", "GPPS level field"
                elif home:
                    scope, how = "LOCAL_COUNTRY", "GPPS applicable entities"
                elif found:
                    scope, how = "LOCAL_OTHER", "GPPS applicable entities"
                else:
                    scope, how = ("LOCAL_COUNTRY" if key == "policy_local" else "GLOBAL"), "source file"
                applic = ("APPLIES" if home else "OTHER_COUNTRIES_ONLY" if found else "NOT_LISTED") if scope == "GLOBAL" else ""
                rows.append(dict(doc_id=doc, doc_type=kind, title=ctx.doc_title[doc], source_file=key, scope=scope,
                                 applicability=applic, decided_by=how,
                                 countries_found=", ".join(found + ([s.country_code] if home else [])),
                                 evidence=" | ".join(f"{k}={sv}" for k, sv, _ in vals)[:300]))
    scope = pd.DataFrame(rows, columns=["doc_id", "doc_type", "title", "source_file", "scope", "applicability",
                                        "decided_by", "countries_found", "evidence"]).drop_duplicates("doc_id")
    missing = [d for d in ctx.doc_text if d not in set(scope.doc_id)]
    if missing:
        scope = pd.concat([scope, pd.DataFrame([dict(
            doc_id=d, doc_type=ctx.doc_type[d], title=ctx.doc_title[d], source_file="-",
            scope="LOCAL_COUNTRY" if ctx.doc_file_scope[d] == "local" else "GLOBAL", applicability="",
            decided_by="source file", countries_found="", evidence="") for d in missing])], ignore_index=True)
    lf = scope[scope.source_file == "policy_local"]
    if len(lf) >= 3:
        other_share = float((lf.scope == "LOCAL_OTHER").mean())
        file_share = float((lf.decided_by == "source file").mean())
        if other_share > 0.5:
            raise IAError("E-SCP-001", f"{other_share:.0%} of the {len(lf)} documents in the local policy file look like "
                                       f"another country's (countries found: "
                                       f"{lf.countries_found.value_counts().head(3).to_dict()})")
        if file_share > 0.5:
            log.event("W-SCP-002", f"{file_share:.0%} of the {len(lf)} local documents matched no country setting")
    n_file = int((scope.decided_by == "source file").sum())
    if n_file:
        log.event("W-SCP-001", f"{n_file} document(s) scoped by their source file (no GPPS level field)", echo=False)
    ctx.scope = scope
    ctx.doc_scope = dict(zip(scope.doc_id, scope.scope))
    ctx.doc_applic = dict(zip(scope.doc_id, scope.applicability))
    write_parquet(scope, os.path.join(s.output_dir, "document_scope.parquet"))
    tab = scope.groupby(["doc_type", "scope"]).size().to_dict()
    log.info(f"  document scope: {tab}")
    return scope

def _md5(t):
    return hashlib.md5(t.encode("utf-8")).hexdigest()


def _clean_emb(t, max_chars):
    t = str(t) if t is not None else ""
    return (t if t.strip() else " ")[:max_chars]


def unit_doc_text(u):
    return u.doc_title + " | " + u.section_title + " | " + u.subsection_title.fillna("") + " | " + u.text


def _load_npz(path, log):
    if not os.path.exists(path):
        return {}
    try:
        z = np.load(path, allow_pickle=False)
        return dict(zip(z["h"].tolist(), z["X"]))
    except Exception as e:
        log.event("W-EMB-001", f"{os.path.basename(path)}: {e}")
        return {}


def _save_npz(path, cache):
    if cache:
        tmp = path + ".tmp.npz"
        np.savez(tmp, h=np.array(list(cache.keys())), X=np.vstack(list(cache.values())))
        os.replace(tmp, path)


def _as_list(x):
    if isinstance(x, list):
        return [str(i).strip() for i in x if str(i).strip()]
    return [p.strip() for p in str(x or "").replace("\n", ";").split(";") if p.strip()]


async def make_query_plans(s, ctx, gw, prompts, log, stop):
    path = os.path.join(s.output_dir, "query_plans.parquet")
    plans = {}
    if os.path.exists(path):
        try:
            p = pd.read_parquet(path)
            plans = {str(a): json.loads(j) for a, j in zip(p.alert, p.plan) if j and j != "{}"}
        except Exception as e:
            log.info(f"  query plan cache unreadable ({e}) - rebuilt")
    atext = ctx.atext
    todo = [a for a in ctx.alerts.alert if a not in plans]
    gw.stage = log.stage = "query_plan"
    if todo:
        prog = Progress(log, "query_plan", len(ctx.alerts), len(ctx.alerts) - len(todo), "query plans",
                        s.execution["progress_every_sec"], gw.usage)
        sem = asyncio.Semaphore(s.execution["query_plan_concurrency"])
        system = prompts.get("query_plan")

        def save():
            write_parquet(pd.DataFrame({"alert": list(plans), "plan": [json.dumps(plans[a], ensure_ascii=False)
                                                                       for a in plans]}), path)

        async def one(a):
            async with sem:
                if stop.is_set():
                    return a, None
                for _ in range(s.execution["answer_retries"]):
                    r = await gw.chat(system, {"alert_id": a, "alert_text": atext[a], "alert_metadata": ctx.meta.get(a, {})},
                                      ref={"alert": a, "doc": "query_plan"})
                    if r.ok:
                        clean, err, _ = prompts.validate("query_plan", r.data)
                        if clean:
                            return a, clean
                        log.event("E-LLM-003", err, echo=False, alert=a)
                    else:
                        log.event(r.code, r.detail, echo=False, alert=a)
                return a, None

        tasks = [asyncio.create_task(one(a)) for a in todo]
        try:
            for i, fut in enumerate(asyncio.as_completed(tasks), 1):
                a, r = await fut
                if r:
                    plans[a] = {k: (_as_list(r.get(k)) if k != "core_mandate" else str(r.get(k) or ""))
                                for k in ("core_mandate", "requirements", "affected_activities", "policy_language",
                                          "procedure_language", "keywords")}
                prog.update(1)
                if i % 20 == 0:
                    save()
        finally:
            for t in tasks:
                if not t.done():
                    t.cancel()
            save()
    missing = [a for a in ctx.alerts.alert if a not in plans]
    if missing:
        raise IAError("E-SHL-001", f"{len(missing)} alert(s) without a query plan, e.g. {missing[:5]} - re-run")
    log.info(f"  query plans {len(plans):,} for {len(ctx.alerts):,} alerts (made now: {len(todo):,})")
    return plans


async def embed_missing(s, texts, cache, path, gw, log, stop):
    need = list(dict.fromkeys(t for t in texts if _md5(t) not in cache))
    if not need:
        return 0
    bs = s.model["embedding_batch_size"]
    sem = asyncio.Semaphore(s.execution["embedding_concurrency"])
    gw.stage = log.stage = "embeddings"
    prog = Progress(log, "embeddings", len(need), 0, "texts embedded", s.execution["progress_every_sec"], gw.usage)
    dim = len(next(iter(cache.values()))) if cache else None

    async def one(batch):
        async with sem:
            if stop.is_set():
                return batch, None
            return batch, await gw.embed(batch)

    tasks = [asyncio.create_task(one(need[i:i + bs])) for i in range(0, len(need), bs)]
    since = 0
    try:
        for fut in asyncio.as_completed(tasks):
            batch, X = await fut
            if X is not None:
                if dim is None:
                    dim = X.shape[1]
                if X.shape[1] != dim:
                    raise IAError("E-EMB-002", f"cache has dimension {dim}, model returned {X.shape[1]}")
                for t, x in zip(batch, X):
                    cache[_md5(t)] = x
                since += len(batch)
            prog.update(len(batch))
            if since >= 2000:
                _save_npz(path, cache)
                since = 0
    finally:
        for t in tasks:
            if not t.done():
                t.cancel()
        _save_npz(path, cache)
    if stop.is_set():
        raise IAError("E-RUN-002", "stopped during embeddings (finished ones are cached)")
    return len(need)


async def build_procedure_shortlist(s, ctx, gw, prompts, log, stop):
    """-> DataFrame alert, doc, doc_type='procedure', search_rank (1 = best)."""
    from sklearn.feature_extraction.text import TfidfVectorizer

    k = s.search["proc_top_k"]
    path = os.path.join(s.output_dir, f"procedure_shortlist_top{k}.parquet")
    alerts = list(ctx.alerts.alert)
    saved = None
    if os.path.exists(path):
        try:
            saved = pd.read_parquet(path)
            saved["alert"] = saved["alert"].astype(str)
        except Exception as e:
            log.info(f"  saved shortlist unreadable ({e}) - rebuilt")
    if saved is not None and set(alerts) <= set(saved.alert):
        out = saved[saved.alert.isin(alerts)].reset_index(drop=True)
        log.info(f"  procedure shortlist from saved file: {len(out):,} pairs")
        return _drop_skipped(out, ctx, log)
    todo = [a for a in alerts if saved is None or a not in set(saved.alert)]
    if saved is not None:
        log.event("W-SHL-001", f"{len(todo)} alert(s) not in the saved shortlist - built fresh for those")

    plans = await make_query_plans(s, ctx, gw, prompts, log, stop)
    atext = ctx.atext

    def queries(a):
        p = plans.get(a, {})
        q = [atext[a][:6000]] + ([p["core_mandate"]] if p.get("core_mandate") else [])
        q += p.get("requirements", []) + p.get("affected_activities", []) + p.get("procedure_language", [])
        return [x for x in dict.fromkeys(q) if x.strip()]

    mc = s.model["embedding_max_chars"]
    pv = ctx.units[ctx.units.doc_type == "procedure"].reset_index(drop=True)
    if pv.empty:
        raise IAError("E-SRC-005", "no visible procedure sections")
    eu_path, eq_path = os.path.join(s.output_dir, "emb_units_alerts.npz"), os.path.join(s.output_dir, "emb_queries.npz")
    EU, EQ = _load_npz(eu_path, log), _load_npz(eq_path, log)
    u_txt = [_clean_emb(t, mc) for t in unit_doc_text(pv).tolist()]
    a_txt = [_clean_emb(atext[a], mc) for a in todo]
    n1 = await embed_missing(s, u_txt + a_txt, EU, eu_path, gw, log, stop)
    n2 = await embed_missing(s, [q for a in todo for q in queries(a)], EQ, eq_path, gw, log, stop)
    log.info(f"  embeddings: new units/alerts {n1:,} | new queries {n2:,} (0 = all from cache)")
    missing = [t for t in u_txt + a_txt if _md5(t) not in EU]
    if missing:
        raise IAError("E-EMB-001", f"{len(missing)} texts still without an embedding - re-run")

    norm = lambda X: X / np.clip(np.linalg.norm(X, axis=1, keepdims=True), 1e-9, None)
    UE = norm(np.vstack([EU[_md5(t)] for t in u_txt]))
    QA = norm(np.vstack([EU[_md5(t)] for t in a_txt]))
    S_a = QA @ UE.T
    S_b = np.zeros_like(S_a)
    for i, a in enumerate(todo):
        Q = norm(np.vstack([EQ[_md5(q)] for q in queries(a) if _md5(q) in EQ] or [QA[i]]))
        S_b[i] = (Q @ UE.T).max(axis=0)
    sr = s.search
    vec = TfidfVectorizer(stop_words="english", ngram_range=(1, sr["tfidf_ngram_max"]), min_df=sr["tfidf_min_df"],
                          max_df=sr["tfidf_max_df"], sublinear_tf=True)
    try:
        U_kw = vec.fit_transform(unit_doc_text(pv))
    except ValueError:                                       # tiny corpora (tests): relax the document-frequency cut
        vec = TfidfVectorizer(stop_words="english", ngram_range=(1, sr["tfidf_ngram_max"]), sublinear_tf=True)
        U_kw = vec.fit_transform(unit_doc_text(pv))
    S_d = (vec.transform([" ".join([atext[a]] + queries(a)[1:] + plans.get(a, {}).get("keywords", []))
                          for a in todo]) @ U_kw.T).toarray()

    def rrf(*mats):
        out = np.zeros_like(mats[0], dtype=np.float32)
        for M in mats:
            out += 1.0 / (sr["rrf_k"] + np.argsort(np.argsort(-M, axis=1), axis=1) + 1)
        return out

    S_f = rrf(S_a, S_b, S_d)
    pdocs = pd.unique(pv.doc_id)
    D_f = pd.DataFrame(S_f.T).groupby(pv.doc_id.values).max().reindex(pdocs).values.T
    rows = []
    for i, a in enumerate(todo):
        order = np.argsort(-D_f[i], kind="stable")[:k]
        rows += [dict(alert=a, doc=pdocs[j], doc_type="procedure", search_rank=r + 1) for r, j in enumerate(order)]
    fresh = pd.DataFrame(rows)
    full = pd.concat([saved, fresh], ignore_index=True) if saved is not None else fresh
    write_parquet(full, path)
    out = full[full.alert.isin(alerts)].reset_index(drop=True)
    return _drop_skipped(out, ctx, log)

"""Load alerts, GPPS documents and the exception lists into one Context object (everything later stages need)."""
import os
import json

from dataclasses import dataclass, field

import pandas as pd

@dataclass
class Context:
    alerts: pd.DataFrame                       # alert, title, text
    meta: dict                                 # alert -> {jurisdiction, regulator, taxonomy}
    units: pd.DataFrame                        # visible sections / sub-sections
    doc_text: dict = field(default_factory=dict)       # doc -> full text with [S<n> | ...] labels (sent to the model)
    doc_units: dict = field(default_factory=dict)      # doc -> [unit_id of S1, S2, ...]
    doc_title: dict = field(default_factory=dict)
    doc_type: dict = field(default_factory=dict)
    doc_src_chars: dict = field(default_factory=dict)
    doc_file_scope: dict = field(default_factory=dict)
    unit_info: dict = field(default_factory=dict)
    framework_docs: dict = field(default_factory=dict)  # doc -> title (read with the framework rule)
    skip_docs: dict = field(default_factory=dict)       # doc -> reason (never read)
    sources_used: dict = field(default_factory=dict)
    # filled by scope.py
    scope: pd.DataFrame = None
    doc_scope: dict = field(default_factory=dict)
    doc_applic: dict = field(default_factory=dict)

    @property
    def atext(self):
        return dict(zip(self.alerts.alert, self.alerts.text))

    def docs_of(self, kind):
        return [d for d in self.doc_text if self.doc_type[d] == kind and d not in self.skip_docs]


def resolve_sources(s, log):
    used, missing = {}, []
    for k, v in s.sources.items():
        r = resolve_source(s.path(v))
        log.info(f"  source {k:14s} {r or 'NOT FOUND: ' + s.path(v)}")
        if r:
            used[k] = r
        else:
            missing.append(f"{k} = {s.path(v)}")
    if missing:
        raise IAError("E-SRC-001", "; ".join(missing))
    return used


def load_exceptions(s, log):
    """exceptions/framework_documents.csv and exceptions/skip_documents.csv (doc_id,title,reason; # = comment)."""
    out = {}
    for name in ("framework_documents", "skip_documents"):
        path = os.path.join(s.path(s.paths["exceptions_dir"]), f"{name}.csv")
        if not os.path.exists(path):
            raise IAError("E-EXC-001", f"{path} not found (keep the file, even with only its header)")
        try:
            df = pd.read_csv(path, comment="#", dtype=str, skipinitialspace=True).fillna("")
        except Exception as e:
            raise IAError("E-EXC-001", f"{path}: {e}")
        if "doc_id" not in df.columns:
            raise IAError("E-EXC-001", f"{path}: header must be doc_id,title,reason")
        df["doc_id"] = df["doc_id"].str.strip().str.upper()
        df = df[df.doc_id != ""]
        col = "title" if name == "framework_documents" else "reason"
        out[name] = {r.doc_id: (getattr(r, col, "") if col in df.columns else "") for r in df.itertuples()}
        log.info(f"  exceptions {name}: {len(out[name])} document(s)")
    return out["framework_documents"], out["skip_documents"]


def copy_reused_caches(s, log):
    src = s.path(s.run["reuse_search_from"]) if s.run["reuse_search_from"] else None
    if not src:
        return
    if not os.path.isdir(src):
        log.info(f"  reuse_search_from not found, ignored: {src}")
        return
    k = s.search["proc_top_k"]
    for f in ("query_plans.parquet", "emb_units_alerts.npz", "emb_queries.npz", f"procedure_shortlist_top{k}.parquet"):
        a, b = os.path.join(src, f), os.path.join(s.output_dir, f)
        if os.path.exists(a) and not os.path.exists(b):
            shutil.copy2(a, b)
            log.info(f"  reused from {src}: {f}")


def load_alerts(s, used, log):
    r2 = read_table(used["rapid2"])
    scope_df = read_table(used["alert_scope"])
    for col in ("RECORD_ID", "TITLE"):
        if col not in r2.columns:
            raise IAError("E-SRC-003", f"rapid2 file: column {col} missing")
    sc = pick(scope_df, "RECORD_ID", "RID")
    if not sc:
        raise IAError("E-SRC-003", "alert_scope file: column RECORD_ID missing")
    scope_ids = set(scope_df[sc].map(N))
    r2 = r2[r2["RECORD_ID"].map(N).isin(scope_ids)].drop_duplicates("RECORD_ID")
    alerts = pd.DataFrame({
        "alert": r2["RECORD_ID"].map(N).values,
        "title": r2["TITLE"].map(clean).values,
        "text": r2.apply(lambda r: " ".join(x for x in (clean(r.get("TITLE")), clean(r.get("SUMM_CONTENT")),
                                                        clean(r.get("SUMM_UPDATE")), clean(r.get("INTRO_TEXT"))) if x),
                         axis=1).values if len(r2) else []})
    g = lambda r, c: clean(r.get(c)) if c in r2.columns else ""
    meta = {N(r["RECORD_ID"]): {"jurisdiction": g(r, "RECORD_JURISDICTION_NAME"), "regulator": g(r, "REGULATOR_CODE"),
                                "taxonomy": " > ".join(x for x in (g(r, "LEVEL_1_RISK_TAXONOMY"), g(r, "LEVEL_2_RISK_TAXONOMY"),
                                                                   g(r, "LEVEL_3_RISK_TAXONOMY")) if x)}
            for _, r in r2.iterrows()}
    want = [N(a) for a in s.run["alert_ids"]]
    if want:
        missing = [a for a in want if a not in set(alerts.alert)]
        if missing:
            log.event("W-SRC-002", f"{len(missing)} id(s) not in scope: {missing[:20]}")
        alerts = alerts[alerts.alert.isin(want)]
    elif s.run["alerts_limit"]:
        alerts = alerts.head(int(s.run["alerts_limit"]))
    alerts = alerts[alerts.text.str.strip() != ""].reset_index(drop=True)
    if alerts.empty:
        raise IAError("E-SRC-004", f"{len(scope_ids):,} ids in scope, {len(r2):,} found in Rapid2, "
                                   f"{len(want)} requested")
    return alerts, meta


def load_units(s, used, log):
    path = os.path.join(s.output_dir, "units.parquet")
    if os.path.exists(path):
        try:
            u = pd.read_parquet(path)
            log.info(f"  sections / sub-sections from cache: {len(u):,} ({path})")
            return u
        except Exception as e:
            log.info(f"  units cache unreadable ({e}) - rebuilding")
    log.info("  parsing GPPS ...")
    pub = bool(s.rules["published_only"])
    pol = pd.concat([load_docs(used["policy_local"], "policy", "local", pub, log),
                     load_docs(used["policy_global"], "policy", "global", pub, log)], ignore_index=True)
    if len(pol):
        pol = pol.drop_duplicates("doc_id")
    proc = load_docs(used["procedure"], "procedure", "global", pub, log)
    docs = pd.concat([pol, proc], ignore_index=True)
    if docs.empty:
        raise IAError("E-SRC-005", "no published GPPS documents found in the three GPPS files")
    u = build_units(docs)
    write_parquet(u, path)
    return u


def build_context(s, log):
    used = resolve_sources(s, log)
    write_json(used, os.path.join(s.output_dir, "sources_used.json"))
    copy_reused_caches(s, log)
    framework, skip = load_exceptions(s, log)
    alerts, meta = load_alerts(s, used, log)
    units_all = load_units(s, used, log)
    for c in ("doc_title", "section_no", "section_title", "subsection_no", "subsection_title", "text"):
        units_all[c] = units_all[c].map(lambda v: "" if v is None or (isinstance(v, float) and math.isnan(v)) else str(v))
    units = units_all[~units_all.hidden.astype(str).str.lower().isin(["true", "1"])].reset_index(drop=True)

    ctx = Context(alerts=alerts, meta=meta, units=units, framework_docs=framework, skip_docs=skip, sources_used=used)
    for doc_id, g in units.groupby("doc_id", sort=False):
        parts, ids = [], []
        for i, r in enumerate(g.itertuples(), start=1):
            sec = f"{(str(r.section_no) + ' ') if r.section_no else ''}{r.section_title}"
            sub = f" > {r.subsection_no} {r.subsection_title}" if r.subsection_no else ""
            parts.append(f"[S{i} | {sec}{sub}]\n{r.text}")
            ids.append(r.unit_id)
            ctx.unit_info[r.unit_id] = dict(
                section_no=str(r.section_no or ""), section_title=r.section_title,
                subsection_no=str(r.subsection_no or ""), subsection_title=r.subsection_title or "",
                gpps_label=(f"Section {r.section_no} {r.section_title}" if r.section_no else r.section_title)
                + (f" > {r.subsection_no} {r.subsection_title}" if r.subsection_no else ""), text=r.text)
        ctx.doc_text[doc_id], ctx.doc_units[doc_id] = "\n\n".join(parts), ids
        ctx.doc_title[doc_id], ctx.doc_src_chars[doc_id] = g.doc_title.iloc[0], int(pd.to_numeric(g.chars).sum())
        ctx.doc_type[doc_id], ctx.doc_file_scope[doc_id] = g.doc_type.iloc[0], g.scope.iloc[0]
    hidden_only = set(units_all.doc_id) - set(ctx.doc_text)
    if hidden_only:
        log.event("W-SRC-003", f"{len(hidden_only)} document(s) have no visible text: {sorted(hidden_only)[:10]}")
    for d in list(framework) + list(skip):
        if d not in ctx.doc_text:
            log.event("W-EXC-001", f"{d} is listed in exceptions/ but not found in GPPS")
    # hand-off files (texts exactly as sent to the model) - used by the audit notebook
    write_parquet(pd.DataFrame({"alert": alerts.alert, "title": alerts.title, "text": alerts.text,
                                "meta": [json.dumps(meta.get(a, {}), ensure_ascii=False) for a in alerts.alert]}),
                  os.path.join(s.output_dir, "alerts.parquet"))
    log.info(f"  alerts {len(alerts):,} | policies {len(ctx.docs_of('policy')):,} | procedures "
             f"{len(ctx.docs_of('procedure')):,} | skipped by exceptions {len([d for d in skip if d in ctx.doc_text])} | "
             f"visible sections / sub-sections {len(units):,}")
    return ctx
