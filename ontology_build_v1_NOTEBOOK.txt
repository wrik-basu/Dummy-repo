# ------------------------------------------------------------------------------
# Readers

def read_any(path):
    """Read parquet (dbdate-safe), xlsx/xls or csv. Full text, nothing truncated."""
    if path.endswith(".parquet"):
        table = pq.read_table(path).replace_schema_metadata(None)
        return table.to_pandas(date_as_object=True)
    if path.endswith((".xlsx", ".xls")):
        return pd.read_excel(path)
    return pd.read_csv(path)


def col(df, key, required=True):
    """Return the first candidate column from COLS[key] that exists in df."""
    for candidate in COLS[key]:
        if candidate in df.columns:
            PICKED[key] = candidate
            return candidate

    if required:
        raise KeyError(
            f"{key}: none of {COLS[key]} found. "
            f"Columns: {list(df.columns)[:40]}"
        )

    PICKED[key] = None
    return None


def pick_col(df, names):
    """First name in `names` that is a column of df, else None."""
    return next((n for n in names if n in df.columns), None)


# ------------------------------------------------------------------------------
# Value cleaning
# ------------------------------------------------------------------------------
def is_blank(value):
    return value is None or (isinstance(value, float) and pd.isna(value))


def cid(value):
    """Clean an ID: drop hidden characters / spaces, 123.0 -> 123, '' if empty."""
    if is_blank(value):
        return ""

    s = re.sub(r"[​ ﻿]", "", str(value)).strip()

    if s.endswith(".0") and s[:-2].isdigit():
        s = s[:-2]

    if s.lower() in ("nan", "none", ""):
        return ""
    return s


def txt(value):
    """Clean text: strip HTML tags, unescape entities, squash spaces. Full length."""
    if is_blank(value):
        return ""
    s = re.sub(r"(?s)<[^>]+>", " ", str(value))
    s = html.unescape(s)
    return re.sub(r"\s+", " ", s).strip()


def slug(value):
    """Text -> ID-safe slug (spaces -> underscores)."""
    return re.sub(r"\s+", "_", txt(value))


def split_multi(value):
    """'A; B | C' -> ['A', 'B', 'C']"""
    return [part.strip() for part in re.split(r"[;|\n]", txt(value)) if part.strip()]


# ------------------------------------------------------------------------------
# Graph builders
# ------------------------------------------------------------------------------
def node(nid, ntype, title="", text="", system="", rec=True, **props):
    """
    Add a node once; later calls only fill missing title / text / props.

    rec=True  : loaded from its own system of record (or a value node)
    rec=False : only referenced by another system (a 'stub' until found)
    """
    if not nid:
        return ""

    n = NODES.get(nid)
    if n is None:
        n = dict(id=nid, type=ntype, title="", text="",
                 source_system=system, record=False, props={})
        NODES[nid] = n

    # Upgrade a stub to a record when its own system loads it
    if rec and not n["record"]:
        n["record"] = True
        n["source_system"] = system

    if title and not n["title"]:
        n["title"] = txt(title)
    if text and not n["text"]:
        n["text"] = txt(text)

    for key, value in props.items():
        if value not in ("", None) and key not in n["props"]:
            n["props"][key] = value

    return nid


def edge(src, tgt, prop, layer, system, **attrs):
    """
    Add one relationship.

    layer:
      FACT          from a system of record (RegMap, Helios, GPPS)
      INTAKE        Rapid2 alert tags (applicability, category, ML tags)
      AI_IMPACT     alert impact assertion (alert -> RGL/RSM/ROB/LR/L1C/1C)
      AI_STRUCTURE  ontology link built with GenAI, not alert-specific
                    (ROB -> policy, ROB -> procedure)
      REFERENCE     RegMap's own human mapping, kept as a check only
    """
    if not src or not tgt:
        return
    EDGES.append(dict(
        source=src,
        target=tgt,
        property=prop,
        layer=layer,
        source_system=system,
        run_id=RUN_ID,
        **attrs,
    ))


def taxonomy_chain(l1, l2, l3, system):
    """Create L1 > L2 > L3 taxonomy nodes, link child -> parent, return deepest id."""
    ids = []
    for level, name in (("L1", l1), ("L2", l2), ("L3", l3)):
        name = txt(name)
        if not name:
            break
        ids.append(node(
            f"TAX-{level}-{slug(name)}", "RiskTaxonomyNode", f"{level} · {name}",
            system=system, level=level, name=name,
        ))

    for child, parent in zip(ids[1:], ids[:-1]):
        edge(child, parent, "broaderRisk", "FACT", system)

    return ids[-1] if ids else ""


def note(label, value):
    REPORT.append((label, value))

        status=txt(r.get("RECORD_STATUS")),
        start_date=str(r.get("RECORD_START_DATE") or ""),
        active=cid(r.get("ACTIVE_INDICATOR")),
        url=txt(r.get("URL_TEXT")),
    )

    # --- Jurisdiction ---------------------------------------------------------
    jur = txt(r.get("RECORD_JURISDICTION_NAME"))
    if jur:
        j = node(f"JUR-{slug(jur)}", "Jurisdiction", jur, system="Rapid2")
        edge(a, j, "inJurisdiction", "INTAKE", "Rapid2")

    # --- Regulator ------------------------------------------------------------
    reg = cid(r.get("REGULATOR_CODE"))
    if reg:
        g_ = node(f"REGULATOR-{reg}", "Regulator", reg, system="Rapid2")
        edge(a, g_, "issuedByRegulator", "INTAKE", "Rapid2")

    # --- Risk steward ---------------------------------------------------------
    stew = txt(r.get("RISK_STEWARD_AREA_NAME"))
    if stew:
        s = node(f"STEW-{slug(stew)}", "Role", stew, system="Rapid2", role="Risk steward")
        edge(a, s, "taggedSteward", "INTAKE", "Rapid2")

    # --- Themes ---------------------------------------------------------------
    for theme in split_multi(r.get("RECORD_THEMES")):
        t = node(f"THEME-{slug(theme)}", "RegulatoryTheme", theme, system="Rapid2")
        edge(a, t, "hasTheme", "INTAKE", "Rapid2")

    # --- Risk taxonomy L1 > L2 > L3 -------------------------------------------
    tax = taxonomy_chain(
        r.get("LEVEL_1_RISK_TAXONOMY"),
        r.get("LEVEL_2_RISK_TAXONOMY"),
        r.get("LEVEL_3_RISK_TAXONOMY"),
        "Rapid2",
    )
    if tax:
        edge(a, tax, "taggedRisk", "INTAKE", "Rapid2")

# Extracts 3 / 4 / 5: RegMap's own RSM -> risk / control mappings.
# Kept as REFERENCE (human mappings at summary level): candidates and checks
# for the AI bridges, never treated as truth.
# ------------------------------------------------------------------------------
def rsm_map(key, target_col, prop, applicable_col=None):
    df = rm(key)
    if target_col not in df.columns:
        note(f"{key}: column missing", target_col)
        return 0

    keep = [c for c in ["Regulation Summary ID", target_col, applicable_col]
            if c and c in df.columns]
    df = df[keep].copy()
    df["_s"] = df["Regulation Summary ID"].map(cid)
    df["_t"] = df[target_col].map(cid)
    df = df[(df["_s"] != "") & (df["_t"] != "")].drop_duplicates(["_s", "_t"])

    for _, r in df.iterrows():
        applicable = txt(r.get(applicable_col)) if applicable_col else ""
        edge(r["_s"], r["_t"], prop, "REFERENCE", "RegMap", applicable=applicable)

    return len(df)

ID_PATTERN = re.compile(r"\b(L1C-\d+|L2C-\d+|LR-\d+|1C-\d+|IC-\d+)\b")


def parse_json(value):
    """json_data cell -> dict ({} if empty)."""
    if isinstance(value, dict):
        return value
    if isinstance(value, str) and value.strip():
        return json.loads(value)
    return {}


def load_gpps(paths, id_col):
    """Concatenate files, keep the latest version per document, parse json_data."""
    df = pd.concat([read_any(p) for p in paths], ignore_index=True)
    df = df.sort_values("version_num", ascending=False).drop_duplicates(id_col)
    df["doc"] = df["json_data"].map(parse_json)
    return df


def obj_rels(segments):
    """Yield (segment title, objRel entry) for every objRel in every segment."""
    for seg in segments or []:
        seg_title = txt(seg.get("sgmntTitlText") or seg.get("procSgmntTitlText") or "")
        for rel in seg.get("objRel") or []:
            yield seg_title, rel


def rel_target(rel):
    """(type code, target id) of one objRel entry."""
    rel_type = str(rel.get("objTypeCde") or rel.get("plcyObjTypeCde") or "").upper()

    for key in ("cntlId", "objId", "plcyObjId", "riskId", "plcyCoreId", "objRefId", "id"):
        if rel.get(key) not in (None, ""):
            return rel_type, cid(rel[key])

    # Fallback: an ID written inside a name / description
    m = ID_PATTERN.search(json.dumps(rel))
    return rel_type, (m.group(1) if m else "")


def control_type(tid):
    return "LibraryControl" if tid.startswith("L1C") else "Control"


rel_stats = {}


def count_rel(label):
    rel_stats[label] = rel_stats.get(label, 0) + 1

    )

    if owner:
        p = node(f"PERSON-{slug(owner)}", "Person", owner, system="GPPS")
        edge(pid, p, "ownedBy", "FACT", "GPPS")

    segments = (d.get("policyApplication") or {}).get("segment")
    for seg_title, rel in obj_rels(segments):
        rel_type, tid = rel_target(rel)
        count_rel(f"policy {rel_type}")

# Base = the FULL linkage output (every decision). Judge verdict joined on.
# Pairs never audited -> NOT_AUDITED. Planted test pairs are not in the base
# files, so they never enter the graph.

def verdict_map(path, src, tgt, level=None):
    """{(source id, target id): judge verdict} from an audit file."""
    a = read_any(path)
    if level is not None and "level" in a.columns:
        a = a[a["level"].astype(str).str.upper() == level]
    v = pick_col(a, COLS["verdict"])
    return {(cid(s), cid(t)): txt(x).upper() for s, t, x in zip(a[src], a[tgt], a[v])}


def ai_links(df, src_col, tgt_col, prop, system,
             src_prefix="", tgt_prefix="", verdicts=None, extra=None, layer="AI_IMPACT"):
    """Add one AI link edge per row (all decisions kept). Returns rows added."""

    # --- resolve optional columns once ----------------------------------------
    c_dec    = pick_col(df, COLS["decision"])
    c_score  = pick_col(df, ["link_score", "max_score"])
    c_conf   = pick_col(df, ["confidence_level", "confidence", "confidence_score", "model_confidence"])
    c_ev_from = pick_col(df, ["evidence_from_alert", "evidence_from_obligation", "model_evidence_alert"])
    c_ev_to  = pick_col(df, ["evidence_from_regulation", "evidence_from_target",
                             "evidence_from_document", "model_evidence_reg"])
    c_reason = pick_col(df, ["reasoning", "model_reasoning"])
    c_ground = pick_col(df, ["grounding_flag"])
    extra    = {k: v for k, v in (extra or {}).items() if v in df.columns}
    verdicts = verdicts or {}

    def get(row, column):
        return txt(row.get(column)) if column else ""

    n = 0
    for _, r in df.iterrows():
        s, t = cid(r[src_col]), cid(r[tgt_col])
        if not s or not t:
            continue

        edge(
            src_prefix + s, tgt_prefix + t, prop, layer, system,
            decision=get(r, c_dec).upper(),
            score=r.get(c_score) if c_score else None,
            confidence=get(r, c_conf),
            evidence_from=get(r, c_ev_from),
            evidence_to=get(r, c_ev_to),
            reasoning=get(r, c_reason),
            grounding=get(r, c_ground),
            judge_verdict=verdicts.get((s, t), "NOT_AUDITED"),
            sme_status="UNREVIEWED",
            **{k: txt(r.get(v)) for k, v in extra.items()},
        )
        n += 1
    return n
