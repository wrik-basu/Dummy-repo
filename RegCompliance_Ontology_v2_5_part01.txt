  const out=document.getElementById("ans"); out.innerHTML="";
  const ranked=route(q), [best,score]=ranked[0];
  const delay=reduce?0:260;
  const go=(i,f)=>askTimer.push(setTimeout(f,delay*i));
  go(0,()=>steps[0].className="on");
  if(score<2){
    go(1,()=>{steps[0].className="done";out.innerHTML=`<div class="nores"><b>No grounded answer.</b> The graph has no path that answers this, so nothing is made up.   const cq=CQX[best], rows=cq.rows;
  go(1,()=>{steps[0].className="done";steps[1].className="on"});

  /* Turn each alert's intake record into ontology nodes and links.
     Shared values (one steward, one legal entity...) become ONE node, so
     the graph shows which alerts share context. */
  function addIntake(d) {
    const have = new Set(d.nodes.map((n) => n.id));
    const node = (id, type, title, extra = {}) => {
      if (!have.has(id)) { d.nodes.push({ id, type, title, text: "", ...extra }); have.add(id); }
      return id;
    };
    const seenE = new Set(d.edges.map((e) => e.source + "|" + e.target + "|" + e.relation));
    const edge = (s, t, r) => {
      const k = s + "|" + t + "|" + r;
      if (!seenE.has(k)) { d.edges.push({ source: s, target: t, relation: r, kind: "CONTEXT" }); seenE.add(k); }
    };
    const slug = (x) => String(x).trim().replace(/\s+/g, "_");
    d.nodes.filter((n) => n.type === "Alert").forEach((a) => {
      const key = a.id.replace(/^ALERT-/, "");
      const x = a.intake || INTAKE[key];
      if (!x) return;
      a.intake = x;
      if (x.ref || x.title) a.title = [x.ref, x.title].filter(Boolean).join(" · ");
      // Location: "Global/United States" -> United States, broader Global
      const parts = String(x.location || "").split("/").map((p) => p.trim()).filter(Boolean);
      let child = null;
      parts.reverse().forEach((p, i) => {
        const id = node("JUR-" + slug(p), "Jurisdiction", p, { col: 1 + i });
        if (child) edge(child, id, "broader_jurisdiction"); else edge(a.id, id, "in_jurisdiction");
        child = id;
      });
      if (x.steward) edge(a.id, node("STEW-" + slug(x.steward), "Steward", x.steward), "tagged_steward");
      if (x.theme)   edge(a.id, node("THEME-" + slug(x.theme), "Theme", x.theme), "has_theme");
      // Risk taxonomy: alert is tagged at L3; L3 -> L2 -> L1 is the hierarchy
      const lv = [[x.l3, "L3", 1], [x.l2, "L2", 2], [x.l1, "L1", 3]].filter((v) => v[0]);
      let prev = null;
      lv.forEach(([name, level, col]) => {
        const id = node("TAX-" + level + "-" + slug(name), "Taxonomy", level + " · " + name, { col, level });
        if (prev) edge(prev, id, "broader_risk"); else edge(a.id, id, "tagged_risk");
        prev = id;
      });
      if (x.lob) edge(a.id, node("LOB-" + slug(x.lob), "LOB", x.lob), "tagged_business");
      if (x.le)  edge(a.id, node("LE-" + slug(x.le), "LE", x.le), "tagged_legal_entity");
    });
  }

  function scopeSet() {
    if (cur === "ALL") return new Set(DATA.nodes.map((n) => n.id));
    const seen = new Set([cur]), queue = [cur];
    (nodeById[cur].story_extra || []).forEach((x) => { if (nodeById[x] && !seen.has(x)) { seen.add(x); queue.push(x); } });
    while (queue.length) {
      const id = queue.shift();
      outE[id].forEach((e) => { if (!seen.has(e.target)) { seen.add(e.target); queue.push(e.target); } });
    }
    return seen;
  }

  // Edges drawn for the active layer.
  function layerEdges(scope) {
    const inScope = (e) => scope.has(e.source) && scope.has(e.target);
    if (layer === "impact") return DATA.edges.filter((e) => inScope(e) && e._layer === "impact");
    if (layer === "context") return DATA.edges.filter((e) => inScope(e) && e._layer === "context");
    if (layer === "structure") return DATA.edges.filter((e) => inScope(e) && e._layer !== "impact" && e._layer !== "context");
    // both: structure everywhere; an alert's impact line is drawn only to
    // records with no structural parent in view (the others carry the
    // impact as the coloured dot, and in the side panel).
    return DATA.edges.filter((e) => {
      if (!inScope(e)) return false;
      if (e._layer !== "impact") return true;          // structure, gaps and intake context
      return !inE[e.target].some((p) => scope.has(p.source) && p._layer !== "impact");
    });
  }

  // Records that take part in the active layer.
  function layerNodes(scope, edges) {
    const s = new Set();
    edges.forEach((e) => { s.add(e.source); s.add(e.target); });
    if (layer !== "structure") scope.forEach((id) => { if (nodeById[id].type === "Alert") s.add(id); });
    if (layer === "structure") scope.forEach((id) => {          // impacted records with no structure link
      const t = nodeById[id].type;
      if (t !== "Alert" && !CTX_TYPES.has(t)) s.add(id);
    });
    return s;
  }

  function computeView() {
    const scope = scopeSet();
    const edges = layerEdges(scope);
    const nodes = layerNodes(scope, edges);
    KIDS = {}; PARENTS = {};
    edges.forEach((e) => {
      (KIDS[e.source] = KIDS[e.source] || []).push(e.target);
      (PARENTS[e.target] = PARENTS[e.target] || []).push(e.source);
    });
    return { scope, edges, nodes };
  }

  function resetReveal() {
    const { nodes } = computeView();
    // Start from the roots: records nothing in view points to (the alerts,
    // or in the structure layer the top of each chain).
    R = new Set([...nodes].filter((id) => !(PARENTS[id] || []).length));
    fresh = new Set(R);
  }
  function reveal(ids) { ids.forEach((k) => { if (!R.has(k)) { R.add(k); fresh.add(k); } }); }
  // An alert opens in two steps, like the real process: first its intake
  // context (applicability, category, ML tags), then what it impacts.
  function expand(id) {
    const kids = KIDS[id] || [];
    if (nodeById[id].type === "Alert") {
      const ctx = kids.filter((k) => CTX_TYPES.has(nodeById[k].type) && !R.has(k));
      if (ctx.length) { reveal(ctx); return; }
    }
    reveal(kids);
  }
  function fold(id) {
    (KIDS[id] || []).forEach((k) => {
      if (!R.has(k)) return;
      const other = (PARENTS[k] || []).some((p) => p !== id && R.has(p));
      if (!other) { fold(k); R.delete(k); }
    });
  }
  function nextStep() {
    [...R].filter((id) => (KIDS[id] || []).some((k) => !R.has(k))).forEach(expand);
    render();
  }
  function revealPath(id) {
    let x = id, guard = 0;
    while (x && !R.has(x) && guard++ < 50) { R.add(x); fresh.add(x); x = (PARENTS[x] || [])[0]; }
  }


  /* -------------------------------------------------------------------
     9. LAYOUT: two lanes, fixed positions for the whole scope
        Regulatory chain : Alert | RGL | RSM | ROB | Policy + Procedure | Gaps
        Risk & control   : Alert | LR  | L1C | 1C  |                   | Gaps
     ------------------------------------------------------------------- */
  function laneOf(id) {
    const n = nodeById[id];
    if (n.type === "Gap") { const p = inE[id][0]; return p ? laneOf(p.source) : "reg"; }
    return (T[n.type] || { lane: "reg" }).lane;
  }

  function layoutGraph(scope) {
    P = {};
    const cols = {}, alerts = [];
    scope.forEach((id) => {
      const n = nodeById[id];
      if (n.type === "Alert") { alerts.push(id); return; }
      const key = laneOf(id) + "|" + colOf(n);
      (cols[key] = cols[key] || []).push(id);
    });

    // Order each column next to its structural parent (falls back to alert).
    const order = {};
    alerts.sort().forEach((id, i) => { order[id] = i / Math.max(1, alerts.length); });
    const parentsOf = (id) => {
      const s = inE[id].filter((e) => scope.has(e.source) && e._layer !== "impact").map((e) => e.source);
      return s.length ? s : inE[id].filter((e) => scope.has(e.source)).map((e) => e.source);
    };
    const typeRank = { Policy: 0, Procedure: 1,
      Jurisdiction: 0, Steward: 1, Theme: 2, Taxonomy: 3, LOB: 4, LE: 5 };
    ["reg", "ctx", "ctl"].forEach((lane) => {
      for (let c = 1; c <= 5; c++) {
        const list = cols[lane + "|" + c];
        if (!list) continue;
        const bary = (id) => {
          const ps = parentsOf(id).filter((p) => p in order);
          return ps.length ? ps.reduce((s, p) => s + order[p], 0) / ps.length : 1;
        };
        list.sort((a, b) =>
          (typeRank[nodeById[a].type] ?? 0) - (typeRank[nodeById[b].type] ?? 0) ||
          bary(a) - bary(b) ||
          String(nodeById[a].title).localeCompare(String(nodeById[b].title)));
        list.forEach((id, i) => { order[id] = i / list.length; });
      }
    });

    const count = (lane) => Math.max(0, ...[1, 2, 3, 4, 5].map((c) => (cols[lane + "|" + c] || []).length));
    const laneH = (lane) => Math.max(1, count(lane)) * ROW_H;
    // Regulatory chain on top, intake context in the middle (next to the
    // alerts), risk & control at the bottom.
    const lanes = {};
    let top = TOP;
    ["reg", "ctx", "ctl"].forEach((lane) => {
      if (lane === "ctx" && !count("ctx")) return;
      lanes[lane] = { top, h: laneH(lane) };
      top += laneH(lane) + LANE_GAP;
    });

    Object.entries(cols).forEach(([key, list]) => {
      const [lane, c] = key.split("|"), L = lanes[lane];
      const off = (L.h - list.length * ROW_H) / 2;
      list.forEach((id, i) => { P[id] = { x: 40 + Number(c) * COL_W, y: L.top + off + i * ROW_H }; });
    });

    const midY = lanes.ctx
      ? lanes.ctx.top + lanes.ctx.h / 2 - NODE_H / 2
      : lanes.reg.top + lanes.reg.h + LANE_GAP / 2 - NODE_H / 2;
    alerts.forEach((id, i) => {
      P[id] = { x: 40, y: midY - ((alerts.length - 1) / 2) * ROW_H * 1.3 + i * ROW_H * 1.3 };
    });
    return { cols, lanes };
  }


  /* -------------------------------------------------------------------
     10. RENDER
     ------------------------------------------------------------------- */
  function render() {
    const { scope, edges, nodes } = computeView();
    const { cols, lanes } = layoutGraph(scope);
    const vis = mode === "all" ? nodes : new Set([...R].filter((id) => nodes.has(id)));
    const shown = edges.filter((e) => vis.has(e.source) && vis.has(e.target));

    // Show-all mode: clicking a box highlights its own links.
    let fs = null;
    if (mode === "all" && selL && selL.kind === "node" && vis.has(selL.id)) {
      fs = new Set([selL.id]);
      shown.forEach((e) => { if (e.source === selL.id || e.target === selL.id) { fs.add(e.source); fs.add(e.target); } });
    } else if (mode === "all" && selL && selL.kind === "edge") {
      const e = DATA.edges[selL.id]; fs = new Set([e.source, e.target]);
    }
    const labelOn = (e) => mode === "step" || (fs && fs.has(e.source) && fs.has(e.target));

    let order = 0;
    const anim = (id) => fresh.has(id) ? ` lv-new" style="animation-delay:${Math.min(order++ * 35, 900)}ms` : "";

    let g = "";
    const width = 40 + 6 * COL_W;

    // --- Lane bands and column titles ---------------------------------
    const CTX_COL = { 1: "TAGGED AT INTAKE", 2: "BROADER", 3: "BROADER" };
    [["reg", "REGULATORY CHAIN · RegMap + GPPS"], ["ctx", "ALERT CONTEXT · intake standardisation & ML contextualisation"],
     ["ctl", "RISK & CONTROL CHAIN · Helios"]].forEach(([lane, title]) => {
      const L = lanes[lane];
      if (!L) return;
      g += `<rect class="lv-lane" x="${COL_W - 40}" y="${L.top - 46}" width="${width - COL_W + 20}" height="${L.h + 62}" rx="10"/>`;
      g += `<text class="lv-lane-t" x="${COL_W - 22}" y="${L.top - 27}">${title}</text>`;
      for (let c = 1; c <= 5; c++) {
        const list = cols[lane + "|" + c];
        if (!list) continue;
        const names = lane === "ctx" ? (CTX_COL[c] || "")
          : [...new Set(list.map((id) => tLabel(nodeById[id].type)))].join(" / ");
        g += `<text class="lv-col-t" x="${40 + c * COL_W}" y="${L.top - 9}">${h(names.toUpperCase())}</text>`;
      }
    });

    // --- Lines ------------------------------------------------------------
    const labels = [], labelSeen = new Set();
    shown.forEach((e) => {
      const a = P[e.source], b = P[e.target];
      if (!a || !b) return;
      const x1 = a.x + NODE_W, y1 = a.y + NODE_H / 2, x2 = b.x, y2 = b.y + NODE_H / 2;
      const dx = Math.max(40, (x2 - x1) / 2);
      const d = `M${x1},${y1} C${x1 + dx},${y1} ${x2 - dx},${y2} ${x2},${y2}`;
      const v = e._layer === "impact" ? vClass(e.judge_verdict) : "";
      const dim = fs && !(fs.has(e.source) && fs.has(e.target)) ? "lv-dim" : "";
      const hl = selL && selL.kind === "edge" && selL.id === e._i ? "hl" : "";
      const an = anim(e.target);
      g += `<path class="lv-edge ${e._layer} ${v ? "v-" + v : ""} ${dim} ${hl}${an}" d="${d}"/>`;
      g += `<path class="lv-hit" data-edge="${e._i}" d="${d}"><title>${h(verb(e.relation))}</title></path>`;
      // The relation is written on the line, just before the box it points to.
      const key = e.target + "|" + e.relation;
      if (labelOn(e) && !labelSeen.has(key)) {
        labelSeen.add(key);
        const t = verb(e.relation), w = t.length * 6.2 + 14;
        labels.push(`<g class="lv-elab ${e._layer}${an}" data-edge="${e._i}">
          <rect x="${x2 - 8 - w}" y="${y2 - 9}" width="${w}" height="18" rx="9"/>
          <text x="${x2 - 8 - w / 2}" y="${y2 + 3.5}" text-anchor="middle">${h(t)}</text>
          <title>${h(t)} · ${h(prop(e.relation))}</title></g>`);
      }
    });
    g += labels.join("");

    // --- Boxes ------------------------------------------------------------
    vis.forEach((id) => {
      const n = nodeById[id], p = P[id];
      if (!p) return;
      const dim = fs && !fs.has(id) ? "lv-dim" : "";
      const isSel = selL && selL.kind === "node" && selL.id === id ? "sel" : "";
      const gap = n.type === "Gap" ? "gap" : "";
      const imp = inE[id].find((e) => e._layer === "impact" && !blank(e.judge_verdict));
      const vc = imp ? vClass(imp.judge_verdict) : "";
      const vcol = { agree: "var(--ok)", disagree: "var(--bad)", uncertain: "var(--warn)" }[vc];
      const label = n.title && n.title !== id ? n.title : id;
      let more = "";
      if (mode === "step" && (KIDS[id] || []).length) {
        const k = (KIDS[id] || []).filter((x) => !vis.has(x)).length;
        const txt = k ? "+" + k : "−", w = txt.length * 7 + 12;
        more = `<g class="lv-more ${k ? "" : "done"}" data-more="${h(id)}" data-n="${k}">
          <rect x="${NODE_W - 22 - w}" y="${NODE_H / 2 - 9}" width="${w}" height="18" rx="9"/>
          <text x="${NODE_W - 22 - w / 2}" y="${NODE_H / 2 + 4}" text-anchor="middle">${txt}</text>
          <title>${k ? "Reveal " + k + " linked record(s)" : "Fold this branch"}</title></g>`;
      }
      g += `<g class="lv-node ${dim} ${isSel} ${gap}${anim(id)}" data-node="${h(id)}" transform="translate(${p.x},${p.y})">
        <title>${h(tLabel(n.type))}: ${h(id)}${n.title && n.title !== id ? "\n" + h(n.title) : ""}</title>
        <rect class="box" width="${NODE_W}" height="${NODE_H}" rx="6"/>
        <rect width="5" height="${NODE_H}" rx="2" fill="${tColor(n.type)}"/>
        <text x="12" y="${NODE_H / 2 + 4}" ${n.type === "Alert" ? 'font-weight="700"' : ""}>${h(short(label, more ? 23 : 30))}</text>
        ${vcol ? `<circle cx="${NODE_W - 10}" cy="${NODE_H / 2}" r="5" fill="${vcol}"><title>Alert impact: ${vText(imp.judge_verdict)}</title></circle>` : ""}
        ${more}
      </g>`;
    });

    $("lv-vp").innerHTML = g;
    fresh = new Set();
    stats(scope);
    toolsUi(vis);
  }

  function stats(scope) {
    const c = {};
    scope.forEach((id) => { const t = nodeById[id].type; c[t] = (c[t] || 0) + 1; });
    const imp = DATA.edges.filter((e) => e._layer === "impact" && scope.has(e.source) && scope.has(e.target));
    const v = { agree: 0, disagree: 0, uncertain: 0, "": 0 };
    imp.forEach((e) => { v[vClass(e.judge_verdict)]++; });
    const order = ["Alert", "RGL", "RSM", "ROB", "Policy", "Procedure", "LR", "L1C", "1C", "Gap"];
    const ctxN = [...scope].filter((id) => CTX_TYPES.has(nodeById[id].type)).length;
    $("lv-stats").innerHTML = (ctxN ? `<div class="stat"><b style="color:var(--context)">${ctxN}</b><span>Intake context tags</span></div>` : "") +
      order.filter((t) => c[t]).map((t) =>
        `<div class="stat"><b style="color:${tColor(t)}">${c[t]}</b><span>${h(tLabel(t))}</span></div>`).join("") +
      `<div class="stat wide"><b>${imp.length}</b><span>alert impact links ·
        <span style="color:var(--ok)">${v.agree} confirmed</span> ·
        <span style="color:var(--bad)">${v.disagree} disputed</span> ·
        <span style="color:var(--warn)">${v.uncertain} to SME</span> · ${v[""]} not judged</span></div>`;
  }

  function toolsUi(vis) {
    const step = mode === "step";
    ["both", "context", "structure", "impact"].forEach((l) => $("lv-l-" + l).setAttribute("aria-pressed", layer === l));
    journey();
    $("lv-m-step").setAttribute("aria-pressed", step);
    $("lv-m-all").setAttribute("aria-pressed", !step);
    $("lv-next").hidden = !step; $("lv-reset").hidden = !step;
    $("lv-next").disabled = ![...vis].some((id) => (KIDS[id] || []).some((k) => !vis.has(k)));
    const L = {
      both: "<b>Everything:</b> the alert's intake context, plus the ontology chain of everything it touches. The coloured dot on a box is the judge's verdict on the alert's impact on it.",
      context: "<b>Alert context:</b> what happens first, at intake: applicability, category and the ML tags (risk steward, theme, risk taxonomy L1–L3, line of business, legal entity, location).",
      structure: "<b>Ontology structure:</b> how the impacted records relate to each other in RegMap, GPPS and Helios, with no alert links drawn.",
      impact: "<b>Alert impact:</b> every record the alert touches directly, one line each, coloured by the judge's verdict.",
    }[layer];
    $("lv-hint").innerHTML = L + (step
      ? " &nbsp;·&nbsp; <b>Step by step:</b> click a box to reveal what it links to (red <b>+N</b>), <b>Next step</b> reveals one more layer, <b>−</b> folds a branch."
      : " &nbsp;·&nbsp; <b>Show all:</b> click a box to highlight its links.");
  }

  const relayout = () => { selL = null; $("lv-grid").classList.remove("open"); resetReveal(); render(); requestAnimationFrame(fit); };
  ["both", "context", "structure", "impact"].forEach((l) => { $("lv-l-" + l).onclick = () => { layer = l; relayout(); }; });

  /* The alert journey, left to right: what happens to an alert before and
     during impact assessment. Values are filled for the selected alert. */
  function journey() {
    const a = cur !== "ALL" ? nodeById[cur] : null, x = a && a.intake;
    const imp = a ? outE[a.id].filter((e) => e._layer === "impact") : DATA.edges.filter((e) => e._layer === "impact");
    const ok = imp.filter((e) => vClass(e.judge_verdict) === "agree").length;
    const ctx = a ? outE[a.id].filter((e) => e._layer === "context").length : 0;
    const steps = [
      ["1", "Alert received", a ? h((x && x.ref) || a.id) : `${DATA.nodes.filter((n) => n.type === "Alert").length} alerts`, "", "var(--impact)"],
      ["2", "Applicable to HSBC?", x ? h(x.applicable) : "Yes / No", "context", "var(--context)"],
      ["3", "Category", x ? h(x.category) + " · " + h(CATEGORY_TXT[x.category] || "") : "RD · Consultation · News · Fine", "context", "var(--context)"],
      ["4", "Contextualise (ML)", x ? `${ctx} tags: steward, theme, risk L1–L3, LOB, LE` : "steward, theme, risk L1–L3, LOB, LE", "context", "var(--context)"],
      ["5", "Impact assessment (AI)", `${imp.length} links · ${ok} judge-confirmed`, "impact", "var(--accent)"],
      ["6", "Ontology chain", "regulation → obligation → policy · risk → control", "structure", "var(--lineage)"],
      ["7", "SME review", "next: confirm or reject each link", "", "var(--muted)"],
    ];
    $("lv-journey").innerHTML = steps.map(([n, t, v, l, c], i) =>
      `${i ? '<span class="ja">→</span>' : ""}<div class="js ${l && l === layer ? "on" : ""}" data-layer="${l}" style="--kc:${c}">
        <span class="k">Step ${n}</span><b>${t}</b><span class="v">${v}</span></div>`).join("");
    $("lv-journey").querySelectorAll(".js").forEach((el) => {
      el.onclick = () => { if (el.dataset.layer) { layer = el.dataset.layer; relayout(); } };
    });
  }
  $("lv-m-step").onclick = () => { mode = "step"; relayout(); };
  $("lv-m-all").onclick  = () => { mode = "all"; render(); requestAnimationFrame(fit); };
  $("lv-next").onclick   = nextStep;
  $("lv-reset").onclick  = () => { selL = null; $("lv-grid").classList.remove("open"); resetReveal(); render(); };

  function clearSel() { selL = null; $("lv-grid").classList.remove("open"); render(); }

  const secT = (n, t, extra = "") => `<div class="pn-sec"><span class="n">${n}</span>${t}${extra}</div>`;

  // A record card: type chip, ID, title, its own text (clamped).
  function card(n, lines = 6) {
    const cls = T[n.type] ? T[n.type].cls : "";
    const desc = cls && byId[cls] ? byId[cls].d : "";
    const title = n.title && n.title !== n.id ? `<div class="pn-title">${h(n.title)}</div>` : "";
    const text = !blank(n.text)
      ? `<div class="pn-text clamp" style="--lines:${lines}">${h(n.text)}</div><button class="pn-more">Show more</button>`
      : `<div class="muted" style="font-size:13px">${h(desc)}</div>`;

      <div class="pn-head"><span class="pn-chip">${h(tShort(n.type))}</span><span class="tag">${h(n.id)}</span></div>
      ${title}${text}</div>`;
  }

  // The relation between two cards.
  function relBlock(e) {
    const [name, sub, col] = LAYER_TXT[e._layer];
    const sent = ontoSentence(e.relation), o = ontoRel(e.relation);
    return `<div class="pn-rel" style="--kc:${col}"><span class="arrow">↓</span>
      <span class="verb">${h(verb(e.relation))}</span>

      ${sent ? `<div class="pn-onto"><span class="pn-ql" style="--qc:var(--muted)">Ontology</span>${sent}

  // Evidence for an AI link: verdict, score, the two quotes, reasoning.
  function evidence(e) {
    if (e.kind === "DERIVED_LINK") return `<div class="pn-evi"><div class="pn-vrow"><span class="pn-rr">DERIVED · no AI</span></div>
      <div class="pn-vrow">Via confirmed control(s) <b style="color:var(--fg)">${h(e.via_controls || "")}</b></div>
      ${!blank(e.sections) ? `<div class="pn-vrow">Policy sections <b style="color:var(--fg)">${h(e.sections)}</b></div>` : ""}
      <div class="muted" style="font-size:12px">${h(e.reasoning || "The obligation is satisfied by a judge-confirmed control, and GPPS maps that control to this policy.")}</div></div>`;
    if (e.kind !== "AI_LINK") return "";
    const s = nodeById[e.source], t = nodeById[e.target];
    const vc = vClass(e.judge_verdict);
    const score = Number(e.score);
    let x = `<div class="pn-evi"><div class="pn-vrow">
      <span class="pn-verdict ${vc}">${vText(e.judge_verdict)}</span>
      ${!blank(e.score) && !isNaN(score) ? `<span class="pn-meter">Score <i><b style="width:${Math.max(0, Math.min(10, score)) * 10}%"></b></i><b style="color:var(--fg)">${h(e.score)}/10</b></span>` : ""}
      ${!blank(e.confidence) ? `<span>Confidence <b style="color:var(--fg)">${h(e.confidence)}</b></span>` : ""}
      ${!blank(e.grounding) ? `<span>Quotes <b style="color:var(--fg)">${h(e.grounding)}</b></span>` : ""}</div>`;
    if (!blank(e.impact_type) || !blank(e.coverage)) x += `<div class="pn-vrow">${
      String(e.impact_type).toUpperCase() === "CHANGE_REQUIRED" ? `<span class="pn-cr">CHANGE REQUIRED</span>`
      : !blank(e.impact_type) ? `<span class="pn-rr">${h(String(e.impact_type).replace(/_/g, " "))}</span>` : ""}${
      !blank(e.coverage) ? ` <span>Coverage <b style="color:var(--fg)">${h(e.coverage)}</b></span>` : ""}</div>`;
    if (!blank(e.change_needed)) x += `<div class="pn-q" style="--qc:var(--bad)"><span class="pn-ql">Change needed</span>${h(e.change_needed)}</div>`;
    if (!blank(e.sections)) x += `<div class="pn-vrow">Linked sections <b style="color:var(--fg)">${h(e.sections)}</b></div>`;
    if (!blank(e.evidence_from))
