/* relationship sets for a class (own + optionally inherited) */
function edgesFor(id, withInherited){
  const inh = withInherited ? ancestors(id) : [];
  const out=[], inn=[];
  [id,...inh].forEach(src=>D.rels.forEach(e=>{ if(!onK.has(e.k)) return;
    const via = src===id ? "" : src;
    if(e.f===src) out.push({...e,via,other:e.t});
    if(e.t===src && e.f!==src) inn.push({...e,via,other:e.f}); }));
  const k=e=>e.r+"|"+e.other, uniq=a=>a.filter((e,i)=>a.findIndex(x=>k(x)===k(e))===i);
  const rank={S:0,I:1,C:2,P:3}, sort=a=>a.sort((x,y)=>rank[x.k]-rank[y.k]||x.r.localeCompare(y.r));
  return {out:sort(uniq(out)), inn:sort(uniq(inn).filter(e=>e.other!=="Entity"))};
}
