def text_of(i):
    return f"{TITLE.get(i, '')}\n{TEXT.get(i, '')}".strip()


def clean_words(s):
    return re.sub(r"[^a-z0-9]+", " ", str(s or "").lower()).strip()


def quote_found(quote, text):
    q = clean_words(quote)
    if len(q) < 12 or q in ("null", "none"):
        return False
    t = clean_words(text)
    if q in t:
        return True
    w = q.split()
    return len(w) >= 6 and (" ".join(w[:8]) in t or " ".join(w[-8:]) in t)


def wilson(k, n, z=1.96):
    if n == 0:
        return (np.nan, np.nan)
    p, d = k / n, 1 + z * z / n
    c = (p + z * z / (2 * n)) / d
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return (max(0, c - h), min(1, c + h))


def save_text_safe(df, path):
    """Parquet-safe save (mixed bool / text judge columns -> text)."""
    out = df.copy()
    for c_ in out.columns:
        if out[c_].dtype == object:
            out[c_] = out[c_].map(lambda v: None if v is None else str(v))
    out.to_parquet(path, index=False)

# ====== CELL 6: RUNNER + FINALISATION (library) ===============================
def norm_id(x):
    return re.sub(r"\s+", "", str(x or "")).upper()


async def run_batches(batches, checkpoint_path, stage):
    global CURRENT_STAGE
    CURRENT_STAGE = stage
    rows = []
    if checkpoint_path and os.path.exists(checkpoint_path):
        prev = pd.read_parquet(checkpoint_path)
        rows = prev[prev["model_decision"] != "ERROR"].to_dict("records")
        print(f"[resume {stage}] {len(rows):,} pairs already judged")
    done = {(r["rob"], r["ic"]) for r in rows}
    todo = [{**b, "items": [it for it in b["items"] if (b["rob"], it["ic"]) not in done]} for b in batches]
    todo = [b for b in todo if b["items"]]
    print(f"[{stage}] {len(todo):,} calls to make")
    sem = asyncio.Semaphore(CONCURRENCY)

    async def one(b):
        async with sem:
            ans = await call_llm(PROMPT_IC, build_payload(b), None, MAX_TOKENS) or {}
        got = {norm_id(x.get("instance_id")): x for x in (ans.get("items") or []) if isinstance(x, dict)}
        out = []
        for it in b["items"]:
            x = got.get(norm_id(it["ic"]), {})
            d = str(x.get("decision", "ERROR")).upper().strip().replace(" ", "_") if x else "ERROR"
            out.append({"rob": b["rob"], "l1c": b["l1c"], "ic": it["ic"], "regmap_applicable": it["regmap_applicable"],
                        "business": it["business"], "legal_entity": it["legal_entity"],
                        "model_decision": d if d in ("LINKED", "NOT_LINKED", "INSUFFICIENT_EVIDENCE") else "ERROR",
                        "coverage": x.get("coverage"), "link_score": x.get("link_score"),
                        "confidence_level": x.get("confidence_level"), "reasoning": x.get("reasoning"),
                        "evidence_from_obligation": x.get("evidence_from_obligation"),
                        "evidence_from_instance": x.get("evidence_from_instance")})
        return out

    tasks = [asyncio.create_task(one(b)) for b in todo]
    n = 0
    for fut in tqdm(asyncio.as_completed(tasks), total=len(tasks), desc=stage):
        rows.extend(await fut)
        n += 1
        if checkpoint_path and n % SAVE_EVERY == 0:
            pd.DataFrame(rows).astype(str).to_parquet(checkpoint_path + ".tmp", index=False)
            os.replace(checkpoint_path + ".tmp", checkpoint_path)
    res = pd.DataFrame(rows)
    if checkpoint_path and len(res):
        res.astype(str).to_parquet(checkpoint_path + ".tmp", index=False)
        os.replace(checkpoint_path + ".tmp", checkpoint_path)
    for c_ in res.columns:
        res[c_] = res[c_].map(lambda v: None if v is None or str(v) in ("None", "nan", "") else v)
    for c_ in ("link_score", "confidence_level"):
        res[c_] = pd.to_numeric(res[c_], errors="coerce")
    res["regmap_applicable"] = res["regmap_applicable"].fillna("")
    print(f"[{stage}] {res['model_decision'].value_counts().to_dict()}")
    return res


def finalise(res):
    out = res.copy()
    ob_ok = [quote_found(q, text_of(r)) for q, r in zip(out["evidence_from_obligation"], out["rob"])]
    ic_ok = [quote_found(q, text_of(i)) for q, i in zip(out["evidence_from_instance"], out["ic"])]
    out["grounding_flag"] = np.select([np.array(ob_ok) & np.array(ic_ok), np.array(ob_ok) | np.array(ic_ok)],
                                      ["GROUNDED", "PARTIAL"], default="UNGROUNDED")
    linked = (out["model_decision"] == "LINKED") & (out["link_score"].fillna(0) >= LINK_THRESHOLD)
    out["final_decision"] = np.select(
        [out["model_decision"] == "ERROR", linked & (out["grounding_flag"] == "GROUNDED"), linked,
         out["model_decision"] == "INSUFFICIENT_EVIDENCE"],
        ["ERROR", "LINKED", "INSUFFICIENT_EVIDENCE", "INSUFFICIENT_EVIDENCE"], default="NOT_LINKED")
    return out

print("runner ready")

async def run_judge(audit, ckpt):
    global CURRENT_STAGE
    CURRENT_STAGE = "p3b_ic_audit"
    key = lambda df: list(zip(df["rob"], df["ic"], df["bucket"]))
    rows, done = [], set()
    if os.path.exists(ckpt):
        prev = pd.read_parquet(ckpt)
        prev = prev[prev["audit_verdict"] != "ERROR"]
        rows, done = prev.to_dict("records"), set(key(prev))
        print(f"[resume] {len(done):,} audits already done")
    todo = audit[[k not in done for k in key(audit)]]
    print(f"[audit] {len(todo):,} calls to make")
    sem = asyncio.Semaphore(CONCURRENCY)

    async def one(r):
        payload = {"obligation_text": text_of(r.rob),
                   "library_control": text_of(r.l1c),
                   "instance_id": r.ic, "business": r.business, "legal_entity": r.legal_entity,
                   "instance_text": text_of(r.ic),
                   "model_output": {"decision": r.model_decision, "coverage": r.coverage,
                                    "link_score": None if pd.isna(r.link_score) else float(r.link_score),
                                    "reasoning": r.reasoning, "evidence_from_obligation": r.evidence_from_obligation,
                                    "evidence_from_instance": r.evidence_from_instance}}
        async with sem:
            a = await call_llm(JUDGE_IC, payload, None, MAX_TOKENS) or {}
        v = str(a.get("verdict", "ERROR")).upper().strip() if a else "ERROR"
        return {"rob": r.rob, "ic": r.ic, "bucket": r.bucket,
                "audit_verdict": v if v in ("AGREE", "DISAGREE", "UNCERTAIN") else "ERROR",
                "audit_error_type": a.get("error_type"), "audit_correct_decision": a.get("correct_decision"),
                "audit_obligation_quote": a.get("obligation_quote"), "audit_instance_quote": a.get("instance_quote"),
                "audit_confidence_level": a.get("confidence_level"), "audit_rationale": a.get("rationale")}

    tasks = [asyncio.create_task(one(r)) for r in todo.itertuples()]
    n = 0
    for fut in tqdm(asyncio.as_completed(tasks), total=len(tasks), desc="p3b ic audit"):
        rows.append(await fut)
        n += 1
        if n % 50 == 0:
            pd.DataFrame(rows).astype(str).to_parquet(ckpt + ".tmp", index=False)
            os.replace(ckpt + ".tmp", ckpt)
    j = pd.DataFrame(rows)
    j.astype(str).to_parquet(ckpt + ".tmp", index=False)
    os.replace(ckpt + ".tmp", ckpt)
    for c_ in j.columns:
        j[c_] = j[c_].map(lambda v: None if v is None or str(v) in ("None", "nan") else v)
    print("[audit]", j["audit_verdict"].value_counts().to_dict())
    return j


judged = await run_judge(audit, f"{P3B_DIR}/ic_audit_ckpt.parquet")
